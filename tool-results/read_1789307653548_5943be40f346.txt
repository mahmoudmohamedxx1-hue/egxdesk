   106→function sanitizeTable(raw: unknown): { columns: string[]; rows: Cell[][] } {
   107→  if (raw === null || typeof raw !== "object") return { columns: [], rows: [] };
   108→  const obj = raw as { columns?: unknown; rows?: unknown };
   109→  const columns = Array.isArray(obj.columns)
   110→    ? obj.columns.filter((c): c is string => typeof c === "string").map((c) => c.slice(0, 80)).slice(0, 30)
   111→    : [];
   112→  const rows = Array.isArray(obj.rows)
   113→    ? obj.rows
   114→        .slice(0, 600)
   115→        .filter((r): r is unknown[] => Array.isArray(r))
   116→        .map((r) =>
   117→          r
   118→            .slice(0, 30)
   119→            .map((c): Cell =>
   120→              typeof c === "number" && Number.isFinite(c)
   121→                ? c
   122→                : typeof c === "string"
   123→                  ? c.slice(0, 300)
   124→                  : null
   125→            )
   126→        )
   127→        .filter((r) => r.length > 0)
   128→    : [];
   129→  return { columns: columns.length ? columns : (rows[0]?.map((_, i) => `#${i + 1}`) ?? []), rows };
   130→}

   132→/** numeric-column detection for client tables → tabular number formats */
   133→function autoFmtFor(rows: Cell[][], colIndex: number): ColFmt {
   134→  let nums = 0;
   135→  let total = 0;
   136→  for (const r of rows.slice(0, 200)) {
   137→    const v = r[colIndex];
   138→    if (v === null || v === "") continue;
   139→    total++;
   140→    if (typeof v === "number") nums++;
   141→  }
   142→  if (total === 0 || nums / total < 0.7) return "text";
   143→  const allInt = rows.every((r) => {
   144→    const v = r[colIndex];
   145→    return v === null || v === "" || (typeof v === "number" && Number.isInteger(v));
   146→  });
   147→  return allInt ? "int" : "num";
   148→}
