   233→def add_heading(text, style, level=0):

   236→    p.bookmark_name = key

   252→def callout_row(stats):
   253→    """stats: list of (big, label) — renders a centered 3-up callout strip."""
   254→    cells, widths = [], []
   255→    n = len(stats)
   256→    gap = 10
   257→    cw = (AVAIL_W - gap * (n - 1)) / n
   258→    row = []
   259→    for i, (big, lab) in enumerate(stats):
   260→        inner = Table([[Paragraph(f"<b>{big}</b>", S_STAT)],
   261→                       [Paragraph(lab, S_STATLB)]], colWidths=[cw])
   262→        inner.setStyle(TableStyle([
   263→            ("BACKGROUND", (0, 0), (-1, -1), CARD_BG),
   264→            ("BOX", (0, 0), (-1, -1), 0.8, BORDER),
   265→            ("TOPPADDING", (0, 0), (-1, 0), 9), ("BOTTOMPADDING", (0, 1), (-1, 1), 9),
   266→            ("TOPPADDING", (0, 1), (-1, 1), 2), ("BOTTOMPADDING", (0, 0), (-1, 0), 2),
   267→            ("VALIGN", (0, 0), (-1, -1), "MIDDLE"),
   268→        ]))
   269→        row.append(inner)
   270→        widths.append(cw)
   271→        if i < n - 1:
   272→            row.append(Spacer(gap, 1))
   273→            widths.append(gap)
   274→    outer = Table([row], colWidths=widths, hAlign="CENTER")
   275→    outer.setStyle(TableStyle([
   276→        ("VALIGN", (0, 0), (-1, -1), "MIDDLE"),
   277→        ("LEFTPADDING", (0, 0), (-1, -1), 0), ("RIGHTPADDING", (0, 0), (-1, -1), 0),
   278→        ("TOPPADDING", (0, 0), (-1, -1), 0), ("BOTTOMPADDING", (0, 0), (-1, -1), 0),
   279→    ]))
   280→    return outer

   282→def bench_table(cols, cap_start, cap_end):
   283→    """Benchmark matrix for selected tool columns between capability rows."""
   284→    sym = {"Y": "\u2713", "P": "part.", "N": "\u2014"}
   285→    head = [Paragraph("<b>Capability</b>", S_TH)] + \
   286→           [Paragraph(f"<b>{t}</b>", S_TH) for t in cols]
   287→    data = [head]
   288→    for row in MATRIX[cap_start:cap_end]:
   289→        label = row[0]
   290→        vals = [row[1 + TOOLS.index(t)] for t in cols]
   291→        r = [Paragraph(label, S_TDL)] + [Paragraph(sym[v], S_TD) for v in vals]
   292→        data.append(r)
   293→    ratios = [0.30] + [0.70 / len(cols)] * len(cols)
   294→    widths = [r * AVAIL_W for r in ratios]
   295→    t = Table(data, colWidths=widths, hAlign="CENTER", repeatRows=1)
   296→    style = [
   297→        ("BACKGROUND", (0, 0), (-1, 0), TABLE_HEADER_COLOR),
   298→        ("VALIGN", (0, 0), (-1, -1), "MIDDLE"),
   299→        ("GRID", (0, 0), (-1, -1), 0.4, BORDER),
   300→        ("LEFTPADDING", (0, 0), (-1, -1), 5), ("RIGHTPADDING", (0, 0), (-1, -1), 5),
   301→        ("TOPPADDING", (0, 0), (-1, -1), 3.5), ("BOTTOMPADDING", (0, 0), (-1, -1), 3.5),
   302→    ]
   303→    for i in range(1, len(data)):
   304→        style.append(("BACKGROUND", (0, i), (-1, i),
   305→                      TABLE_ROW_ODD if i % 2 else TABLE_ROW_EVEN))
   306→    # highlight EGX Desk column
   307→    if "EGX Desk" in cols:
   308→        c = cols.index("EGX Desk") + 1
   309→        style.append(("BACKGROUND", (c, 1), (c, -1), CARD_BG))
   310→    t.setStyle(TableStyle(style))
   311→    return t
