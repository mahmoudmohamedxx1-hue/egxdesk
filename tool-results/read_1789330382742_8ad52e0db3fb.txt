    57→/** The terracotta 12-ray sunburst — the assistant's avatar mark. */
    58→function SunburstMark({ className }: { className?: string }) {
    59→  const rays = Array.from({ length: 12 }, (_, i) => i * 30);
    60→  return (
    61→    <svg viewBox="0 0 24 24" fill="none" className={className} aria-hidden>
    62→      {rays.map((a) => {
    63→        const rad = (a * Math.PI) / 180;
    64→        return (
    65→          <line key={a} x1={12 + Math.sin(rad) * 3.4} y1={12 - Math.cos(rad) * 3.4}
    66→            x2={12 + Math.sin(rad) * 10.2} y2={12 - Math.cos(rad) * 10.2}
    67→            stroke="currentColor" strokeWidth="2.2" strokeLinecap="round" />
    68→        );
    69→      })}
    70→    </svg>
    71→  );
    72→}

   109→  const [focused, setFocused] = useState(false);
   110→  const [mounted, setMounted] = useState(false);
   111→
   112→  // model layer state
   113→  const [modelId, setModelId] = useState<string>(MODEL_INSTANT);
   114→  const [engineStatus, setEngineStatus] = useState<EngineStatus>({ phase: "idle" });
   115→  const [webgpuOk, setWebgpuOk] = useState<boolean | null>(null);
   116→  const [registry, setRegistry] = useState<RegistryModel[] | null>(null);
   117→  const [showAll, setShowAll] = useState(false);
   118→  const [modelQuery, setModelQuery] = useState("");

   120→  // mic

   122→  const recogRef = useRef<unknown>(null);

   138→      if (m) setModelId(m);
   139→    } catch {}
   140→    const off = onEngineStatus((s) => setEngineStatus(s));
   141→    setEngineStatus(currentEngineStatus());

   144→      off();

   156→      }

   167→    }
   168→  }, [open]);
   169→
   170→  // keep the transcript pinned to the newest content
   171→  useEffect(() => {
   172→    const el = scrollRef.current;
   173→    if (el) el.scrollTop = el.scrollHeight;

   185→
   186→  const persist = useCallback((next: Msg[]) => {
   187→    try {
   188→      localStorage.setItem(CHAT_KEY, JSON.stringify(next.slice(-MAX_STORED)));

   190→  }, []);
   191→
   192→  const pushMsg = useCallback((m: Msg) => {
   193→    setMsgs((prev) => {
   194→      const next = [...prev, m];
   195→      persist(next);
   196→      return next;
   197→    });

   199→
   200→  const patchLast = useCallback((patch: Partial<Msg>) => {
   201→    setMsgs((prev) => {
   202→      if (!prev.length) return prev;

   206→      return next;

   223→

   228→    return res;
   229→  }, [toolCtx]);
   230→
   231→  // ── the full agent loop ──
   232→  const ask = useCallback(async (rawText: string) => {
   233→    const text = rawText.trim();

   235→    setInput("");

   238→    setMsgs(history);
   239→    persist(history);
   240→
   241→    // ── INSTANT mode: zero-model fast path ──
   242→    if (modelId === MODEL_INSTANT) {
   243→      const hit = instantRoute(text);
   244→      if (hit) {
   245→        const res = await executeTool(hit.tool, hit.args);
   246→        pushMsg({
   247→          role: "assistant",
   248→          text: res.text,
   249→          steps: [{ tool: hit.tool, ok: res.ok }],
   250→          ts: Date.now(),
   251→          ...(res.ok ? {} : { error: true }),
   252→        });
   253→      } else {
   254→        pushMsg({
   255→          role: "assistant",
   256→          text: `**${tt(T.aiInstantFallbackTitle, lang)}**\n\n${tt(T.aiInstantFallbackBody, lang)}`,
   257→          ts: Date.now(),
   258→        });
   259→      }
   260→      return;
   261→    }
   262→
   263→    // ── CLOUD mode (GLM via /api/assistant) ──
   264→    if (modelId === MODEL_CLOUD) {
   265→      const ac = new AbortController();
   266→      abortRef.current = ac;
   267→      setBusy("think");
   268→      try {
   269→        const res = await fetch("/api/assistant", {
   270→          method: "POST",
   271→          headers: { "content-type": "application/json" },
   272→          signal: ac.signal,
   273→          body: JSON.stringify({
   274→            stage: "plan",
   275→            lang,
   276→            context: { view: view.name, ticker: view.ticker },
   277→            messages: history.slice(-8).map((m) => ({ role: m.role, content: m.text })),
   278→          }),
   279→        });
   280→        if (!res.ok) throw new Error(`HTTP ${res.status}`);
   281→        const j = (await res.json()) as { tool?: string; args?: Record<string, unknown>; reply?: string; error?: string };
   282→        if (j.tool && j.args != null) {
   283→          const toolRes = await executeTool(j.tool, j.args);
   284→          setBusy("answer");
   285→          let answered = false;
   286→          try {
   287→            const ac2 = new AbortController();
   288→            abortRef.current = ac2;
   289→            const res2 = await fetch("/api/assistant", {
   290→              method: "POST",
   291→              headers: { "content-type": "application/json" },
   292→              signal: ac2.signal,
   293→              body: JSON.stringify({
   294→                stage: "answer",
   295→                lang,
   296→                question: text,
   297→                tool: j.tool,
   298→                args: j.args,
   299→                result: toolRes.data ?? { ok: toolRes.ok, text: toolRes.text },
   300→              }),
   301→            });
   302→            if (res2.ok) {
   303→              const j2 = (await res2.json()) as { reply?: string; error?: string };
   304→              if (j2.reply && j2.reply.trim()) {
   305→                pushMsg({ role: "assistant", text: j2.reply.trim(), steps: [{ tool: j.tool, ok: toolRes.ok }], ts: Date.now() });
   306→                answered = true;
   307→              }
   308→            }
   309→          } catch {}
   310→          if (!answered) {
   311→            pushMsg({ role: "assistant", text: toolRes.text, steps: [{ tool: j.tool, ok: toolRes.ok }], ts: Date.now() });
   312→          }
   313→        } else if (j.reply && j.reply.trim()) {
   314→          pushMsg({ role: "assistant", text: j.reply.trim(), ts: Date.now() });
   315→        } else {
   316→          throw new Error(j.error ?? "no action");
   317→        }
   318→      } catch (err) {
   319→        const aborted = err instanceof DOMException && err.name === "AbortError";
   320→        pushMsg({
   321→          role: "assistant",
   322→          text: aborted ? `⏹ ${tt(T.aiInputStop, lang)}` : tt(T.aiErrorGeneric, lang),
   323→          ts: Date.now(),
   324→          ...(aborted ? {} : { error: true }),
   325→        });
   326→      } finally {
   327→        abortRef.current = null;
   328→        setBusy(null);
   329→      }
   330→      return;
   331→    }
   332→
   333→    // ── ON-DEVICE mode (WebLLM) ──
   334→    setBusy("think");
   335→    try {
   336→      const planMsgs = [
   337→        { role: "system" as const, content: planSystemPrompt(lang, view.name, view.ticker) },
   338→        ...history.slice(-4).map((m) => ({ role: m.role as "user" | "assistant", content: m.text.slice(0, 1200) })),
   339→      ];
   340→      const out = await engineChat(modelId, planMsgs);
   341→      const parsed = parseToolJson(out);
   342→      if (parsed?.tool) {
   343→        const toolRes = await executeTool(parsed.tool, parsed.args ?? {});
   344→        setBusy("answer");
   345→        pushMsg({ role: "assistant", text: "", steps: [{ tool: parsed.tool, ok: toolRes.ok }], ts: Date.now() });
   346→        const ansMsgs = [
   347→          { role: "system" as const, content: answerSystemPrompt(lang, parsed.tool, parsed.args ?? {}, toolRes.data ?? { ok: toolRes.ok, text: toolRes.text }, text) },
   348→        ];
   349→        const final = await engineChat(modelId, ansMsgs, (full) => patchLast({ text: full }));
   350→        patchLast({ text: final.trim() || toolRes.text });
   351→      } else if (parsed?.reply) {
   352→        pushMsg({ role: "assistant", text: parsed.reply.trim(), ts: Date.now() });
   353→      } else {
   354→        // the small model failed JSON — fall back to the instant router
   355→        const hit = instantRoute(text);
   356→        if (hit) {
   357→          const res = await executeTool(hit.tool, hit.args);
   358→          pushMsg({ role: "assistant", text: res.text, steps: [{ tool: hit.tool, ok: res.ok }], ts: Date.now() });
   359→        } else {
   360→          pushMsg({ role: "assistant", text: out.trim() || tt(T.aiErrorGeneric, lang), ts: Date.now() });
   361→        }
   362→      }
   363→    } catch (err) {
   364→      const msg = err instanceof Error ? err.message : "error";
   365→      const cancelled = /cancel/i.test(msg);
   366→      pushMsg({
   367→        role: "assistant",
   368→        text: cancelled
   369→          ? `⏹ ${tt(T.aiInputStop, lang)}`
   370→          : `${tt(T.aiErrorGeneric, lang)}${engineStatus.phase === "error" ? ` — ${engineStatus.message.slice(0, 90)}` : ""}`,
   371→        ts: Date.now(),
   372→        ...(cancelled ? {} : { error: true }),
   373→      });
   374→    } finally {
   375→      setBusy(null);
   376→    }
   377→  }, [busy, modelId, lang, view, msgs, executeTool, pushMsg, patchLast, engineStatus]);

   388→      localStorage.setItem(MODEL_KEY, id);

   392→        toast(tt(T.aiNoWebGPU, lang));
   393→        return;
   394→      }

   396→    } else {
   397→      // leaving on-device mode frees the VRAM

   399→    }
   400→  }, [webgpuOk, lang, toast]);

   408→  }, [menuOpen, registry]);

   410→  // ── mic (Web Speech API — free, client-based; Chrome/Edge/Safari) ──

   412→    ("SpeechRecognition" in window || "webkitSpeechRecognition" in window);
   413→
   414→  const toggleMic = useCallback(() => {
   415→    if (!srSupported) return;
   416→    type SR = { lang: string; interimResults: boolean; continuous: boolean; start: () => void; stop: () => void; onresult: ((e: { results?: ArrayLike<ArrayLike<{ transcript?: string }>> }) => void) | null; onend: (() => void) | null; onerror: (() => void) | null };
   417→    const w = window as unknown as { SpeechRecognition?: new () => SR; webkitSpeechRecognition?: new () => SR };
   418→    const Ctor = w.SpeechRecognition ?? w.webkitSpeechRecognition;
   419→    if (!Ctor) return;

   421→      (recogRef.current as SR | null)?.stop();
   422→      return;
   423→    }
   424→    const rec = new Ctor();
   425→    rec.lang = lang === "ar" ? "ar-EG" : "en-US";

   427→    rec.continuous = false;

   431→    };
   432→    rec.onend = () => setListening(false);
   433→    rec.onerror = () => setListening(false);
   434→    recogRef.current = rec;
   435→    setListening(true);
   436→    try {
   437→      rec.start();
   438→    } catch {
   439→      setListening(false);
   440→    }
   441→  }, [srSupported, listening, lang]);
   442→
   443→  const canSend = !busy && input.trim().length > 0;
   444→  const ringOn = focused || busy != null;
   445→  const chipLabel = modelChipLabel(modelId);
   446→  const registryFiltered = (registry ?? []).filter((m) =>
   447→    !modelQuery || m.id.toLowerCase().includes(modelQuery.toLowerCase())
   448→  );
   449→
   450→  // one progress row shared by featured + registry views
   451→  const progressFor = (id: string): { progress: number; text: string } | null => {

   453→      return { progress: engineStatus.progress, text: engineStatus.text };

   462→        <button
   463→          onClick={() => setOpen(true)}
   464→          aria-label={tt(T.aiAssistOpen, lang)}
   465→          title={tt(T.aiAssistOpen, lang)}
   466→          className="ai-orb fixed z-40 bottom-[calc(env(safe-area-inset-bottom)+1.25rem)] end-5 flex h-12 w-12 items-center justify-center rounded-full transition-transform hover:scale-105 active:scale-95"
   467→        >
   468→          <Sparkles className="h-5 w-5" aria-hidden />
   469→        </button>
   470→      )}
   471→
   472→      <AnimatePresence>
   473→        {open && (
   474→          <motion.div
   475→            initial={{ opacity: 0, y: 24, scale: 0.97 }}
   476→            animate={{ opacity: 1, y: 0, scale: 1 }}
   477→            exit={{ opacity: 0, y: 16, scale: 0.97 }}

   479→            role="dialog"
   480→            aria-label={tt(T.aiAssistTitle, lang)}
   481→            className="fixed z-50 flex flex-col overflow-hidden rounded-2xl border bg-card shadow-2xl
   482→              inset-x-3 top-14 bottom-3
   483→              sm:inset-auto sm:bottom-[calc(env(safe-area-inset-bottom)+1.5rem)] sm:end-6 sm:top-auto

   488→              <span className="flex h-8 w-8 items-center justify-center rounded-full" style={{ backgroundColor: "var(--chat-accent-soft)", color: "var(--chat-accent)" }}>
   489→                <Bot className="h-4.5 w-4.5" aria-hidden />
