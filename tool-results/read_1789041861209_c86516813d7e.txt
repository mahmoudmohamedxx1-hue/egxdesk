    22→ *

    29→// ── rate limiting (in-memory, per IP — a personal tool, not a public API) ──

    31→const RATE_LIMIT = 30; // requests per hour
    32→const rateMap = new Map<string, number[]>();
    33→
    34→function rateLimited(ip: string): boolean {
    35→  const now = Date.now();
    36→  const window = 60 * 60_000;
    37→  const arr = (rateMap.get(ip) ?? []).filter((t) => now - t < window);
    38→  if (arr.length >= RATE_LIMIT) {
    39→    rateMap.set(ip, arr);
    40→    return true;
    41→  }
    42→  arr.push(now);
    43→  rateMap.set(ip, arr);
    44→  if (rateMap.size > 500) {
    45→    // drop stale entries so the map can never grow unbounded
    46→    for (const [k, v] of rateMap) if (v.every((t) => now - t >= window)) rateMap.delete(k);
    47→  }
    48→  return false;
    49→}

    77→function clampLimit(raw: unknown, max: number, dflt: number): number {
    78→  const n = typeof raw === "number" ? Math.floor(raw) : Number(raw);
    79→  if (!Number.isFinite(n) || n <= 0) return dflt;
    80→  return Math.min(n, max);
    81→}

    83→/** First balanced JSON object in the model output (handles ```json fences),
    84→ *  with two repair passes for common LLM quirks: control characters (raw
    85→ *  newlines) inside string literals, and replies truncated mid-string (no
    86→ *  closing brace) — the tail is then recovered with a regex. */
    87→function tryParse(s: string): Record<string, unknown> | null {
    88→  try {
    89→    const p = JSON.parse(s);
    90→    return p && typeof p === "object" ? (p as Record<string, unknown>) : null;
    91→  } catch {
    92→    return null;
    93→  }
    94→}
    95→
    96→/** Escape raw \n \r \t that appear INSIDE string literals (invalid JSON that
    97→ *  models sometimes emit); whitespace outside strings is left alone. */
    98→function repairControlChars(s: string): string {
    99→  let out = "";
   100→  let inStr = false;
   101→  let esc = false;
   102→  for (const ch of s) {
   103→    if (esc) {
   104→      out += ch;
   105→      esc = false;
   106→      continue;
   107→    }
   108→    if (ch === "\\") {
   109→      out += ch;
   110→      esc = true;
   111→      continue;
   112→    }
   113→    if (ch === '"') {
   114→      inStr = !inStr;
   115→      out += ch;
   116→      continue;
   117→    }
   118→    if (inStr && (ch === "\n" || ch === "\r" || ch === "\t")) {
   119→      out += "\\n";
   120→      continue;
   121→    }
   122→    out += ch;
   123→  }
   124→  return out;
   125→}

   127→function extractJson(raw: string): Record<string, unknown> | null {
   128→  if (!raw) return null;
   129→  let s = raw.trim();
   130→  const fence = s.match(/```(?:json)?\s*([\s\S]*?)```/i);
   131→  if (fence) s = fence[1].trim();
   132→
   133→  // 1) balanced-brace scan → parse → repaired parse
   134→  const start = s.indexOf("{");
   135→  if (start !== -1) {
   136→    let depth = 0;
   137→    let inStr = false;
   138→    let esc = false;
   139→    for (let i = start; i < s.length; i++) {
   140→      const c = s[i];
   141→      if (esc) {
   142→        esc = false;
   143→        continue;
   144→      }
   145→      if (c === "\\") {
   146→        if (inStr) esc = true;
   147→        continue;
   148→      }
   149→      if (c === '"') {
   150→        inStr = !inStr;
   151→        continue;
   152→      }
   153→      if (inStr) continue;
   154→      if (c === "{") depth++;
   155→      else if (c === "}") {
   156→        depth--;
   157→        if (depth === 0) {
   158→          const span = s.slice(start, i + 1);
   159→          return tryParse(span) ?? tryParse(repairControlChars(span));
   160→        }
   161→      }
   162→    }
   163→  }
   164→
   165→  // 2) truncated reply (no closing brace): recover {"final": "… from the tail
   166→  const mFinal = s.match(/"final"\s*:\s*"([\s\S]*)/);
   167→  if (mFinal) {
   168→    const tail = mFinal[1].replace(/"\s*\}?\s*$/, "").trim();
   169→    if (tail.length > 0) return { final: tail };
   170→  }
   171→
   172→  // 3) a tool call that survived truncation: {"tool": "name"…
   173→  const mTool = s.match(/"tool"\s*:\s*"([A-Za-z_][A-Za-z0-9_]*)"/);
   174→  if (mTool) {
   175→    const mArgs = s.match(/"args"\s*:\s*(\{[\s\S]*?)[}]?\s*$/);
   176→    let args: Record<string, unknown> = {};
   177→    if (mArgs) {
   178→      const repaired = repairControlChars(`${mArgs[1]}}`);
   179→      const parsed = tryParse(repaired);
   180→      if (parsed) args = parsed;
   181→    }
   182→    return { tool: mTool[1], args };
   183→  }
   184→  return null;
   185→}

   541→export async function POST(req: Request) {
   542→  const ip =
   543→    req.headers.get("x-forwarded-for")?.split(",")[0]?.trim() ||
   544→    req.headers.get("x-real-ip") ||
   545→    "local";
   546→  if (rateLimited(ip)) {
   547→    return NextResponse.json({ error: "rate limited — try again later" }, { status: 429 });
   548→  }
   549→
   550→  // validate the conversation the client sends (last 12 turns, last is user)
   551→  let body: { messages?: unknown; lang?: unknown; debug?: unknown };
   552→  try {
   553→    body = (await req.json()) as { messages?: unknown; lang?: unknown };
   554→  } catch {
   555→    return NextResponse.json({ error: "invalid json" }, { status: 400 });
   556→  }
   557→  const lang = body.lang === "en" ? "en" : "ar";
   558→  const rawMsgs = Array.isArray(body.messages) ? body.messages : [];
   559→  const history: ChatMsg[] = rawMsgs
   560→    .filter(
   561→      (m): m is ChatMsg =>
   562→        m !== null &&
   563→        typeof m === "object" &&
   564→        ((m as ChatMsg).role === "user" || (m as ChatMsg).role === "assistant") &&
   565→        typeof (m as ChatMsg).content === "string" &&
   566→        (m as ChatMsg).content.length > 0 &&
   567→        (m as ChatMsg).content.length <= 4000
   568→    )
   569→    .slice(-12);
   570→  if (history.length === 0 || history[history.length - 1].role !== "user") {
   571→    return NextResponse.json({ error: "messages required (last must be user)" }, { status: 400 });
   572→  }
   573→
   574→  const zai = await getZai();
   575→  const msgs: { role: "user" | "assistant"; content: string }[] = [
   576→    { role: "assistant", content: systemPrompt(lang) },
   577→    ...history.map((m) => ({ role: m.role, content: m.content })),
   578→  ];
   579→  const steps: AgentStep[] = [];
   580→  let corrections = 0;
   581→  const debugRaw: string[] = [];
   582→  const debug = body.debug === true;
   583→
   584→  for (let round = 0; round < MAX_TOOL_CALLS + 3; round++) {
   585→    if (steps.length >= MAX_TOOL_CALLS) break;
   586→    let raw = "";
   587→    try {
   588→      const completion = await zai.chat.completions.create({
   589→        messages: msgs,
   590→        thinking: { type: "disabled" },
   591→      });
   592→      raw = completion.choices[0]?.message?.content ?? "";
   593→    } catch (err) {
   594→      return NextResponse.json(
   595→        { error: "model unavailable", detail: err instanceof Error ? err.message : "unknown" },
   596→        { status: 502 }
   597→      );
   598→    }
   599→    if (debug) debugRaw.push(raw.slice(0, 800));
   600→
   601→    const parsed = extractJson(raw);
   602→    if (!parsed || (!("tool" in parsed) && !("final" in parsed))) {
   603→      corrections++;
   604→      if (corrections > 2) break;
   605→      msgs.push({
   606→        role: "user",
   607→        content:
   608→          'Format error. Reply with exactly ONE JSON object, no fences: {"tool": "<name>", "args": {...}} to call a tool, or {"final": "<markdown answer>"} to answer.',
   609→      });
   610→      continue;
   611→    }
   612→
   613→    if (typeof parsed.final === "string" && parsed.final.trim().length > 0) {
   614→      return NextResponse.json(
   615→        { answer: parsed.final.trim(), steps, disclaimer: true, ...(debug ? { debugRaw } : {}) },
   616→        { headers: { "Cache-Control": "no-store" } }
   617→      );
   618→    }
   619→
   620→    const toolName = typeof parsed.tool === "string" ? parsed.tool : "";
   621→    const tool = TOOLS.find((t) => t.name === toolName);
   622→    if (!tool) {
   623→      msgs.push({
   624→        role: "user",
   625→        content: `Unknown tool "${toolName}". Available tools: ${TOOLS.map((t) => t.name).join(", ")}.`,
   626→      });
   627→      continue;
   628→    }
   629→
   630→    const args = (parsed.args && typeof parsed.args === "object" ? parsed.args : {}) as Record<string, unknown>;
   631→    let result: unknown;
   632→    try {
   633→      result = await tool.run(args);
   634→    } catch (err) {
   635→      result = { error: err instanceof Error ? err.message : "tool failed" };
   636→    }
   637→    const ok = !(result && typeof result === "object" && "error" in (result as Record<string, unknown>));
   638→    steps.push({ tool: toolName, args, ok });
   639→
   640→    msgs.push({ role: "user", content: JSON.stringify(result).slice(0, 6000) });
   641→  }
   642→
   643→  // loop exhausted without a final answer — honest fallback, never invented
   644→  const fallback =
   645→    lang === "ar"
   646→      ? "وصلتُ لحد الأدوات المتاحة دون إجابة كاملة. جرّب إعادة السؤال بصيغة أبسط (مثال: «ما حالة السوق الآن؟» أو «quote لسهم COMI»)."
   647→      : "I ran out of tool budget without a complete answer. Try rephrasing (e.g. \"market overview\" or \"quote for COMI\").";
   648→  return NextResponse.json(
   649→    { answer: fallback, steps, disclaimer: true, ...(debug ? { debugRaw } : {}) },
   650→    { headers: { "Cache-Control": "no-store" } }
   651→  );
   652→}
