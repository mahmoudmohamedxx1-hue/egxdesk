    92→/** The terracotta asterisk mark — an original 12-ray sunburst that plays the
    93→ *  role of the assistant's avatar (inspired by reference AI chats, not a
    94→ *  copy of any logo). */
    95→function ClaudeMark({ className }: { className?: string }) {
    96→  const rays = Array.from({ length: 12 }, (_, i) => i * 30);
    97→  return (
    98→    <svg viewBox="0 0 24 24" fill="none" className={className} aria-hidden>
    99→      {rays.map((a) => {
   100→        const rad = (a * Math.PI) / 180;
   101→        const x1 = 12 + Math.sin(rad) * 3.4;
   102→        const y1 = 12 - Math.cos(rad) * 3.4;
   103→        const x2 = 12 + Math.sin(rad) * 10.2;
   104→        const y2 = 12 - Math.cos(rad) * 10.2;
   105→        return <line key={a} x1={x1} y1={y1} x2={x2} y2={y2} stroke="currentColor" strokeWidth="2.4" strokeLinecap="round" />;
   106→      })}
   107→    </svg>
   108→  );
   109→}

   202→  const [ready, setReady] = useState(false);
   203→  const [input, setInput] = useState("");
   204→  const [busy, setBusy] = useState(false);
   205→  const [deep, setDeep] = useState(false);
   206→  const [elapsed, setElapsed] = useState(0);
   207→  const [liveSteps, setLiveSteps] = useState<AgentStep[]>([]);
   208→  const [liveNote, setLiveNote] = useState<string | null>(null);
   209→  const [streamText, setStreamText] = useState("");
   210→  const [historyOpen, setHistoryOpen] = useState(false);
   211→  const [historyLoading, setHistoryLoading] = useState(false);
   212→  const [historyList, setHistoryList] = useState<HistoryRow[]>([]);
   213→  const [usage, setUsage] = useState<UsageSummary | null>(null);
   214→  const chatIdRef = useRef<string>("");
   215→  const lastQueryRef = useRef<string | null>(null);
   216→  const bottomRef = useRef<HTMLDivElement | null>(null);
   217→  const abortRef = useRef<AbortController | null>(null);
   218→
   219→  // restore the chat from the device (SSR-safe mount read); a shared
   220→  // ?q=… link prefills the composer (never auto-sends — it would burn the
   221→  // recipient's AI quota without their consent)
   222→  useEffect(() => {
   223→    try {
   224→      const raw = localStorage.getItem(CHAT_KEY);
   225→      if (raw) {
   226→        const arr = JSON.parse(raw);
   227→        if (Array.isArray(arr)) {
   228→          setMessages(
   229→            arr
   230→              .filter((m): m is AgentMsg => m && typeof m === "object" && (m.role === "user" || m.role === "assistant") && typeof m.content === "string")
   231→              .slice(-MAX_STORED)
   232→          );
   233→        }
   234→      }
   235→      let id = localStorage.getItem(CHAT_ID_KEY);
   236→      if (!id) {
   237→        id = newChatId();
   238→        localStorage.setItem(CHAT_ID_KEY, id);
   239→      }
   240→      chatIdRef.current = id;
   241→      setDeep(localStorage.getItem(THINK_KEY) === "1");
   242→    } catch {}
   243→    const q = bootParam("q");
   244→    if (q) setInput(q.slice(0, 2000));
   245→    setReady(true);
   246→  }, []);
   247→
   248→  const persist = (next: AgentMsg[]) => {
   249→    setMessages(next);
   250→    try {
   251→      localStorage.setItem(CHAT_KEY, JSON.stringify(next.slice(-MAX_STORED)));
   252→    } catch {}
   253→  };
   254→
   255→  const toggleDeep = () => {
   256→    setDeep((d) => {
   257→      const next = !d;
   258→      try {
   259→        localStorage.setItem(THINK_KEY, next ? "1" : "0");
   260→      } catch {}
   261→      return next;
   262→    });
   263→  };
   264→
   265→  // elapsed-seconds timer while the agent works
   266→  useEffect(() => {
   267→    if (!busy) {
   268→      setElapsed(0);
   269→      return;
   270→    }
   271→    const t = setInterval(() => setElapsed((e) => e + 1), 1000);
   272→    return () => clearInterval(t);
   273→  }, [busy]);
   274→
   275→  // keep the newest message in view (also while the answer streams in)
   276→  useEffect(() => {
   277→    bottomRef.current?.scrollIntoView({ behavior: "smooth", block: "end" });
   278→  }, [messages, busy, liveSteps, streamText]);
   279→
   280→  const ask = async (text: string) => {
   281→    const q = text.trim();
   282→    if (!q || busy) return;
   283→    lastQueryRef.current = q;
   284→    const userMsg: AgentMsg = { role: "user", content: q, ts: Date.now() };
   285→    const history = [...messages, userMsg];
   286→    persist(history);
   287→    setInput("");
   288→    setBusy(true);
   289→    setLiveSteps([]);
   290→    setLiveNote(null);
   291→    setStreamText("");
   292→    const ac = new AbortController();
   293→    abortRef.current = ac;
   294→    let streamSoFar = "";
   295→    try {
   296→      const res = await fetch("/api/agent", {
   297→        method: "POST",
   298→        headers: { "Content-Type": "application/json" },
   299→        signal: ac.signal,
   300→        body: JSON.stringify({
   301→          messages: history.filter((m) => !m.error).map((m) => ({ role: m.role, content: m.content })),
   302→          lang,
   303→          deviceId: getDeviceId(), // usage metering + per-user hourly limit
   304→          deep, // the composer's extended-thinking toggle (Task 22-b)
   305→        }),
   306→      });
   307→      const ct = res.headers.get("content-type") ?? "";
   308→      // plain-JSON error path (429 rate limit / 400 validation / gateway html)
   309→      if (!res.ok || !ct.includes("text/event-stream")) {
   310→        const json = (await res.json().catch(() => ({}))) as { error?: string };
   311→        persist([
   312→          ...history,
   313→          {
   314→            role: "assistant",
   315→            content: json.error ? `${tt(T.agentError, lang)} (${json.error})` : tt(T.agentError, lang),
   316→            error: true,
   317→            ts: Date.now(),
   318→          },
   319→        ]);
   320→        return;
   321→      }
   322→      // SSE stream: tool steps arrive as they run, the answer previews live
   323→      const reader = res.body?.getReader();
   324→      if (!reader) throw new Error("no stream body");
   325→      const dec = new TextDecoder();
   326→      let buf = "";
   327→      let gotTerminal = false;
   328→      const stepsAcc: AgentStep[] = [];
   329→      const handleEvent = (evt: Record<string, unknown>) => {
   330→        if (evt.type === "step") {
   331→          stepsAcc.push({ tool: String(evt.tool ?? ""), args: (evt.args as Record<string, unknown>) ?? {}, ok: evt.ok !== false });
   332→          setLiveSteps([...stepsAcc]);
   333→        } else if (evt.type === "status") {
   334→          setLiveNote(typeof evt.note === "string" ? evt.note : null);
   335→        } else if (evt.type === "delta" && typeof evt.text === "string") {
   336→          streamSoFar += evt.text;
   337→          setStreamText(streamSoFar);
   338→        } else if (evt.type === "done" && typeof evt.answer === "string") {
   339→          gotTerminal = true;
   340→          const finalSteps = Array.isArray(evt.steps) ? (evt.steps as AgentStep[]) : stepsAcc;
   341→          const next: AgentMsg[] = [...history, { role: "assistant", content: evt.answer, steps: finalSteps, ts: Date.now() }];
   342→          persist(next);
   343→          saveChat(next); // server-side history (fire-and-forget)
   344→        } else if (evt.type === "error") {
   345→          gotTerminal = true;
   346→          persist([
   347→            ...history,
   348→            {
   349→              role: "assistant",
   350→              content: evt.message ? `${tt(T.agentError, lang)} (${evt.message})` : tt(T.agentError, lang),
   351→              error: true,
   352→              ts: Date.now(),
   353→            },
   354→          ]);
   355→        }
   356→      };
   357→      for (;;) {
   358→        const { done, value } = await reader.read();
   359→        if (done) break;
   360→        buf += dec.decode(value, { stream: true });
   361→        let sep: number;
   362→        while ((sep = buf.indexOf("\n\n")) !== -1) {
   363→          const rawEvt = buf.slice(0, sep);
   364→          buf = buf.slice(sep + 2);
   365→          for (const line of rawEvt.split("\n")) {
   366→            if (!line.startsWith("data:")) continue;
   367→            try {
   368→              handleEvent(JSON.parse(line.slice(5).trim()) as Record<string, unknown>);
   369→            } catch {
   370→              /* partial line — the next chunk completes it */
   371→            }
   372→          }
   373→        }
   374→      }
   375→      if (!gotTerminal) throw new Error("stream ended without a terminal event");
   376→    } catch (err) {
   377→      // user pressed stop — keep whatever streamed as the (partial) answer
   378→      if (err instanceof DOMException && err.name === "AbortError") {
   379→        if (streamSoFar.trim()) {
   380→          persist([...history, { role: "assistant", content: streamSoFar, ts: Date.now() }]);
   381→        } else {
   382→          persist([...history, { role: "assistant", content: tt(T.agentStopped, lang), error: true, ts: Date.now() }]);
   383→        }
   384→      } else {
   385→        persist([...history, { role: "assistant", content: tt(T.agentError, lang), error: true, ts: Date.now() }]);
   386→      }
   387→    } finally {
   388→      abortRef.current = null;
   389→      setBusy(false);
   390→      setLiveSteps([]);
   391→      setLiveNote(null);
   392→      setStreamText("");
   393→    }
   394→  };
   395→
   396→  const stop = () => {
   397→    abortRef.current?.abort();
   398→  };
   399→
   400→  const retry = () => {
   401→    if (lastQueryRef.current && !busy) void ask(lastQueryRef.current);
   402→  };
   403→
   404→  // ── server-side chat history (auto-saved after every exchange) ──
   405→
   406→  const saveChat = (msgs: AgentMsg[]) => {
   407→    const id = chatIdRef.current;
   408→    if (!id || msgs.length === 0) return;
   409→    const title = (msgs.find((m) => m.role === "user")?.content ?? "").slice(0, 80);
   410→    void fetch("/api/agent/chats", {
   411→      method: "PUT",
   412→      headers: { "Content-Type": "application/json" },
   413→      body: JSON.stringify({
   414→        deviceId: getDeviceId(),
   415→        id,
   416→        title,
   417→        messages: msgs.filter((m) => !m.error).slice(-80),
   418→      }),
   419→    }).catch(() => {}); // best-effort — the next exchange retries
   420→  };
   421→
   422→  const loadHistory = async () => {
   423→    setHistoryLoading(true);
   424→    try {
   425→      const res = await fetch(`/api/agent/chats?deviceId=${encodeURIComponent(getDeviceId())}`);
   426→      const json = (await res.json()) as { chats?: HistoryRow[] };
   427→      setHistoryList(json.chats ?? []);
   428→    } catch {}
   429→    setHistoryLoading(false);

   431→    try {
   432→      const res = await fetch("/api/usage");
   433→      if (res.ok) setUsage((await res.json()) as UsageSummary);
   434→    } catch {}
   435→  };
   436→
   437→  const loadChat = async (id: string) => {
   438→    if (busy) return;
   439→    try {
   440→      const res = await fetch(`/api/agent/chats?deviceId=${encodeURIComponent(getDeviceId())}&id=${encodeURIComponent(id)}`);
   441→      if (!res.ok) return;
   442→      const json = (await res.json()) as { messages?: AgentMsg[] };
   443→      const msgs = (json.messages ?? [])
   444→        .filter((m) => m && (m.role === "user" || m.role === "assistant") && typeof m.content === "string")
   445→        .slice(-MAX_STORED);
   446→      persist(msgs);
   447→      chatIdRef.current = id;
   448→      try {
   449→        localStorage.setItem(CHAT_ID_KEY, id);
   450→      } catch {}
   451→      setHistoryOpen(false);
   452→    } catch {}
   453→  };
   454→
   455→  const deleteChat = async (id: string) => {
   456→    try {
   457→      await fetch(`/api/agent/chats?deviceId=${encodeURIComponent(getDeviceId())}&id=${encodeURIComponent(id)}`, {
   458→        method: "DELETE",
   459→      });

   464→      }
   465→    } catch {}
   466→  };

   469→    const id = newChatId();
   470→    chatIdRef.current = id;
   471→    try {
   472→      localStorage.setItem(CHAT_ID_KEY, id);
   473→    } catch {}
   474→    persist([]);
   475→    setHistoryOpen(false);
   476→  };
   477→
   478→  const newChat = () => {
   479→    startNewChat();
   480→    toast(tt(T.agentClear, lang));
   481→  };

   485→  /** the composer's bottom-left chips: tools · extended thinking · model */
   486→  const toolbar = (
   487→    <>
   488→      {/* tools chip */}
   489→      <Popover>
   490→        <PopoverTrigger asChild>
   491→          <button
   492→            type="button"
   493→            aria-label={tt(T.agentAttachTitle, lang)}
   494→            title={tt(T.agentAttachTitle, lang)}
   495→            className="inline-flex h-7 items-center gap-1.5 rounded-full border px-2.5 text-[11px] transition-colors hover:opacity-80"
   496→            style={{ color: "var(--chat-muted)" }}
   497→          >
   498→            <Wrench className="h-3 w-3" aria-hidden />
   499→            <span className="hidden sm:inline">{tt(T.agentToolsInfo, lang)}</span>
