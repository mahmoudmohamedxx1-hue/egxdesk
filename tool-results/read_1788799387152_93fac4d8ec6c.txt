   221→   221→// ── helpers ──
   222→   222→
   223→   223→function parseNum(s: string): number | null {
   224→   224→  if (s === "") return null;
   225→   225→  const v = Number(s);
   226→   226→  return Number.isFinite(v) ? v : null;
   227→   227→}
   228→   228→
   229→   229→/** bound passes only when the metric exists and is inside [min,max] */
   230→   230→function passesBound(v: number | null, b: Bound): boolean {
   231→   231→  const min = parseNum(b.min);
   232→   232→  const max = parseNum(b.max);
   233→   233→  if (min === null && max === null) return true;
   234→   234→  if (v === null || v === undefined || !Number.isFinite(v)) return false;
   235→   235→  if (min !== null && v < min) return false;
   236→   236→  if (max !== null && v > max) return false;
   237→   237→  return true;
   238→   238→}
   239→   239→
   240→   240→function passesMin(v: number | null, s: string, scale = 1): boolean {
   241→   241→  const min = parseNum(s);
   242→   242→  if (min === null) return true;
   243→   243→  if (v === null || v === undefined || !Number.isFinite(v)) return false;
   244→   244→  return v >= min * scale;
   245→   245→}
   246→   246→
   247→   247→function rangePos(r: CompanyRow): number | null {
   248→   248→  if (r.high52 === null || r.low52 === null || r.close == null) return null;
   249→   249→  const span = r.high52 - r.low52;
   250→   250→  if (!(span > 0)) return null;
   251→   251→  return (r.close - r.low52) / span;
   252→   252→}
   253→   253→
   254→   254→function boundChipLabel(label: string, b: Bound): string {
   255→   255→  if (b.min && b.max) return `${label}: ${b.min}–${b.max}`;
   256→   256→  if (b.min) return `${label} ≥ ${b.min}`;
   257→   257→  return `${label} ≤ ${b.max}`;
   258→   258→}

   316→   316→
   317→   317→export function ScreenerView() {
   318→   318→  const { lang, navigate } = useApp();
   319→   319→  const { data } = useLiveData<{ session: SessionMeta; total: number; rows: CompanyRow[] }>("/api/companies");
   320→   320→  const [f, setF] = useState<Filters>(DEFAULT_FILTERS);
   321→   321→  const [panelOpen, setPanelOpen] = useState(true);
   322→   322→  const [sortKey, setSortKey] = useState<ColKey>("marketCap");
   323→   323→  const [desc, setDesc] = useState(true);
   324→   324→
   325→   325→  const rows = data?.rows ?? null;
   326→   326→  const total = data?.total ?? rows?.length ?? 0;
   327→   327→
   328→   328→  const sectors = useMemo(() => {
   329→   329→    if (!rows) return [];
   330→   330→    const seen = new Map<string, string>();
   331→   331→    rows.forEach((r) => seen.set(r.sectorCode, lang === "ar" ? r.sectorAr : r.sectorEn));
   332→   332→    return Array.from(seen.entries())
   333→   333→      .map(([code, name]) => ({ code, name }))
   334→   334→      .sort((a, b) => a.name.localeCompare(b.name, lang === "ar" ? "ar" : "en"));
   335→   335→  }, [rows, lang]);
   336→   336→
   337→   337→  const applyPatch = (patch: Partial<Filters>) => setF((prev) => ({ ...prev, ...patch }));
   338→   338→
   339→   339→  const filtered = useMemo(() => {
   340→   340→    if (!rows) return null;
   341→   341→    const needle = f.q.trim().toLowerCase();
   342→   342→    const out = rows.filter((r) => {
   343→   343→      if (needle) {
   344→   344→        const hay = `${r.ticker} ${r.name} ${r.nameAr ?? ""} ${r.sectorEn} ${r.sectorAr}`.toLowerCase();
   345→   345→        if (!hay.includes(needle) && !rowMatchesArabic(r.ticker, f.q.trim())) return false;
   346→   346→      }
   347→   347→      if (f.sector && r.sectorCode !== f.sector) return false;
   348→   348→      if (!passesBound(r.close, f.price)) return false;
   349→   349→      if (!passesBound(r.changePct, f.change)) return false;
   350→   350→      if (!passesBound(r[f.perfPeriod], f.perf)) return false;
   351→   351→      if (!passesBound(r.marketCap != null ? r.marketCap / 1e6 : null, f.cap)) return false;
   352→   352→      if (!passesBound(r.pe, f.pe)) return false;
   353→   353→      if (!passesBound(r.pb ?? null, f.pb)) return false;
   354→   354→      if (!passesBound(r.divYield, f.yield)) return false;
   355→   355→      if (!passesBound(r.roe ?? null, f.roe)) return false;
   356→   356→      if (!passesBound(r.debtToEquity ?? null, f.de)) return false;
   357→   357→      if (!passesBound(r.eps, f.eps)) return false;
   358→   358→      if (!passesMin(r.volume, f.volumeMin)) return false;
   359→   359→      if (!passesMin(r.valueTraded != null ? r.valueTraded / 1e6 : null, f.valueMin)) return false;
   360→   360→      if (!passesMin(r.volumeRatio, f.volRatioMin)) return false;
   361→   361→      if (f.range52) {
   362→   362→        const pos = rangePos(r);
   363→   363→        if (pos === null) return false;
   364→   364→        if (f.range52 === "high" && pos < 0.95) return false;
   365→   365→        if (f.range52 === "low" && pos > 0.1) return false;
   366→   366→      }
   367→   367→      return true;
   368→   368→    });
   369→   369→    // sort: rows missing the sort metric sink to the bottom either direction
   370→   370→    const col = COLUMNS.find((c) => c.key === sortKey)!;
   371→   371→    const withVal = out.filter((r) => col.val(r) !== null && col.val(r) !== undefined);
   372→   372→    const without = out.filter((r) => col.val(r) === null || col.val(r) === undefined);
   373→   373→    const sorted = [...withVal].sort((a, b) => {
   374→   374→      const va = col.val(a);
   375→   375→      const vb = col.val(b);
   376→   376→      let cmp: number;
   377→   377→      if (typeof va === "string" || typeof vb === "string") {
   378→   378→        cmp = String(va).localeCompare(String(vb));
   379→   379→      } else {
   380→   380→        cmp = (va as number) - (vb as number);
   381→   381→      }
   382→   382→      return desc ? -cmp : cmp;
   383→   383→    });
   384→   384→    return [...sorted, ...without];
   385→   385→  }, [rows, f, sortKey, desc]);
   386→   386→
   387→   387→  // active-filter chips
   388→   388→  const chips: { id: string; label: string; clear: () => void }[] = [];
   389→   389→  if (f.q.trim()) chips.push({ id: "q", label: `"${f.q.trim()}"`, clear: () => applyPatch({ q: "" }) });
   390→   390→  if (f.sector) {
   391→   391→    const s = sectors.find((x) => x.code === f.sector);
   392→   392→    if (s) chips.push({ id: "sector", label: s.name, clear: () => applyPatch({ sector: "" }) });
   393→   393→  }
   394→   394→  const boundDefs: [keyof Filters, string][] = [
   395→   395→    ["price", tt(T.filterPrice, lang)],
   396→   396→    ["change", tt(T.filterChange, lang)],
   397→   397→    ["pe", tt(T.filterPe, lang)],
   398→   398→    ["pb", tt(T.filterPb, lang)],
   399→   399→    ["yield", tt(T.filterYield, lang)],
   400→   400→    ["roe", tt(T.filterRoe, lang)],
   401→   401→    ["de", tt(T.filterDe, lang)],
   402→   402→    ["eps", tt(T.filterEps, lang)],
   403→   403→    ["cap", tt(T.filterCap, lang)],
   404→   404→  ];
   405→   405→  boundDefs.forEach(([key, label]) => {
   406→   406→    const b = f[key] as Bound;
   407→   407→    if (b.min || b.max) {
   408→   408→      chips.push({
   409→   409→        id: `b-${key}`,
   410→   410→        label: boundChipLabel(label, b),
   411→   411→        clear: () => applyPatch({ [key]: EMPTY_BOUND } as Partial<Filters>),
   412→   412→      });
   413→   413→    }
   414→   414→  });
   415→   415→  if (f.perf.min || f.perf.max) {
   416→   416→    const periodLabel = tt(PERF_OPTIONS.find(([p]) => p === f.perfPeriod)![1], lang);
   417→   417→    chips.push({
   418→   418→      id: "perf",
   419→   419→      label: boundChipLabel(`${tt(T.filterPerf, lang)} (${periodLabel})`, f.perf),
   420→   420→      clear: () => applyPatch({ perf: EMPTY_BOUND }),
   421→   421→    });
   422→   422→  }
   423→   423→  if (f.volumeMin) chips.push({ id: "vol", label: `${tt(T.filterVolume, lang)} ≥ ${f.volumeMin}`, clear: () => applyPatch({ volumeMin: "" }) });
   424→   424→  if (f.valueMin) chips.push({ id: "val", label: `${tt(T.filterValue, lang)} ≥ ${f.valueMin}`, clear: () => applyPatch({ valueMin: "" }) });
   425→   425→  if (f.volRatioMin) chips.push({ id: "vr", label: `${tt(T.filterVolRatio, lang)} ≥ ${f.volRatioMin}`, clear: () => applyPatch({ volRatioMin: "" }) });
   426→   426→  if (f.range52) {
