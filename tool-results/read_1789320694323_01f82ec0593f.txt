    85→/** G19 — trendline segments persisted per symbol on the device. */
    86→type Seg = { a: { x: string; y: number }; b: { x: string; y: number } };
    87→const LINES_KEY = "egx-trendlines";
    88→
    89→function loadLines(symbol: string): Seg[] {
    90→  try {
    91→    const raw = localStorage.getItem(LINES_KEY);
    92→    if (!raw) return [];
    93→    const map = JSON.parse(raw) as Record<string, Seg[] | undefined>;
    94→    const arr = map?.[symbol];
    95→    if (!Array.isArray(arr)) return [];
    96→    return arr.filter(
    97→      (s) => s && s.a && s.b && typeof s.a.x === "string" && Number.isFinite(s.a.y) && typeof s.b.x === "string" && Number.isFinite(s.b.y)
    98→    );
    99→  } catch {
   100→    return [];
   101→  }
   102→}
   103→
   104→function saveLines(symbol: string, segs: Seg[]): void {
   105→  try {
   106→    const raw = localStorage.getItem(LINES_KEY);
   107→    const map = raw ? (JSON.parse(raw) as Record<string, Seg[]>) : {};
   108→    map[symbol] = segs;
   109→    localStorage.setItem(LINES_KEY, JSON.stringify(map));
   110→  } catch {}
   111→}

   113→export function PriceChart({ symbol, defaultRange = "6M" }: { symbol: string; defaultRange?: string }) {
   114→  const { lang } = useApp();
   115→  const [range, setRange] = useState(defaultRange);
   116→  const [data, setData] = useState<ChartResponse | null>(null);
   117→  const [error, setError] = useState<string | null>(null);
   118→  const [loading, setLoading] = useState(true);
   119→  const [inds, setInds] = useState<Record<IndKey, boolean>>({
   120→    sma20: true,
   121→    sma50: false,
   122→    ema20: false,
   123→    bb: false,
   124→    volma: false,
   125→    rsi: false,
   126→    macd: false,
   127→  });
   128→  // G8: chart modes
   129→  const [logScale, setLogScale] = useState(false);
   130→  const [compare, setCompare] = useState<string>("none");
   131→  const [cmpData, setCmpData] = useState<ChartResponse | null>(null);
   132→  const [cmpNote, setCmpNote] = useState<string | null>(null);
   133→  // G19: trendline drawing
   134→  const [drawMode, setDrawMode] = useState(false);
   135→  const [pending, setPending] = useState<{ x: string; y: number } | null>(null);
   136→  const [segments, setSegments] = useState<Seg[]>([]);
   137→  const mounted = useRef(true);
   138→
   139→  useEffect(() => {
   140→    mounted.current = true;
   141→    let cancelled = false;
   142→    const load = async () => {
   143→      try {
   144→        const res = await fetch(`/api/chart?symbol=${encodeURIComponent(symbol)}&range=${range}`, {
   145→          cache: "no-store",
   146→        });
   147→        const json = (await res.json()) as ChartResponse;
   148→        if (!res.ok || json.error) throw new Error(json.error ?? `chart ${res.status}`);
   149→        if (!cancelled) {
   150→          setData(json);
   151→          setError(null);
   152→        }
   153→      } catch (e: unknown) {
   154→        if (!cancelled) setError(e instanceof Error ? e.message : "chart unavailable");
   155→      } finally {
   156→        if (!cancelled) setLoading(false);
   157→      }
   158→    };
   159→    load();
   160→    return () => {
   161→      cancelled = true;
   162→      mounted.current = false;
   163→    };
   164→  }, [symbol, range]);
   165→
   166→  // while the index archive warms up, keep polling until real rows appear
   167→  useEffect(() => {
   168→    if (!data?.warming) return;
   169→    const t = setInterval(async () => {
   170→      try {
   171→        const res = await fetch(`/api/chart?symbol=${encodeURIComponent(symbol)}&range=${range}`, {
   172→          cache: "no-store",
   173→        });
   174→        if (!res.ok) return;
   175→        const json = (await res.json()) as ChartResponse;
   176→        if (json.error || json.points.length < 2) return;
   177→        if (mounted.current) {
   178→          setData(json);
   179→          setError(null);
   180→        }
   181→      } catch {
   182→        /* keep polling */
   183→      }
   184→    }, 8_000);
   185→    return () => clearInterval(t);
   186→  }, [data?.warming, symbol, range]);
   187→
   188→  // G19 — restore this symbol's trendlines after mount (SSR-safe)
   189→  useEffect(() => {
   190→    setSegments(loadLines(symbol));
   191→    setPending(null);
   192→    setDrawMode(false);
   193→  }, [symbol]);
   194→
   195→  // G8 — fetch the comparison series for the same range (indices only)
   196→  useEffect(() => {
   197→    if (compare === "none") {
   198→      setCmpData(null);
   199→      setCmpNote(null);
   200→      return;
   201→    }
   202→    let cancelled = false;
   203→    (async () => {
   204→      setCmpNote(null);
   205→      try {
   206→        const res = await fetch(`/api/chart?symbol=${compare}&range=${range}`, { cache: "no-store" });
   207→        const json = (await res.json()) as ChartResponse;
   208→        if (!res.ok || json.error || json.points.length < 2) throw new Error("unavailable");
   209→        if (!cancelled) {
   210→          setCmpData(json);
   211→          setCmpNote(null);
   212→        }
   213→      } catch {
   214→        if (!cancelled) {
   215→          setCmpData(null);
   216→          setCmpNote(lang === "ar" ? "سلسلة المقارنة غير متاحة لهذا النطاق" : "Comparison series unavailable for this range");
   217→        }
   218→      }
   219→    })();
   220→    return () => {
   221→      cancelled = true;
   222→    };
   223→  }, [compare, range, lang]);
   224→
   225→  if (loading && !data) {
   226→    return (
   227→      <div className="space-y-2">
   228→        <div className="flex gap-1.5">
   229→          {[0, 1, 2, 3, 4].map((i) => (
   230→            <Skeleton key={i} className="h-7 w-14" />
   231→          ))}
   232→        </div>
   233→        <Skeleton className="h-64 w-full" />
   234→      </div>
   235→    );
   236→  }
   237→
   238→  if (error && !data) {
   239→    return <p className="py-8 text-center text-sm text-muted-foreground">{tt(T.chartUnavailable, lang)}</p>;
   240→  }
   241→  if (!data) return null;
   242→
   243→  const points = data.points;
   244→  const up = (data.changePct ?? 0) >= 0;
   245→  const stroke = up ? "var(--up)" : "var(--down)";
   246→  const showVolume = points.some((p) => typeof p.volume === "number");
   247→  const n = points.length;
   248→
   249→  // indicator availability per range length
   250→  const canOverlay = n >= 21;
   251→  const canRsi = n >= 15;
   252→  const canMacd = n >= 36;
   253→  const canVolMa = showVolume && n >= 21;
   254→  const avail: Record<IndKey, boolean> = {
   255→    sma20: canOverlay,
   256→    sma50: canOverlay,
   257→    ema20: canOverlay,
   258→    bb: canOverlay,
   259→    volma: canVolMa,
   260→    rsi: canRsi,
   261→    macd: canMacd,
   262→  };
   263→  const anyAvail = Object.values(avail).some(Boolean);
   264→  const activeInds = INDICATORS.filter((d) => inds[d.key] && avail[d.key]);
   265→  const useVolMa = inds.volma && canVolMa;
   266→  const showVolMaToggle = showVolume && n >= 21; // toggle visible whenever volume + length allow
   267→
   268→  // compute series
   269→  const closes = points.map((p) => p.close);
   270→  const sSma20 = inds.sma20 && canOverlay ? smaSeries(closes, 20) : null;
   271→  const sSma50 = inds.sma50 && canOverlay ? smaSeries(closes, 50) : null;
   272→  const sEma20 = inds.ema20 && canOverlay ? emaFull(closes, 20) : null;
   273→  const sBb = inds.bb && canOverlay ? bollingerSeries(closes, 20, 2) : null;
   274→  const sRsi = inds.rsi && canRsi ? rsiSeries(closes, 14) : null;
   275→  const sMacd = inds.macd && canMacd ? macdSeries(closes) : null;
   276→  const sVolMa = useVolMa
   277→    ? smaSeries(
   278→        points.map((p) => (typeof p.volume === "number" ? p.volume : 0)),
   279→        20,
   280→      )
   281→    : null;
   282→
   283→  const chartData = points.map((p, i) => ({
   284→    date: p.date.slice(2), // YY-MM-DD compact tick label
   285→    close: p.close,
   286→    volume: p.volume ?? undefined,
   287→    sma20: sSma20?.[i] ?? undefined,
   288→    sma50: sSma50?.[i] ?? undefined,
   289→    ema20: sEma20?.[i] ?? undefined,
   290→    bbUp: sBb?.up[i] ?? undefined,
   291→    bbMid: sBb?.mid[i] ?? undefined,
   292→    bbLo: sBb?.lo[i] ?? undefined,
   293→    volMa: sVolMa?.[i] ?? undefined,
   294→    rsi: sRsi?.[i] ?? undefined,
   295→    macd: sMacd?.macd[i] ?? undefined,
   296→    macdSignal: sMacd?.signal[i] ?? undefined,
   297→    macdHist: sMacd?.hist[i] ?? undefined,
   298→  }));
   299→
   300→  // G8 compare mode: rebase both series to 100 at the window start and merge
   301→  // onto the main date axis (union of dates, forward-filled). Plain function
   302→  // (not a hook): this sits after the early returns above.
   303→  const compareOn = compare !== "none" && !!cmpData && cmpData!.points.length >= 2;
   304→  const cmpLabel = compareOn
   305→    ? tt(COMPARE_OPTIONS.find((o) => o.key === compare) ?? { ar: compare, en: compare }, lang)
   306→    : "";
   307→  const compareData = (() => {
   308→    if (!compareOn || !cmpData) return null;
   309→    const cmpMap = new Map(cmpData.points.map((p) => [p.date, p.close]));
   310→    const mainMap = new Map(points.map((p) => [p.date, p.close]));
   311→    const dates = Array.from(new Set([...mainMap.keys(), ...cmpMap.keys()])).sort();
   312→    let m0: number | null = null;
   313→    let c0: number | null = null;
   314→    let mLast = 100;
   315→    let cLast = 100;
   316→    const rows: { date: string; main100: number; cmp100: number; close?: number }[] = [];
   317→    for (const d of dates) {
   318→      const m = mainMap.get(d);
   319→      const c = cmpMap.get(d);
   320→      if (m != null && m0 === null) m0 = m;
   321→      if (c != null && c0 === null) c0 = c;
   322→      if (m != null && m0) mLast = (m / m0) * 100;
   323→      if (c != null && c0) cLast = (c / c0) * 100;
   324→      rows.push({ date: d.slice(2), main100: mLast, cmp100: cLast, close: m });
   325→    }
   326→    return rows;
   327→  })();
   328→
   329→  // trendline click capture (G19): snaps to the session under the cursor
   330→  const handleChartClick = (state: { activeLabel?: string; activeTooltipIndex?: number }) => {
   331→    if (!drawMode || compareOn) return;
   332→    const idx = state.activeTooltipIndex;
   333→    if (idx == null || idx < 0 || idx >= chartData.length) return;
   334→    const row = chartData[idx];
   335→    const y = row.close;
   336→    if (typeof y !== "number" || !Number.isFinite(y)) return;
   337→    const point = { x: row.date, y };
   338→    if (!pending) {
   339→      setPending(point);
   340→    } else {
   341→      const next = [...segments, { a: pending, b: point }];
   342→      setSegments(next);
   343→      saveLines(symbol, next);
   344→      setPending(null);
   345→      setDrawMode(false);
   346→    }
   347→  };
   348→
   349→  const syncId = `pc-${data.symbol}-${data.range}`;
   350→  const lastIdx = n - 1;
   351→  const lastRsi = sRsi ? sRsi[lastIdx] : null;
   352→  const lastMacd = sMacd ? sMacd.macd[lastIdx] : null;
   353→
   354→  // tooltip overlay rows for enabled indicators
   355→  const tipExtras: { key: string; label: string; color: string; fmt: (v: number) => string }[] = [];
   356→  if (sSma20) tipExtras.push({ key: "sma20", label: tt(T.indSma20, lang), color: "var(--c1)", fmt: (v) => fmtNum(v) });
   357→  if (sSma50) tipExtras.push({ key: "sma50", label: tt(T.indSma50, lang), color: "var(--c2)", fmt: (v) => fmtNum(v) });
   358→  if (sEma20) tipExtras.push({ key: "ema20", label: tt(T.indEma20, lang), color: "var(--c5)", fmt: (v) => fmtNum(v) });
   359→  if (sBb) {
   360→    tipExtras.push({ key: "bbUp", label: `BB ↑`, color: "var(--c4)", fmt: (v) => fmtNum(v) });
   361→    tipExtras.push({ key: "bbLo", label: `BB ↓`, color: "var(--c4)", fmt: (v) => fmtNum(v) });
   362→  }
   363→  if (sVolMa)
   364→    tipExtras.push({ key: "volMa", label: tt(T.volMaLine, lang), color: "var(--c1)", fmt: (v) => fmtInt(v) });
   365→
   366→  const rsiState =
   367→    lastRsi === null ? null : lastRsi >= 70 ? "overbought" : lastRsi <= 30 ? "oversold" : "neutral";
   368→
   369→  return (
   370→    <div dir="ltr" className="space-y-3">
   371→      {/* header stats */}
   372→      <div className="flex items-end justify-between flex-wrap gap-2">
   373→        <div className="flex items-baseline gap-3 flex-wrap">
   374→          {data.last !== null && <span className="num text-2xl font-bold tracking-tight">{fmtNum(data.last)}</span>}
   375→          {data.changePct !== null && (
   376→            <span className={`num text-sm font-semibold ${directionClass(data.changePct)}`}>
   377→              {fmtPct(data.changePct)} <span className="text-muted-foreground font-normal">({tt(T.rangeChange, lang)})</span>
   378→            </span>
   379→          )}
   380→          {data.currency === "EGP" && <span className="text-[10px] text-muted-foreground">EGP</span>}
   381→          {data.kind === "index" && <span className="text-[10px] text-muted-foreground">{tt(T.indexPoints, lang)}</span>}
   382→        </div>
   383→        <div className="num text-[11px] text-muted-foreground">
   384→          {tt(T.chartHigh, lang)} {fmtNum(data.high, 1)} · {tt(T.chartLow, lang)} {fmtNum(data.low, 1)}
   385→        </div>
   386→      </div>
   387→
   388→      {/* range buttons + chart modes (G8) */}
   389→      <div className="flex items-center gap-1.5 flex-wrap" role="tablist" aria-label={tt(T.chartRange, lang)}>
   390→        {data.availableRanges.map((r) => {
   391→          const active = r === data.range;
   392→          return (
   393→            <button
   394→              key={r}
   395→              role="tab"
   396→              aria-selected={active}
   397→              onClick={() => setRange(r)}
   398→              className={`rounded-md border px-2.5 py-1 text-xs transition-colors ${
   399→                active ? "bg-secondary font-semibold border-ring" : "text-muted-foreground hover:bg-accent/50"
   400→              }`}
   401→            >
   402→              {tt(RANGE_LABELS[r] ?? { ar: r, en: r }, lang)}
   403→            </button>
   404→          );
   405→        })}
   406→        <button
   407→          onClick={() => setLogScale((v) => !v)}
   408→          aria-pressed={logScale}
   409→          title={tt(T.logScaleLabel, lang)}
   410→          className={`rounded-md border px-2.5 py-1 text-xs transition-colors ${
   411→            logScale ? "bg-secondary font-semibold border-ring" : "text-muted-foreground hover:bg-accent/50"
   412→          }`}
   413→        >
   414→          {tt(T.logScaleLabel, lang)}
   415→        </button>
   416→        <select
   417→          value={compare}
   418→          onChange={(e) => setCompare(e.target.value)}
   419→          aria-label={tt(T.compareChartLabel, lang)}
   420→          title={tt(T.compareChartNote, lang)}
   421→          className="h-7 rounded-md border bg-card px-1.5 text-[11px] text-foreground"
   422→        >
   423→          <option value="none">{tt(T.compareAddIndex, lang)}</option>
   424→          {COMPARE_OPTIONS.map((o) => (
   425→            <option key={o.key} value={o.key}>
   426→              {lang === "ar" ? o.ar : o.en}
   427→            </option>
   428→          ))}
   429→        </select>
   430→        {cmpNote && <span className="text-[10px] text-muted-foreground">{cmpNote}</span>}
   431→        {data.asOf && <span className="num ms-auto text-[10px] text-muted-foreground">{data.asOf}</span>}
   432→      </div>
   433→
   434→      {/* indicator toggles + trendline tools (G19) */}
   435→      {points.length >= 2 && (
   436→        <div className="flex items-center gap-1.5 flex-wrap">
   437→          <span className="text-[10px] font-medium text-muted-foreground me-1">{tt(T.indicatorsLabel, lang)}:</span>
   438→          {INDICATORS.map((d) => {
   439→            const on = inds[d.key];
   440→            const ok = avail[d.key] && !compareOn; // overlays are absolute-price — hidden in compare mode
   441→            return (
   442→              <button
   443→                key={d.key}
   444→                onClick={() => ok && setInds((s) => ({ ...s, [d.key]: !s[d.key] }))}
   445→                disabled={!ok}
   446→                title={ok ? undefined : compareOn ? tt(T.compareChartNote, lang) : tt(T.indNotEnough, lang)}
   447→                aria-pressed={on}
   448→                className={`inline-flex items-center gap-1.5 rounded-full border px-2 py-0.5 text-[11px] transition-colors ${
   449→                  !ok
   450→                    ? "opacity-40 cursor-not-allowed text-muted-foreground"
   451→                    : on
   452→                      ? "bg-secondary font-semibold border-ring"
   453→                      : "text-muted-foreground hover:bg-accent/50"
   454→                }`}
   455→              >
   456→                <span className="h-2 w-2 rounded-full" style={{ background: d.color }} aria-hidden />
   457→                {tt(d.t, lang)}
   458→              </button>
   459→            );
   460→          })}
   461→          {showVolMaToggle && !compareOn && (
   462→            <button
   463→              onClick={() => setInds((s) => ({ ...s, volma: !s.volma }))}
   464→              aria-pressed={inds.volma}
   465→              className={`inline-flex items-center gap-1.5 rounded-full border px-2 py-0.5 text-[11px] transition-colors ${
   466→                inds.volma
   467→                  ? "bg-secondary font-semibold border-ring"
   468→                  : "text-muted-foreground hover:bg-accent/50"
