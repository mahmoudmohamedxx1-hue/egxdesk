     1→"use client";
     2→
     3→import { useState } from "react";
     4→import { useApp } from "../market/app-context";
     5→import { useLiveData } from "../market/use-live-data";
     6→import type { CompanyRow, NewsRow, SessionMeta } from "../market/types";
     7→import { T, tt, dn } from "@/lib/i18n";
     8→import { fmtNum, fmtValue, fmtPct, fmtInt, fmtDateAr, fmtTimeAr, directionClass } from "@/lib/format";
     9→import { WatchStar } from "../market/watch-star";
    10→import { ChangeCell } from "../market/change-cell";
    11→import { PerfChart, RangeBar } from "../market/perf-chart";
    12→import { PriceChart } from "../market/price-chart";
    13→import { StatementsPanel } from "../market/statements-panel";
    14→import { TechnicalPanel } from "../market/technical-panel";
    15→import { Skeleton } from "@/components/ui/skeleton";
    16→import { Button } from "@/components/ui/button";
    17→import { Volume2, Calculator, TrendingUp, TrendingDown, ExternalLink, RefreshCw, Zap, CalendarClock, FileSpreadsheet } from "lucide-react";
    18→
    19→type CompanyData = {
    20→  session: SessionMeta;
    21→  company: CompanyRow & {
    22→    industry: string | null;
    23→    netMarginTTM: number | null;
    24→    revenueTTM: number | null;
    25→    floatShares: number | null;
    26→    avgTurnover30: number | null;
    27→    high1M: number | null;
    28→    low1M: number | null;
    29→    beta: number | null;
    30→    updateMode: string | null;
    31→    employees?: number | null;
    32→  };
    33→  sectorAgg: {
    34→    count: number;
    35→    nameAr: string;
    36→    nameEn: string;
    37→    pe: number | null;
    38→    pb: number | null;
    39→    roe: number | null;
    40→    eps: number | null;
    41→    divYield: number | null;
    42→    beta: number | null;
    43→    avgChangePct: number | null;
    44→  };
    45→  peers: CompanyRow[];
    46→  news: NewsRow[];
    47→  disclosures: NewsRow[];
    48→  signals: {
    49→    streak: { direction: "up" | "down"; count: number; since: string } | null;
    50→    nextEarnings: string | null;
    51→    unusualVolume: boolean;
    52→    volumeRatio: number | null;
    53→    near52High: boolean;
    54→    near52Low: boolean;
    55→    computedFrom: string;
    56→  } | null;
    57→};
    58→
    59→export function CompanyView({ ticker, panel }: { ticker: string; panel: string }) {
    60→  const { lang, navigate } = useApp();
    61→  const { data, error, refresh } = useLiveData<CompanyData>(`/api/company/${encodeURIComponent(ticker)}`, 60_000);
    62→  const [activePanel, setActivePanel] = useState(panel);
    63→  const [speaking, setSpeaking] = useState(false);
    64→  const [prevPanel, setPrevPanel] = useState(panel);
    65→  if (prevPanel !== panel) {
    66→    // URL panel changed (e.g. navigating from a peer link) — adjust state during render
    67→    setPrevPanel(panel);
    68→    setActivePanel(panel);
    69→  }
    70→
    71→  if (error && !data) {
    72→    return (
    73→      <div className="rounded-lg border bg-card p-8 text-center space-y-3">
    74→        <p className="font-medium">{tt(T.errorLoad, lang)}</p>
    75→        <Button size="sm" onClick={refresh}>
    76→          <RefreshCw className="h-3.5 w-3.5 me-1.5" />
    77→          {tt(T.retry, lang)}
    78→        </Button>
    79→      </div>
    80→    );
    81→  }
    82→
    83→  if (!data || !data.company) {
    84→    if (data && (data as { error?: string }).error === "no such company") {
    85→      return (
    86→        <div className="py-16 text-center space-y-2">
    87→          <p className="text-lg font-semibold">{lang === "ar" ? "لا توجد شركة بهذا الرمز" : "No such company"}</p>
    88→          <button onClick={() => navigate("market")} className="text-sm text-primary hover:underline">
    89→            {tt(T.browseMarket, lang)}
    90→          </button>
    91→        </div>
    92→      );
    93→    }
    94→    return (
    95→      <div className="space-y-4">
    96→        <Skeleton className="h-16 w-full" />
    97→        <Skeleton className="h-48 w-full" />
    98→        <Skeleton className="h-64 w-full" />
    99→      </div>
   100→    );
   101→  }
   102→
   103→  const c = data.company;
   104→
   105→  function speakQuote() {
   106→    if (typeof window === "undefined" || !window.speechSynthesis) return;
   107→    if (speaking) { window.speechSynthesis.cancel(); setSpeaking(false); return; }
   108→    const dir = c.changePct >= 0
   109→      ? (lang === "ar" ? "مرتفع" : "up")
   110→      : (lang === "ar" ? "منخفض" : "down");
   111→    const text = lang === "ar"
   112→      ? `${c.ticker}. ${dn(c, lang)}. آخر سعر ${fmtNum(c.close)} جنيه، ${dir} بنسبة ${fmtNum(Math.abs(c.changePct))} بالمئة.`
   113→      : `${c.ticker}. ${c.name}. Last ${fmtNum(c.close)} Egyptian pounds, ${dir} ${fmtNum(Math.abs(c.changePct))} percent.`;
   114→    const u = new SpeechSynthesisUtterance(text);
   115→    u.lang = lang === "ar" ? "ar-EG" : "en-US";
   116→    u.onend = () => setSpeaking(false);
   117→    u.onerror = () => setSpeaking(false);
   118→    window.speechSynthesis.speak(u);
   119→    setSpeaking(true);
   120→  }
   121→
   122→  const panels = [
   123→    { key: "overview", t: T.panelOverview },
   124→    { key: "technical", t: T.panelTechnical },
   125→    { key: "statements", t: T.panelStatements },
   126→    { key: "fundamentals", t: T.panelFundamentals },
   127→    { key: "disclosures", t: T.panelDisclosures },
   128→    { key: "activity", t: T.panelActivity },
   129→    { key: "news", t: T.panelNews },
   130→  ];
   131→
   132→  const perf = [
   133→    { label: "1W", value: c.perfW },
   134→    { label: "1M", value: c.perf1M },
   135→    { label: "3M", value: c.perf3M },
   136→    { label: "6M", value: c.perf6M },
   137→    { label: "YTD", value: c.perfYTD },
   138→    { label: "1Y", value: c.perfY },
   139→    { label: "3Y", value: c.perf3Y },
   140→    { label: "5Y", value: c.perf5Y },
   141→  ];
   142→
   143→  return (
   144→    <div className="space-y-5">
   145→      {/* ticker row */}
   146→      <div className="flex items-center flex-wrap gap-2">
   147→        <span className="num text-lg font-bold">{c.ticker}</span>
   148→        <WatchStar ticker={c.ticker} />
   149→        <button
   150→          onClick={speakQuote}
   151→          className="inline-flex items-center gap-1.5 rounded-md border px-2.5 py-1.5 text-xs hover:bg-accent/50 transition-colors"
   152→        >
   153→          <Volume2 className={`h-3.5 w-3.5 ${speaking ? "animate-pulse" : ""}`} />
   154→          {tt(T.listenBrief, lang)}
   155→        </button>
   156→        <button
   157→          onClick={() => navigate("tools")}
   158→          className="inline-flex items-center gap-1.5 rounded-md border px-2.5 py-1.5 text-xs hover:bg-accent/50 transition-colors"
   159→        >
   160→          <Calculator className="h-3.5 w-3.5" />
   161→          {lang === "ar" ? "احسب عائد الكوبون" : "Compute coupon return"}
   162→        </button>
   163→      </div>
   164→
   165→      {/* identity + quote */}
   166→      <div className="rounded-lg border bg-card p-4 sm:p-5">
   167→        <div className="flex flex-wrap items-start justify-between gap-4">
   168→          <div className="min-w-0 space-y-1">
   169→            <p className="text-xs text-muted-foreground">
   170→              {lang === "ar" ? c.sectorAr : c.sectorEn}
   171→              {c.industry ? ` · ${c.industry}` : ""}
   172→            </p>
   173→            <h1 className="text-xl sm:text-2xl font-bold tracking-tight">{dn(c, lang)}</h1>
   174→            {lang === "ar" && c.nameAr && c.nameAr !== c.name && (
   175→              <p className="text-xs text-muted-foreground truncate">{c.name}</p>
   176→            )}
   177→            {lang === "en" && c.nameAr && (
   178→              <p className="text-xs text-muted-foreground truncate" dir="rtl">{c.nameAr}</p>
   179→            )}
   180→            <p className="num text-xs text-muted-foreground">EGX · EGP · {tt(T.delayed, lang)}</p>
   181→          </div>
   182→          <div className="text-end space-y-0.5">
   183→            <p className="text-xs text-muted-foreground">{tt(T.lastClose, lang)}</p>
   184→            <p className="num text-3xl font-bold tracking-tight">{fmtNum(c.close)}</p>
   185→            <p className={`num text-sm font-semibold ${directionClass(c.changePct)} flex items-center gap-1.5 justify-end`}>
   186→              {c.changePct >= 0 ? <TrendingUp className="h-3.5 w-3.5" /> : <TrendingDown className="h-3.5 w-3.5" />}
   187→              {c.changePct >= 0 ? "+" : ""}{fmtNum(c.changeAbs)} ({fmtPct(c.changePct)})
   188→            </p>
   189→            <p className="num text-[10px] text-muted-foreground">{data.session.lastSession}</p>
   190→          </div>
   191→        </div>
   192→
   193→        {/* stat strip */}
   194→        <div className="mt-4 grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-6 gap-x-4 gap-y-3 border-t pt-4">
   195→          <Stat label={tt(T.marketCap, lang)} value={`EGP ${fmtValue(c.marketCap)}`} />
   196→          <Stat label={tt(T.week, lang)} value={fmtPct(c.perfW)} cls={directionClass(c.perfW)} />
   197→          <Stat label={tt(T.month, lang)} value={fmtPct(c.perf1M)} cls={directionClass(c.perf1M)} />
   198→          <Stat label={tt(T.ytd, lang)} value={fmtPct(c.perfYTD)} cls={directionClass(c.perfYTD)} />
   199→          <Stat label={tt(T.volume, lang)} value={fmtInt(c.volume)} sub={`${tt(T.inSession, lang)} · ${tt(T.avg10, lang)} ${fmtInt(c.avgVolume)}`} />
   200→          <Stat label="P/E" value={c.pe ? fmtNum(c.pe, 1) : "—"} sub={c.eps ? `EPS ${fmtNum(c.eps)}` : undefined} />
   201→        </div>
   202→      </div>
   203→
   204→      {/* panel tabs */}
   205→      <div className="flex items-center gap-1 overflow-x-auto thin-scroll border-b" role="tablist">
   206→        {panels.map((p) => (
   207→          <button
   208→            key={p.key}
   209→            role="tab"
   210→            aria-selected={activePanel === p.key}
   211→            onClick={() => setActivePanel(p.key)}
   212→            className={`whitespace-nowrap px-3 py-2 text-sm -mb-px border-b-2 transition-colors ${
   213→              activePanel === p.key ? "border-primary font-semibold" : "border-transparent text-muted-foreground hover:text-foreground"
   214→            }`}
   215→          >
   216→            {tt(p.t, lang)}
   217→            {p.key === "news" && data.news.length > 0 && (
   218→              <span className="num ms-1.5 text-xs text-muted-foreground">{data.news.length}</span>
   219→            )}
   220→            {p.key === "disclosures" && data.disclosures.length > 0 && (
   221→              <span className="num ms-1.5 text-xs text-muted-foreground">{data.disclosures.length}</span>
   222→            )}
   223→          </button>
   224→        ))}
   225→      </div>
   226→
   227→      {/* OVERVIEW */}
   228→      {activePanel === "overview" && (
   229→        <div className="space-y-5">
   230→          <section className="rounded-lg border bg-card p-4">
   231→            <div className="flex items-center justify-between mb-3 flex-wrap gap-2">
   232→              <h2 className="font-bold">{tt(T.priceChart, lang)}</h2>
   233→              <span className="num text-xs text-muted-foreground">{c.ticker} · EGX</span>
   234→            </div>
   235→            <PriceChart key={c.ticker} symbol={c.ticker} defaultRange="6M" />
   236→          </section>
   237→
   238→          {data.signals && <SignalsCard signals={data.signals} lang={lang} />}
   239→
   240→          <section className="rounded-lg border bg-card p-4">
   241→            <div className="flex items-center justify-between mb-3 flex-wrap gap-2">
   242→              <h2 className="font-bold">{tt(T.performance, lang)}</h2>
   243→            </div>
   244→            <PerfChart perf={perf} lang={lang} />
   245→            <p className="mt-2 text-[11px] text-muted-foreground">{tt(T.perfNote, lang)}</p>
   246→          </section>
   247→
   248→          <section className="rounded-lg border bg-card p-4">
   249→            <h2 className="font-bold mb-3">{tt(T.range52, lang)}</h2>
   250→            <RangeBar
   251→              low={c.low52}
   252→              high={c.high52}
   253→              close={c.close}
   254→              labelLow={lang === "ar" ? "أدنى ٥٢ أسبوعاً" : "52-week low"}
   255→              labelHigh={lang === "ar" ? "أعلى ٥٢ أسبوعاً" : "52-week high"}
   256→            />
   257→            {c.high1M !== null && c.low1M !== null && (
   258→              <p className="num mt-3 text-[11px] text-muted-foreground">
   259→                {lang === "ar" ? "مدى الشهر:" : "1-month range:"} {fmtNum(c.low1M, 1)} – {fmtNum(c.high1M, 1)}
   260→              </p>
   261→            )}
   262→          </section>
   263→        </div>
   264→      )}
   265→
   266→      {/* TECHNICAL ANALYSIS */}
   267→      {activePanel === "technical" && <TechnicalPanel key={c.ticker} ticker={c.ticker} />}
   268→
   269→      {/* STATEMENTS */}
   270→      {activePanel === "statements" && (
   271→        <StatementsPanel key={c.ticker} ticker={c.ticker} />
   272→      )}
   273→
   274→      {/* DISCLOSURES (press-derived) */}
   275→      {activePanel === "disclosures" && (
   276→        <section className="rounded-lg border bg-card divide-y">
   277→          <div className="px-4 py-3">
   278→            <h2 className="font-bold flex items-center gap-2">
   279→              <FileSpreadsheet className="h-4 w-4 text-primary" />
   280→              {tt(T.panelDisclosures, lang)}
   281→            </h2>
   282→            <p className="mt-1 text-xs text-muted-foreground max-w-3xl leading-relaxed">{tt(T.disclosuresNote, lang)}</p>
   283→          </div>
   284→          {data.disclosures.length === 0 && (
   285→            <p className="px-4 py-10 text-center text-sm text-muted-foreground">{tt(T.noDisclosures, lang)}</p>
   286→          )}
   287→          {data.disclosures.map((n) => (
   288→            <article key={n.id} className="px-4 py-3">
   289→              <div className="flex items-center gap-2 mb-1 flex-wrap">
   290→                <span className="rounded-sm bg-secondary px-1.5 py-0.5 text-[10px] font-medium">{n.source}</span>
   291→                <span className="num text-[11px] text-muted-foreground">{fmtDateAr(n.publishedAt)} · {fmtTimeAr(n.publishedAt)}</span>
   292→              </div>
   293→              <a href={n.link} target="_blank" rel="noopener noreferrer" className="text-sm leading-snug font-medium hover:underline inline-flex items-start gap-1.5">
   294→                {n.title}
   295→                <ExternalLink className="h-3 w-3 mt-1 shrink-0 text-muted-foreground" />
   296→              </a>
   297→              {n.snippet && <p className="mt-1 text-xs text-muted-foreground line-clamp-2">{n.snippet}</p>}
   298→            </article>
   299→          ))}
   300→        </section>
   301→      )}
   302→
   303→      {/* ACTIVITY */}
   304→      {activePanel === "activity" && (
   305→        <section className="rounded-lg border bg-card p-4 space-y-4">
   306→          <h2 className="font-bold">{tt(T.panelActivity, lang)}</h2>
   307→          <div className="grid grid-cols-2 sm:grid-cols-4 gap-4">
   308→            <Stat label={tt(T.volume, lang)} value={fmtInt(c.volume)} sub={tt(T.inSession, lang)} />
   309→            <Stat label={tt(T.avg10, lang)} value={fmtInt(c.avgVolume)} />
   310→            <Stat label={tt(T.valueTraded, lang)} value={`EGP ${fmtValue(c.valueTraded)}`} sub={tt(T.inSession, lang)} />
   311→            <Stat label={lang === "ar" ? "الحجم ÷ المعتاد" : "Vol ÷ usual"} value={c.volumeRatio !== null ? `${fmtNum(c.volumeRatio, 1)}×` : "—"} sub={tt(T.inSession, lang)} />
   312→          </div>
   313→          <div className="rounded-md bg-secondary/70 p-3 flex items-center justify-between">
   314→            <span className="text-xs text-muted-foreground">{lang === "ar" ? "الحجم ÷ المعتاد" : "Volume ÷ usual"}</span>
   315→            <span className="num text-lg font-bold text-primary">
   316→              {c.avgVolume ? fmtNum(c.volume / c.avgVolume, 1) + "×" : "—"}
   317→            </span>
   318→          </div>
   319→          <div>
   320→            <div className="flex items-center justify-between mb-1 text-xs">
   321→              <span className="text-muted-foreground">{lang === "ar" ? "مقابل متوسط ١٠ جلسات" : "vs 10-session average"}</span>
   322→              <span className="num">{fmtInt(c.volume)} / {fmtInt(c.avgVolume)}</span>
   323→            </div>
   324→            <div className="h-2 rounded-full bg-secondary overflow-hidden" dir="ltr">
   325→              <div className="h-full rounded-full bg-primary" style={{ width: `${Math.min((c.volume / Math.max(c.avgVolume ?? 1, 1)) * 50, 100)}%` }} />
   326→            </div>
   327→          </div>
   328→          {c.avgTurnover30 !== null && (
   329→            <Stat label={lang === "ar" ? "متوسط قيمة التداول (٣٠ يوماً)" : "Avg turnover (30d)"} value={`EGP ${fmtValue(c.avgTurnover30)}`} />
   330→          )}
   331→          <p className="text-[11px] text-muted-foreground">{tt(T.unusualNote, lang)}</p>
   332→        </section>
   333→      )}
   334→
   335→      {/* FUNDAMENTALS */}
   336→      {activePanel === "fundamentals" && (
   337→        <div className="space-y-4">
   338→          <section className="rounded-lg border bg-card p-4">
   339→            <h2 className="font-bold mb-1">{tt(T.fundamentals, lang)}</h2>
   340→            <p className="text-xs text-muted-foreground mb-3">
   341→              {tt(T.vsSector, lang)} — {lang === "ar" ? data.sectorAgg.nameAr : data.sectorAgg.nameEn}{" "}
   342→              <span className="num">({data.sectorAgg.count} {tt(T.companies, lang)})</span>
   343→            </p>
   344→            <div className="grid grid-cols-2 sm:grid-cols-3 gap-x-4 gap-y-3">
   345→              <VsSector label="P/E" value={c.pe} agg={data.sectorAgg.pe} fmt={(v) => fmtNum(v, 1)} />
   346→              <VsSector label={lang === "ar" ? "م/د (القيمة الدفترية)" : "P/B (book)"} value={c.pb} agg={data.sectorAgg.pb} fmt={(v) => fmtNum(v, 2)} />
   347→              <VsSector label={lang === "ar" ? "العائد على حقوق الملكية" : "ROE"} value={c.roe} agg={data.sectorAgg.roe} fmt={(v) => `${fmtNum(v, 1)}%`} higherBetter />
   348→              <VsSector label="EPS" value={c.eps} agg={data.sectorAgg.eps} fmt={(v) => fmtNum(v)} higherBetter />
   349→              <VsSector label={lang === "ar" ? "عائد التوزيعات" : "Div yield"} value={c.divYield} agg={data.sectorAgg.divYield} fmt={(v) => `${fmtNum(v, 1)}%`} higherBetter />
   350→              <VsSector label={tt(T.beta, lang)} value={c.beta} agg={data.sectorAgg.beta} fmt={(v) => fmtNum(v, 2)} />
   351→              <MetricOnly label={tt(T.netMargin, lang)} value={c.netMarginTTM !== null ? `${fmtNum(c.netMarginTTM, 1)}%` : "—"} />
   352→              <MetricOnly label={lang === "ar" ? "صافي الربح (١٢ شهراً)" : "Net income (TTM)"} value={c.netIncomeTTM !== null ? `EGP ${fmtValue(c.netIncomeTTM)}` : "—"} />
   353→              <MetricOnly label={tt(T.revenueTtm, lang)} value={c.revenueTTM !== null ? `EGP ${fmtValue(c.revenueTTM)}` : "—"} />
   354→              <MetricOnly label={lang === "ar" ? "الدين / حقوق الملكية" : "Debt / equity"} value={c.debtToEquity !== null ? fmtNum(c.debtToEquity, 2) : "—"} />
   355→              <MetricOnly label={lang === "ar" ? "هامش الربح الإجمالي" : "Gross margin"} value={c.grossMarginTTM !== null ? `${fmtNum(c.grossMarginTTM, 1)}%` : "—"} />
   356→              <MetricOnly label={lang === "ar" ? "صافي الدين" : "Net debt"} value={c.netDebt !== null ? `EGP ${fmtValue(c.netDebt)}` : "—"} />
   357→              <MetricOnly label={tt(T.floatShares, lang)} value={c.floatShares !== null ? fmtInt(c.floatShares) : "—"} />
   358→              <MetricOnly label={lang === "ar" ? "الموظفون" : "Employees"} value={c.employees !== null && c.employees !== undefined ? fmtInt(c.employees) : "—"} />
   359→              <MetricOnly label={tt(T.marketCap, lang)} value={`EGP ${fmtValue(c.marketCap)}`} />
   360→            </div>
   361→          </section>
   362→
   363→          {/* peers */}
   364→          <section className="rounded-lg border bg-card overflow-hidden">
   365→            <div className="border-b px-4 py-3">
   366→              <h2 className="font-bold">{tt(T.peers, lang)}</h2>
   367→            </div>
   368→            <div className="overflow-x-auto thin-scroll">
   369→              <table className="w-full text-sm">
   370→                <thead className="border-b bg-card">
   371→                  <tr className="text-[11px] text-muted-foreground">
   372→                    <th className="text-start font-medium px-3 py-2">{tt(T.colTicker, lang)}</th>
   373→                    <th className="text-start font-medium px-3 py-2 hidden md:table-cell">{tt(T.colName, lang)}</th>
   374→                    <th className="text-end font-medium px-3 py-2">{tt(T.colClose, lang)}</th>
   375→                    <th className="text-end font-medium px-3 py-2">{tt(T.colChange, lang)}</th>
   376→                    <th className="text-end font-medium px-3 py-2 hidden sm:table-cell">{tt(T.marketCap, lang)}</th>
   377→                    <th className="text-end font-medium px-3 py-2 hidden md:table-cell">{tt(T.colPe, lang)}</th>
   378→                  </tr>
   379→                </thead>
   380→                <tbody className="divide-y">

   514→function VsSector({ label, value, agg, fmt, higherBetter }: {
   515→  label: string; value: number | null | undefined; agg: number | null | undefined;
   516→  fmt: (v: number) => string; higherBetter?: boolean;
   517→}) {
   518→  const has = typeof value === "number" && typeof agg === "number";
   519→  let better: boolean | null = null;
   520→  if (has) {
   521→    better = higherBetter ? value > agg : value < agg;
   522→  }
   523→  return (
   524→    <div className="border-b border-dotted pb-2">
   525→      <p className="text-xs text-muted-foreground">{label}</p>
   526→      <div className="flex items-baseline gap-2 flex-wrap">
   527→        <span className="num text-sm font-semibold">{value !== null && value !== undefined ? fmt(value) : "—"}</span>
   528→        {has && (
   529→          <span className={`num text-[10px] px-1 rounded ${better ? "text-up bg-up-soft" : "text-down bg-down-soft"}`}>
   530→            {better ? "▲" : "▼"} {fmt(agg!)}
   531→          </span>
   532→        )}
   533→      </div>
   534→    </div>
   535→  );
   536→}
