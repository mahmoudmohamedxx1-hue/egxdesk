    93→/** G19 — trendline segments persisted per symbol on the device. */
    94→type Seg = { a: { x: string; y: number }; b: { x: string; y: number } };
    95→const LINES_KEY = "egx-trendlines";
    96→
    97→function loadLines(symbol: string): Seg[] {
    98→  try {
    99→    const raw = localStorage.getItem(LINES_KEY);
   100→    if (!raw) return [];
   101→    const map = JSON.parse(raw) as Record<string, Seg[] | undefined>;
   102→    const arr = map?.[symbol];
   103→    if (!Array.isArray(arr)) return [];
   104→    return arr.filter(
   105→      (s) => s && s.a && s.b && typeof s.a.x === "string" && Number.isFinite(s.a.y) && typeof s.b.x === "string" && Number.isFinite(s.b.y)
   106→    );
   107→  } catch {
   108→    return [];
   109→  }
   110→}
   111→
   112→function saveLines(symbol: string, segs: Seg[]): void {
   113→  try {
   114→    const raw = localStorage.getItem(LINES_KEY);
   115→    const map = raw ? (JSON.parse(raw) as Record<string, Seg[]>) : {};
   116→    map[symbol] = segs;
   117→    localStorage.setItem(LINES_KEY, JSON.stringify(map));
   118→  } catch {}
   119→}
