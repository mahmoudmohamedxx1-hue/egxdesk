    58→    58→function SunburstMark({ className }: { className?: string }) {
    59→    59→  const rays = Array.from({ length: 12 }, (_, i) => i * 30);
    60→    60→  return (
    61→    61→    <svg viewBox="0 0 24 24" fill="none" className={className} aria-hidden>
    62→    62→      {rays.map((a) => {
    63→    63→        const rad = (a * Math.PI) / 180;
    64→    64→        return (
    65→    65→          <line key={a} x1={12 + Math.sin(rad) * 3.4} y1={12 - Math.cos(rad) * 3.4}
    66→    66→            x2={12 + Math.sin(rad) * 10.2} y2={12 - Math.cos(rad) * 10.2}
    67→    67→            stroke="currentColor" strokeWidth="2.2" strokeLinecap="round" />
    68→    68→        );
    69→    69→      })}
    70→    70→    </svg>
    71→    71→  );
    72→    72→}
