   221→// ── helpers ──
   222→
   223→function parseNum(s: string): number | null {
   224→  if (s === "") return null;
   225→  const v = Number(s);
   226→  return Number.isFinite(v) ? v : null;
   227→}
   228→
   229→/** bound passes only when the metric exists and is inside [min,max] */
   230→function passesBound(v: number | null, b: Bound): boolean {
   231→  const min = parseNum(b.min);
   232→  const max = parseNum(b.max);
   233→  if (min === null && max === null) return true;
   234→  if (v === null || v === undefined || !Number.isFinite(v)) return false;
   235→  if (min !== null && v < min) return false;
   236→  if (max !== null && v > max) return false;
   237→  return true;
   238→}
   239→
   240→function passesMin(v: number | null, s: string, scale = 1): boolean {
   241→  const min = parseNum(s);
   242→  if (min === null) return true;
   243→  if (v === null || v === undefined || !Number.isFinite(v)) return false;
   244→  return v >= min * scale;
   245→}
   246→
   247→function rangePos(r: CompanyRow): number | null {
   248→  if (r.high52 === null || r.low52 === null || r.close == null) return null;
   249→  const span = r.high52 - r.low52;
   250→  if (!(span > 0)) return null;
   251→  return (r.close - r.low52) / span;
   252→}
   253→
   254→function boundChipLabel(label: string, b: Bound): string {
   255→  if (b.min && b.max) return `${label}: ${b.min}–${b.max}`;
   256→  if (b.min) return `${label} ≥ ${b.min}`;
   257→  return `${label} ≤ ${b.max}`;
   258→}

   317→export function ScreenerView() {
   318→  const { lang, navigate } = useApp();
   319→  const { data } = useLiveData<{ session: SessionMeta; total: number; rows: CompanyRow[] }>("/api/companies");
   320→  const [f, setF] = useState<Filters>(DEFAULT_FILTERS);
   321→  const [panelOpen, setPanelOpen] = useState(true);
   322→  const [sortKey, setSortKey] = useState<ColKey>("marketCap");
   323→  const [desc, setDesc] = useState(true);
   324→
   325→  const rows = data?.rows ?? null;
   326→  const total = data?.total ?? rows?.length ?? 0;
   327→
   328→  const sectors = useMemo(() => {
   329→    if (!rows) return [];
   330→    const seen = new Map<string, string>();
   331→    rows.forEach((r) => seen.set(r.sectorCode, lang === "ar" ? r.sectorAr : r.sectorEn));
   332→    return Array.from(seen.entries())
   333→      .map(([code, name]) => ({ code, name }))
   334→      .sort((a, b) => a.name.localeCompare(b.name, lang === "ar" ? "ar" : "en"));
   335→  }, [rows, lang]);
   336→
   337→  const applyPatch = (patch: Partial<Filters>) => setF((prev) => ({ ...prev, ...patch }));
   338→
   339→  const filtered = useMemo(() => {
   340→    if (!rows) return null;
   341→    const needle = f.q.trim().toLowerCase();
   342→    const out = rows.filter((r) => {
   343→      if (needle) {
   344→        const hay = `${r.ticker} ${r.name} ${r.nameAr ?? ""} ${r.sectorEn} ${r.sectorAr}`.toLowerCase();
   345→        if (!hay.includes(needle) && !rowMatchesArabic(r.ticker, f.q.trim())) return false;
   346→      }
   347→      if (f.sector && r.sectorCode !== f.sector) return false;
   348→      if (!passesBound(r.close, f.price)) return false;
   349→      if (!passesBound(r.changePct, f.change)) return false;
   350→      if (!passesBound(r[f.perfPeriod], f.perf)) return false;
   351→      if (!passesBound(r.marketCap != null ? r.marketCap / 1e6 : null, f.cap)) return false;
   352→      if (!passesBound(r.pe, f.pe)) return false;
   353→      if (!passesBound(r.pb ?? null, f.pb)) return false;
   354→      if (!passesBound(r.divYield, f.yield)) return false;
   355→      if (!passesBound(r.roe ?? null, f.roe)) return false;
   356→      if (!passesBound(r.debtToEquity ?? null, f.de)) return false;
   357→      if (!passesBound(r.eps, f.eps)) return false;
   358→      if (!passesMin(r.volume, f.volumeMin)) return false;
   359→      if (!passesMin(r.valueTraded != null ? r.valueTraded / 1e6 : null, f.valueMin)) return false;
   360→      if (!passesMin(r.volumeRatio, f.volRatioMin)) return false;
   361→      if (f.range52) {
   362→        const pos = rangePos(r);
   363→        if (pos === null) return false;
   364→        if (f.range52 === "high" && pos < 0.95) return false;
   365→        if (f.range52 === "low" && pos > 0.1) return false;
   366→      }
   367→      return true;
   368→    });
   369→    // sort: rows missing the sort metric sink to the bottom either direction
   370→    const col = COLUMNS.find((c) => c.key === sortKey)!;
   371→    const withVal = out.filter((r) => col.val(r) !== null && col.val(r) !== undefined);
   372→    const without = out.filter((r) => col.val(r) === null || col.val(r) === undefined);
   373→    const sorted = [...withVal].sort((a, b) => {
   374→      const va = col.val(a);
   375→      const vb = col.val(b);
   376→      let cmp: number;
   377→      if (typeof va === "string" || typeof vb === "string") {
   378→        cmp = String(va).localeCompare(String(vb));
   379→      } else {
   380→        cmp = (va as number) - (vb as number);
   381→      }
   382→      return desc ? -cmp : cmp;
   383→    });
   384→    return [...sorted, ...without];
   385→  }, [rows, f, sortKey, desc]);
   386→
   387→  // active-filter chips
   388→  const chips: { id: string; label: string; clear: () => void }[] = [];
   389→  if (f.q.trim()) chips.push({ id: "q", label: `"${f.q.trim()}"`, clear: () => applyPatch({ q: "" }) });
   390→  if (f.sector) {
   391→    const s = sectors.find((x) => x.code === f.sector);
   392→    if (s) chips.push({ id: "sector", label: s.name, clear: () => applyPatch({ sector: "" }) });
   393→  }
   394→  const boundDefs: [keyof Filters, string][] = [
   395→    ["price", tt(T.filterPrice, lang)],
   396→    ["change", tt(T.filterChange, lang)],
   397→    ["pe", tt(T.filterPe, lang)],
   398→    ["pb", tt(T.filterPb, lang)],
   399→    ["yield", tt(T.filterYield, lang)],
   400→    ["roe", tt(T.filterRoe, lang)],
   401→    ["de", tt(T.filterDe, lang)],
   402→    ["eps", tt(T.filterEps, lang)],
   403→    ["cap", tt(T.filterCap, lang)],
   404→  ];
   405→  boundDefs.forEach(([key, label]) => {
   406→    const b = f[key] as Bound;
   407→    if (b.min || b.max) {
   408→      chips.push({
   409→        id: `b-${key}`,
   410→        label: boundChipLabel(label, b),
   411→        clear: () => applyPatch({ [key]: EMPTY_BOUND } as Partial<Filters>),
   412→      });
   413→    }
   414→  });
   415→  if (f.perf.min || f.perf.max) {
   416→    const periodLabel = tt(PERF_OPTIONS.find(([p]) => p === f.perfPeriod)![1], lang);
   417→    chips.push({
   418→      id: "perf",
   419→      label: boundChipLabel(`${tt(T.filterPerf, lang)} (${periodLabel})`, f.perf),
   420→      clear: () => applyPatch({ perf: EMPTY_BOUND }),
   421→    });
   422→  }
   423→  if (f.volumeMin) chips.push({ id: "vol", label: `${tt(T.filterVolume, lang)} ≥ ${f.volumeMin}`, clear: () => applyPatch({ volumeMin: "" }) });
   424→  if (f.valueMin) chips.push({ id: "val", label: `${tt(T.filterValue, lang)} ≥ ${f.valueMin}`, clear: () => applyPatch({ valueMin: "" }) });
   425→  if (f.volRatioMin) chips.push({ id: "vr", label: `${tt(T.filterVolRatio, lang)} ≥ ${f.volRatioMin}`, clear: () => applyPatch({ volRatioMin: "" }) });
   426→  if (f.range52) {
   427→    chips.push({
   428→      id: "r52",
   429→      label: f.range52 === "high" ? tt(T.nearHigh, lang) : tt(T.nearLow, lang),
   430→      clear: () => applyPatch({ range52: "" }),
   431→    });
   432→  }
   433→
   434→  const matchCount = filtered?.length ?? 0;
   435→  const headerSort = (c: ColDef) => {
   436→    if (sortKey === c.key) setDesc(!desc);
   437→    else {
   438→      setSortKey(c.key);
   439→      setDesc(true);
   440→    }
   441→  };
   442→
   443→  return (
   444→    <div className="space-y-4">
   445→      {/* title */}
   446→      <div className="flex items-baseline justify-between flex-wrap gap-2">
   447→        <h1 className="text-2xl font-bold tracking-tight">{tt(T.screenerTitle, lang)}</h1>
   448→        {filtered && (
   449→          <p className="num text-xs text-muted-foreground">
   450→            <span className="font-semibold text-foreground">{matchCount}</span> {tt(T.screenerMatch, lang)}{" "}
   451→            {total} · {data?.session.lastSession} · {tt(T.delayed, lang)}
   452→          </p>
   453→        )}
   454→      </div>
   455→      <p className="text-xs text-muted-foreground -mt-2">{tt(T.screenerNote, lang)}</p>
   456→
   457→      {/* presets */}
   458→      <div className="flex items-center gap-1.5 flex-wrap">
   459→        <span className="inline-flex items-center gap-1 text-[11px] text-muted-foreground me-1">
   460→          <Zap className="h-3 w-3" />
   461→        </span>
   462→        {PRESETS.map((p) => {
   463→          const active =
   464→            ("change" in p.patch &&
   465→              p.patch.change?.min === f.change.min &&
   466→              p.patch.change?.max === f.change.max) ||
   467→            ("yield" in p.patch && p.patch.yield?.min === f.yield.min) ||
   468→            ("pe" in p.patch && p.patch.pe?.max === f.pe.max) ||
   469→            ("volRatioMin" in p.patch && p.patch.volRatioMin === f.volRatioMin) ||
   470→            ("cap" in p.patch && p.patch.cap?.min === f.cap.min);
   471→          return (
   472→            <button
   473→              key={p.key}
   474→              onClick={() =>
   475→                active
   476→                  ? setF(DEFAULT_FILTERS)
   477→                  : applyPatch(p.patch)
   478→              }
   479→              className={`rounded-full border px-2.5 py-1 text-[11px] transition-colors ${
   480→                active
   481→                  ? "bg-secondary font-semibold border-ring"
   482→                  : "text-muted-foreground hover:bg-accent/50 hover:text-foreground"
   483→              }`}
   484→            >
   485→              {tt(p.t, lang)}
   486→            </button>
   487→          );
   488→        })}
   489→      </div>
   490→
   491→      {/* filter panel */}
   492→      <div className="rounded-lg border bg-card">
   493→        <button
   494→          onClick={() => setPanelOpen(!panelOpen)}
   495→          className="flex w-full items-center justify-between gap-2 px-3 py-2.5 text-sm"
   496→          aria-expanded={panelOpen}
   497→        >
   498→          <span className="flex items-center gap-2 font-medium">
   499→            <SlidersHorizontal className="h-3.5 w-3.5 text-muted-foreground" />
   500→            {tt(T.screenerFilters, lang)}
   501→            {chips.length > 0 && (
   502→              <span className="num rounded-full bg-secondary px-1.5 text-[10px] font-semibold">{chips.length}</span>
   503→            )}
   504→          </span>
   505→          <span className="flex items-center gap-2">
   506→            {chips.length > 0 && (
   507→              <span
   508→                role="button"
   509→                tabIndex={0}
