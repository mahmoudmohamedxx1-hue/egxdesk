    31→const RATE_LIMIT = 60; // requests per hour (long uncapped answers + web search)
    32→const rateMap = new Map<string, number[]>();
    33→
    34→function rateLimited(ip: string): boolean {
    35→  const now = Date.now();
    36→  const window = 60 * 60_000;
    37→  const arr = (rateMap.get(ip) ?? []).filter((t) => now - t < window);
    38→  if (arr.length >= RATE_LIMIT) {
    39→    rateMap.set(ip, arr);
    40→    return true;
    41→  }
    42→  arr.push(now);
    43→  rateMap.set(ip, arr);
    44→  if (rateMap.size > 500) {
    45→    // drop stale entries so the map can never grow unbounded
    46→    for (const [k, v] of rateMap) if (v.every((t) => now - t >= window)) rateMap.delete(k);
    47→  }
    48→  return false;
    49→}

    77→function clampLimit(raw: unknown, max: number, dflt: number): number {
    78→  const n = typeof raw === "number" ? Math.floor(raw) : Number(raw);
    79→  if (!Number.isFinite(n) || n <= 0) return dflt;
    80→  return Math.min(n, max);

    83→/** First balanced JSON object in the model output (handles ```json fences),
    84→ *  with two repair passes for common LLM quirks: control characters (raw
    85→ *  newlines) inside string literals, and replies truncated mid-string (no
    86→ *  closing brace) — the tail is then recovered with a regex. */
    87→function tryParse(s: string): Record<string, unknown> | null {
    88→  try {
    89→    const p = JSON.parse(s);
    90→    return p && typeof p === "object" ? (p as Record<string, unknown>) : null;
    91→  } catch {
    92→    return null;
    93→  }
    94→}
    95→
    96→/** Escape raw \n \r \t that appear INSIDE string literals (invalid JSON that
    97→ *  models sometimes emit); whitespace outside strings is left alone. */
    98→function repairControlChars(s: string): string {
    99→  let out = "";
   100→  let inStr = false;
   101→  let esc = false;
   102→  for (const ch of s) {
   103→    if (esc) {
   104→      out += ch;
   105→      esc = false;
   106→      continue;
   107→    }
   108→    if (ch === "\\") {
   109→      out += ch;
   110→      esc = true;
   111→      continue;
   112→    }
   113→    if (ch === '"') {
   114→      inStr = !inStr;
   115→      out += ch;
   116→      continue;
   117→    }
   118→    if (inStr && (ch === "\n" || ch === "\r" || ch === "\t")) {
   119→      out += "\\n";
   120→      continue;
   121→    }
   122→    out += ch;
   123→  }
   124→  return out;
   125→}
   126→
   127→/** Extract every balanced top-level {...} span in order (used by extractJson
   128→ *  so reasoning prose around the reply object is ignored). */
   129→function* topLevelJsonObjects(s: string): Generator<string> {
   130→  let i = 0;
   131→  while (i < s.length) {
   132→    const start = s.indexOf("{", i);
   133→    if (start === -1) return;
   134→    let depth = 0;
   135→    let inStr = false;
   136→    let esc = false;
   137→    let closed = false;
   138→    for (let j = start; j < s.length; j++) {
   139→      const c = s[j];
   140→      if (esc) {
   141→        esc = false;
   142→        continue;
   143→      }
   144→      if (c === "\\") {
   145→        if (inStr) esc = true;
   146→        continue;
   147→      }
   148→      if (c === '"') {
   149→        inStr = !inStr;
   150→        continue;
   151→      }
   152→      if (inStr) continue;
   153→      if (c === "{") depth++;
   154→      else if (c === "}") {
   155→        depth--;
   156→        if (depth === 0) {
   157→          yield s.slice(start, j + 1);
   158→          i = j + 1;
   159→          closed = true;
   160→          break;
   161→        }
   162→      }
   163→    }
   164→    if (!closed) return; // truncated from here on — stop
   165→  }
   166→}

   168→function extractJson(raw: string): Record<string, unknown> | null {
   169→  if (!raw) return null;
   170→  let s = raw.trim();
   171→  // thinking-enabled models sometimes inline chain-of-thought — strip it first
   172→  s = s.replace(/<think>[\s\S]*?<\/think>/gi, "");
   173→  const fence = s.match(/```(?:json)?\s*([\s\S]*?)```/i);
   174→  if (fence) s = fence[1].trim();
   175→
   176→  // 1) scan every balanced object; the FIRST one that follows the reply
   177→  //    protocol (has "tool" or "final") wins — prose or examples before it
   178→  //    are ignored. A parsed non-protocol object is kept as a last resort so
   179→  //    the caller can issue a format correction (original behavior).
   180→  let firstParsed: Record<string, unknown> | null = null;
   181→  for (const span of topLevelJsonObjects(s)) {
   182→    const p = tryParse(span) ?? tryParse(repairControlChars(span));
   183→    if (!p) continue;
   184→    if (firstParsed === null) firstParsed = p;
   185→    if ("tool" in p || "final" in p) return p;
   186→  }
   187→  if (firstParsed) return firstParsed;
   188→
   189→  // 2) truncated reply (no closing brace): recover {"final": "… from the tail
   190→  const mFinal = s.match(/"final"\s*:\s*"([\s\S]*)/);
   191→  if (mFinal) {
   192→    const tail = mFinal[1].replace(/"\s*\}?\s*$/, "").trim();
   193→    if (tail.length > 0) return { final: tail };
   194→  }
   195→
   196→  // 3) a tool call that survived truncation: {"tool": "name"…
   197→  const mTool = s.match(/"tool"\s*:\s*"([A-Za-z_][A-Za-z0-9_]*)"/);
   198→  if (mTool) {
   199→    const mArgs = s.match(/"args"\s*:\s*(\{[\s\S]*?)[}]?\s*$/);
   200→    let args: Record<string, unknown> = {};
   201→    if (mArgs) {
   202→      const repaired = repairControlChars(`${mArgs[1]}}`);
   203→      const parsed = tryParse(repaired);
   204→      if (parsed) args = parsed;
   205→    }
   206→    return { tool: mTool[1], args };
   207→  }
   208→  return null;
   209→}

   600→const MAX_TOOL_CALLS = 10;

   602→export async function POST(req: Request) {
   603→  const ip =
   604→    req.headers.get("x-forwarded-for")?.split(",")[0]?.trim() ||
   605→    req.headers.get("x-real-ip") ||
   606→    "local";
   607→  if (rateLimited(ip)) {
   608→    return NextResponse.json({ error: "rate limited — try again later" }, { status: 429 });
   609→  }
   610→
   611→  // validate the conversation the client sends (last 12 turns, last is user)
   612→  let body: { messages?: unknown; lang?: unknown; debug?: unknown };
   613→  try {
   614→    body = (await req.json()) as { messages?: unknown; lang?: unknown };
   615→  } catch {
   616→    return NextResponse.json({ error: "invalid json" }, { status: 400 });
   617→  }
   618→  const lang = body.lang === "en" ? "en" : "ar";
   619→  const rawMsgs = Array.isArray(body.messages) ? body.messages : [];
   620→  const history: ChatMsg[] = rawMsgs
   621→    .filter(
   622→      (m): m is ChatMsg =>
   623→        m !== null &&
   624→        typeof m === "object" &&
   625→        ((m as ChatMsg).role === "user" || (m as ChatMsg).role === "assistant") &&
   626→        typeof (m as ChatMsg).content === "string" &&
   627→        (m as ChatMsg).content.length > 0 &&
   628→        (m as ChatMsg).content.length <= 8000
   629→    )
   630→    .slice(-24);
   631→  if (history.length === 0 || history[history.length - 1].role !== "user") {
   632→    return NextResponse.json({ error: "messages required (last must be user)" }, { status: 400 });
   633→  }
   634→
   635→  const zai = await getZai();
   636→  const msgs: { role: "user" | "assistant"; content: string }[] = [
   637→    { role: "assistant", content: systemPrompt(lang) },
   638→    ...history.map((m) => ({ role: m.role, content: m.content })),
   639→  ];
   640→  const steps: AgentStep[] = [];
   641→  let corrections = 0;
   642→  const debugRaw: string[] = [];
   643→  const debug = body.debug === true;
   644→
   645→  for (let round = 0; round < MAX_TOOL_CALLS + 3; round++) {
   646→    if (steps.length >= MAX_TOOL_CALLS) break;
   647→    let raw = "";
   648→    try {
   649→      // reasoning depth: round 0 (fast JSON tool-picking or a quick
   650→      // conversational reply) runs with thinking off; every later round —
   651→      // i.e. once real tool data is on the table, the synthesis moment — runs
   652→      // with chain-of-thought ON so the final answer is genuine reasoning,
   653→      // not a shallow template.
   654→      const deep = round > 0;
   655→      const completion = await zai.chat.completions.create({
   656→        messages: msgs,
   657→        thinking: { type: deep ? "enabled" : "disabled" },
   658→      });
   659→      raw = completion.choices[0]?.message?.content ?? "";
   660→    } catch (err) {
   661→      return NextResponse.json(
   662→        { error: "model unavailable", detail: err instanceof Error ? err.message : "unknown" },
   663→        { status: 502 }
   664→      );
   665→    }
   666→    if (debug) debugRaw.push(raw.slice(0, 800));
   667→
   668→    const parsed = extractJson(raw);
   669→    if (!parsed || (!("tool" in parsed) && !("final" in parsed))) {
   670→      corrections++;
   671→      if (corrections > 2) break;
   672→      msgs.push({
   673→        role: "user",
   674→        content:
   675→          'Format error. Reply with exactly ONE JSON object, no fences: {"tool": "<name>", "args": {...}} to call a tool, or {"final": "<markdown answer>"} to answer.',
   676→      });
   677→      continue;
   678→    }
   679→
   680→    if (typeof parsed.final === "string" && parsed.final.trim().length > 0) {
   681→      return NextResponse.json(
   682→        { answer: parsed.final.trim(), steps, model: "GLM", disclaimer: true, ...(debug ? { debugRaw } : {}) },
   683→        { headers: { "Cache-Control": "no-store" } }
   684→      );
   685→    }
   686→
   687→    const toolName = typeof parsed.tool === "string" ? parsed.tool : "";
   688→    const tool = TOOLS.find((t) => t.name === toolName);
   689→    if (!tool) {
   690→      msgs.push({
   691→        role: "user",
   692→        content: `Unknown tool "${toolName}". Available tools: ${TOOLS.map((t) => t.name).join(", ")}.`,
   693→      });
   694→      continue;
   695→    }
   696→
   697→    const args = (parsed.args && typeof parsed.args === "object" ? parsed.args : {}) as Record<string, unknown>;
   698→    let result: unknown;
   699→    try {
   700→      result = await tool.run(args);
   701→    } catch (err) {
   702→      result = { error: err instanceof Error ? err.message : "tool failed" };
   703→    }
   704→    const ok = !(result && typeof result === "object" && "error" in (result as Record<string, unknown>));
   705→    steps.push({ tool: toolName, args, ok });
   706→
   707→    msgs.push({ role: "user", content: JSON.stringify(result).slice(0, 9000) });
   708→  }
   709→
   710→  // loop exhausted without a final answer — NEVER waste the collected data:
   711→  // force one synthesis round from the tool results already in context
   712→  if (steps.length > 0) {
   713→    msgs.push({
   714→      role: "user",
   715→      content:
   716→        'Tool budget exhausted. Reply NOW with your final answer using ONLY the tool data collected above — do not request more tools; if a requested stock was not found, say so plainly. Format: {"final": "<markdown answer>"}',
   717→    });
   718→    try {
   719→      const completion = await zai.chat.completions.create({
   720→        messages: msgs,
   721→        thinking: { type: "enabled" },
   722→      });
   723→      const raw = completion.choices[0]?.message?.content ?? "";
   724→      if (debug) debugRaw.push(raw.slice(0, 800));
   725→      const parsed = extractJson(raw);
   726→      if (parsed && typeof parsed.final === "string" && parsed.final.trim().length > 0) {
   727→        return NextResponse.json(
   728→          { answer: parsed.final.trim(), steps, model: "GLM", disclaimer: true, ...(debug ? { debugRaw } : {}) },
   729→          { headers: { "Cache-Control": "no-store" } }
   730→        );
   731→      }
   732→    } catch {
   733→      // fall through to the honest fallback below
   734→    }
   735→  }
   736→
   737→  // no tools ever ran / synthesis also failed — honest fallback, never invented
   738→  const fallback =
   739→    lang === "ar"
   740→      ? "وصلتُ لحد الأدوات المتاحة دون إجابة كاملة. جرّب إعادة السؤال بصيغة أبسط (مثال: «ما حالة السوق الآن؟» أو «quote لسهم COMI»)."
   741→      : "I ran out of tool budget without a complete answer. Try rephrasing (e.g. \"market overview\" or \"quote for COMI\").";
   742→  return NextResponse.json(
   743→    { answer: fallback, steps, model: "GLM", disclaimer: true, ...(debug ? { debugRaw } : {}) },
   744→    { headers: { "Cache-Control": "no-store" } }
   745→  );
   746→}
