     1→"use client";
     2→
     3→import { useState } from "react";
     4→import { useApp } from "../market/app-context";
     5→import { useLiveData } from "../market/use-live-data";
     6→import type { CompanyRow, NewsRow, SessionMeta } from "../market/types";
     7→import { T, tt, dn } from "@/lib/i18n";
     8→import { fmtNum, fmtValue, fmtPct, fmtInt, fmtDateAr, fmtTimeAr, directionClass } from "@/lib/format";
     9→import { WatchStar } from "../market/watch-star";
    10→import { ChangeCell } from "../market/change-cell";
    11→import { PerfChart, RangeBar } from "../market/perf-chart";
    12→import { PriceChart } from "../market/price-chart";
    13→import { StatementsPanel } from "../market/statements-panel";
    14→import { Skeleton } from "@/components/ui/skeleton";
    15→import { Button } from "@/components/ui/button";
    16→import { Volume2, Calculator, TrendingUp, TrendingDown, ExternalLink, RefreshCw, Zap, CalendarClock, FileSpreadsheet } from "lucide-react";
    17→
    18→type CompanyData = {
    19→  session: SessionMeta;
    20→  company: CompanyRow & {
    21→    industry: string | null;
    22→    netMarginTTM: number | null;
    23→    revenueTTM: number | null;
    24→    floatShares: number | null;
    25→    avgTurnover30: number | null;
    26→    high1M: number | null;
    27→    low1M: number | null;
    28→    beta: number | null;
    29→    updateMode: string | null;
    30→    employees?: number | null;
    31→  };
    32→  sectorAgg: {
    33→    count: number;
    34→    nameAr: string;
    35→    nameEn: string;
    36→    pe: number | null;
    37→    pb: number | null;
    38→    roe: number | null;
    39→    eps: number | null;
    40→    divYield: number | null;
    41→    beta: number | null;
    42→    avgChangePct: number | null;
    43→  };
    44→  peers: CompanyRow[];
    45→  news: NewsRow[];
    46→  disclosures: NewsRow[];
    47→  signals: {
    48→    streak: { direction: "up" | "down"; count: number; since: string } | null;
    49→    nextEarnings: string | null;
    50→    unusualVolume: boolean;
    51→    volumeRatio: number | null;
    52→    near52High: boolean;
    53→    near52Low: boolean;
    54→    computedFrom: string;
    55→  } | null;
    56→};
    57→
    58→export function CompanyView({ ticker, panel }: { ticker: string; panel: string }) {
    59→  const { lang, navigate } = useApp();
    60→  const { data, error, refresh } = useLiveData<CompanyData>(`/api/company/${encodeURIComponent(ticker)}`, 60_000);
    61→  const [activePanel, setActivePanel] = useState(panel);
    62→  const [speaking, setSpeaking] = useState(false);
    63→  const [prevPanel, setPrevPanel] = useState(panel);
    64→  if (prevPanel !== panel) {
    65→    // URL panel changed (e.g. navigating from a peer link) — adjust state during render
    66→    setPrevPanel(panel);
    67→    setActivePanel(panel);
    68→  }
    69→
    70→  if (error && !data) {
    71→    return (
    72→      <div className="rounded-lg border bg-card p-8 text-center space-y-3">
    73→        <p className="font-medium">{tt(T.errorLoad, lang)}</p>
    74→        <Button size="sm" onClick={refresh}>
    75→          <RefreshCw className="h-3.5 w-3.5 me-1.5" />
    76→          {tt(T.retry, lang)}
    77→        </Button>
    78→      </div>
    79→    );
    80→  }
    81→
    82→  if (!data || !data.company) {
    83→    if (data && (data as { error?: string }).error === "no such company") {
    84→      return (
    85→        <div className="py-16 text-center space-y-2">
    86→          <p className="text-lg font-semibold">{lang === "ar" ? "لا توجد شركة بهذا الرمز" : "No such company"}</p>
    87→          <button onClick={() => navigate("market")} className="text-sm text-primary hover:underline">
    88→            {tt(T.browseMarket, lang)}
    89→          </button>
    90→        </div>
    91→      );
    92→    }
    93→    return (
    94→      <div className="space-y-4">
    95→        <Skeleton className="h-16 w-full" />
    96→        <Skeleton className="h-48 w-full" />
    97→        <Skeleton className="h-64 w-full" />
    98→      </div>
    99→    );
   100→  }
   101→
   102→  const c = data.company;
   103→
   104→  function speakQuote() {
   105→    if (typeof window === "undefined" || !window.speechSynthesis) return;
   106→    if (speaking) { window.speechSynthesis.cancel(); setSpeaking(false); return; }
   107→    const dir = c.changePct >= 0
   108→      ? (lang === "ar" ? "مرتفع" : "up")
   109→      : (lang === "ar" ? "منخفض" : "down");
   110→    const text = lang === "ar"
   111→      ? `${c.ticker}. ${dn(c, lang)}. آخر سعر ${fmtNum(c.close)} جنيه، ${dir} بنسبة ${fmtNum(Math.abs(c.changePct))} بالمئة.`
   112→      : `${c.ticker}. ${c.name}. Last ${fmtNum(c.close)} Egyptian pounds, ${dir} ${fmtNum(Math.abs(c.changePct))} percent.`;
   113→    const u = new SpeechSynthesisUtterance(text);
   114→    u.lang = lang === "ar" ? "ar-EG" : "en-US";
   115→    u.onend = () => setSpeaking(false);
   116→    u.onerror = () => setSpeaking(false);
   117→    window.speechSynthesis.speak(u);
   118→    setSpeaking(true);
   119→  }
   120→
   121→  const panels = [
   122→    { key: "overview", t: T.panelOverview },
   123→    { key: "statements", t: T.panelStatements },
   124→    { key: "fundamentals", t: T.panelFundamentals },
   125→    { key: "disclosures", t: T.panelDisclosures },
   126→    { key: "activity", t: T.panelActivity },
   127→    { key: "news", t: T.panelNews },
   128→  ];
   129→
   130→  const perf = [
   131→    { label: "1W", value: c.perfW },
   132→    { label: "1M", value: c.perf1M },
   133→    { label: "3M", value: c.perf3M },
   134→    { label: "6M", value: c.perf6M },
   135→    { label: "YTD", value: c.perfYTD },
   136→    { label: "1Y", value: c.perfY },
   137→    { label: "3Y", value: c.perf3Y },
   138→    { label: "5Y", value: c.perf5Y },
   139→  ];
   140→
   141→  return (
   142→    <div className="space-y-5">
   143→      {/* ticker row */}
   144→      <div className="flex items-center flex-wrap gap-2">
   145→        <span className="num text-lg font-bold">{c.ticker}</span>
   146→        <WatchStar ticker={c.ticker} />
   147→        <button
   148→          onClick={speakQuote}
   149→          className="inline-flex items-center gap-1.5 rounded-md border px-2.5 py-1.5 text-xs hover:bg-accent/50 transition-colors"
   150→        >
   151→          <Volume2 className={`h-3.5 w-3.5 ${speaking ? "animate-pulse" : ""}`} />
   152→          {tt(T.listenBrief, lang)}
   153→        </button>
   154→        <button
   155→          onClick={() => navigate("tools")}
   156→          className="inline-flex items-center gap-1.5 rounded-md border px-2.5 py-1.5 text-xs hover:bg-accent/50 transition-colors"
   157→        >
   158→          <Calculator className="h-3.5 w-3.5" />
   159→          {lang === "ar" ? "احسب عائد الكوبون" : "Compute coupon return"}
   160→        </button>
   161→      </div>
   162→
   163→      {/* identity + quote */}
   164→      <div className="rounded-lg border bg-card p-4 sm:p-5">
   165→        <div className="flex flex-wrap items-start justify-between gap-4">
   166→          <div className="min-w-0 space-y-1">
   167→            <p className="text-xs text-muted-foreground">
   168→              {lang === "ar" ? c.sectorAr : c.sectorEn}
   169→              {c.industry ? ` · ${c.industry}` : ""}
   170→            </p>
   171→            <h1 className="text-xl sm:text-2xl font-bold tracking-tight">{dn(c, lang)}</h1>
   172→            {lang === "ar" && c.nameAr && c.nameAr !== c.name && (
   173→              <p className="text-xs text-muted-foreground truncate">{c.name}</p>
   174→            )}
   175→            {lang === "en" && c.nameAr && (
   176→              <p className="text-xs text-muted-foreground truncate" dir="rtl">{c.nameAr}</p>
   177→            )}
   178→            <p className="num text-xs text-muted-foreground">EGX · EGP · {tt(T.delayed, lang)}</p>
   179→          </div>
   180→          <div className="text-end space-y-0.5">
   181→            <p className="text-xs text-muted-foreground">{tt(T.lastClose, lang)}</p>
   182→            <p className="num text-3xl font-bold tracking-tight">{fmtNum(c.close)}</p>
   183→            <p className={`num text-sm font-semibold ${directionClass(c.changePct)} flex items-center gap-1.5 justify-end`}>
   184→              {c.changePct >= 0 ? <TrendingUp className="h-3.5 w-3.5" /> : <TrendingDown className="h-3.5 w-3.5" />}
   185→              {c.changePct >= 0 ? "+" : ""}{fmtNum(c.changeAbs)} ({fmtPct(c.changePct)})
   186→            </p>
   187→            <p className="num text-[10px] text-muted-foreground">{data.session.lastSession}</p>
   188→          </div>
   189→        </div>
   190→
   191→        {/* stat strip */}
   192→        <div className="mt-4 grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-6 gap-x-4 gap-y-3 border-t pt-4">
   193→          <Stat label={tt(T.marketCap, lang)} value={`EGP ${fmtValue(c.marketCap)}`} />
   194→          <Stat label={tt(T.week, lang)} value={fmtPct(c.perfW)} cls={directionClass(c.perfW)} />
   195→          <Stat label={tt(T.month, lang)} value={fmtPct(c.perf1M)} cls={directionClass(c.perf1M)} />
   196→          <Stat label={tt(T.ytd, lang)} value={fmtPct(c.perfYTD)} cls={directionClass(c.perfYTD)} />
   197→          <Stat label={tt(T.volume, lang)} value={fmtInt(c.volume)} sub={`${tt(T.inSession, lang)} · ${tt(T.avg10, lang)} ${fmtInt(c.avgVolume)}`} />
   198→          <Stat label="P/E" value={c.pe ? fmtNum(c.pe, 1) : "—"} sub={c.eps ? `EPS ${fmtNum(c.eps)}` : undefined} />
   199→        </div>
   200→      </div>
   201→
   202→      {/* panel tabs */}
   203→      <div className="flex items-center gap-1 overflow-x-auto thin-scroll border-b" role="tablist">
   204→        {panels.map((p) => (
   205→          <button
   206→            key={p.key}
   207→            role="tab"
   208→            aria-selected={activePanel === p.key}
   209→            onClick={() => setActivePanel(p.key)}
   210→            className={`whitespace-nowrap px-3 py-2 text-sm -mb-px border-b-2 transition-colors ${
   211→              activePanel === p.key ? "border-primary font-semibold" : "border-transparent text-muted-foreground hover:text-foreground"
   212→            }`}
   213→          >
   214→            {tt(p.t, lang)}
   215→            {p.key === "news" && data.news.length > 0 && (
   216→              <span className="num ms-1.5 text-xs text-muted-foreground">{data.news.length}</span>
   217→            )}
   218→            {p.key === "disclosures" && data.disclosures.length > 0 && (
   219→              <span className="num ms-1.5 text-xs text-muted-foreground">{data.disclosures.length}</span>
   220→            )}
   221→          </button>
   222→        ))}
   223→      </div>
   224→
   225→      {/* OVERVIEW */}
   226→      {activePanel === "overview" && (
   227→        <div className="space-y-5">
   228→          <section className="rounded-lg border bg-card p-4">
   229→            <div className="flex items-center justify-between mb-3 flex-wrap gap-2">
   230→              <h2 className="font-bold">{tt(T.priceChart, lang)}</h2>
   231→              <span className="num text-xs text-muted-foreground">{c.ticker} · EGX</span>
   232→            </div>
   233→            <PriceChart key={c.ticker} symbol={c.ticker} defaultRange="6M" />
   234→          </section>
   235→
   236→          {data.signals && <SignalsCard signals={data.signals} lang={lang} />}
   237→
   238→          <section className="rounded-lg border bg-card p-4">
   239→            <div className="flex items-center justify-between mb-3 flex-wrap gap-2">
   240→              <h2 className="font-bold">{tt(T.performance, lang)}</h2>
   241→            </div>
   242→            <PerfChart perf={perf} lang={lang} />
   243→            <p className="mt-2 text-[11px] text-muted-foreground">{tt(T.perfNote, lang)}</p>
   244→          </section>
   245→
   246→          <section className="rounded-lg border bg-card p-4">
   247→            <h2 className="font-bold mb-3">{tt(T.range52, lang)}</h2>
   248→            <RangeBar
   249→              low={c.low52}
   250→              high={c.high52}
   251→              close={c.close}
   252→              labelLow={lang === "ar" ? "أدنى ٥٢ أسبوعاً" : "52-week low"}
   253→              labelHigh={lang === "ar" ? "أعلى ٥٢ أسبوعاً" : "52-week high"}
   254→            />
   255→            {c.high1M !== null && c.low1M !== null && (
   256→              <p className="num mt-3 text-[11px] text-muted-foreground">
   257→                {lang === "ar" ? "مدى الشهر:" : "1-month range:"} {fmtNum(c.low1M, 1)} – {fmtNum(c.high1M, 1)}
   258→              </p>
   259→            )}
   260→          </section>
   261→        </div>
   262→      )}
   263→
   264→      {/* STATEMENTS */}
   265→      {activePanel === "statements" && (
   266→        <StatementsPanel key={c.ticker} ticker={c.ticker} />
   267→      )}
   268→
   269→      {/* DISCLOSURES (press-derived) */}
   270→      {activePanel === "disclosures" && (
   271→        <section className="rounded-lg border bg-card divide-y">
   272→          <div className="px-4 py-3">
   273→            <h2 className="font-bold flex items-center gap-2">
   274→              <FileSpreadsheet className="h-4 w-4 text-primary" />
   275→              {tt(T.panelDisclosures, lang)}
   276→            </h2>
   277→            <p className="mt-1 text-xs text-muted-foreground max-w-3xl leading-relaxed">{tt(T.disclosuresNote, lang)}</p>
   278→          </div>
   279→          {data.disclosures.length === 0 && (
   280→            <p className="px-4 py-10 text-center text-sm text-muted-foreground">{tt(T.noDisclosures, lang)}</p>
   281→          )}
   282→          {data.disclosures.map((n) => (
   283→            <article key={n.id} className="px-4 py-3">
   284→              <div className="flex items-center gap-2 mb-1 flex-wrap">
   285→                <span className="rounded-sm bg-secondary px-1.5 py-0.5 text-[10px] font-medium">{n.source}</span>
   286→                <span className="num text-[11px] text-muted-foreground">{fmtDateAr(n.publishedAt)} · {fmtTimeAr(n.publishedAt)}</span>
   287→              </div>
   288→              <a href={n.link} target="_blank" rel="noopener noreferrer" className="text-sm leading-snug font-medium hover:underline inline-flex items-start gap-1.5">
   289→                {n.title}
   290→                <ExternalLink className="h-3 w-3 mt-1 shrink-0 text-muted-foreground" />
   291→              </a>
   292→              {n.snippet && <p className="mt-1 text-xs text-muted-foreground line-clamp-2">{n.snippet}</p>}
   293→            </article>
   294→          ))}
   295→        </section>
   296→      )}
   297→
   298→      {/* ACTIVITY */}
   299→      {activePanel === "activity" && (
   300→        <section className="rounded-lg border bg-card p-4 space-y-4">
   301→          <h2 className="font-bold">{tt(T.panelActivity, lang)}</h2>
   302→          <div className="grid grid-cols-2 sm:grid-cols-4 gap-4">
   303→            <Stat label={tt(T.volume, lang)} value={fmtInt(c.volume)} sub={tt(T.inSession, lang)} />
   304→            <Stat label={tt(T.avg10, lang)} value={fmtInt(c.avgVolume)} />
   305→            <Stat label={tt(T.valueTraded, lang)} value={`EGP ${fmtValue(c.valueTraded)}`} sub={tt(T.inSession, lang)} />
   306→            <Stat label={lang === "ar" ? "الحجم ÷ المعتاد" : "Vol ÷ usual"} value={c.volumeRatio !== null ? `${fmtNum(c.volumeRatio, 1)}×` : "—"} sub={tt(T.inSession, lang)} />
   307→          </div>
   308→          <div className="rounded-md bg-secondary/70 p-3 flex items-center justify-between">
   309→            <span className="text-xs text-muted-foreground">{lang === "ar" ? "الحجم ÷ المعتاد" : "Volume ÷ usual"}</span>
   310→            <span className="num text-lg font-bold text-primary">
   311→              {c.avgVolume ? fmtNum(c.volume / c.avgVolume, 1) + "×" : "—"}
   312→            </span>
   313→          </div>
   314→          <div>
   315→            <div className="flex items-center justify-between mb-1 text-xs">
   316→              <span className="text-muted-foreground">{lang === "ar" ? "مقابل متوسط ١٠ جلسات" : "vs 10-session average"}</span>
   317→              <span className="num">{fmtInt(c.volume)} / {fmtInt(c.avgVolume)}</span>
   318→            </div>
   319→            <div className="h-2 rounded-full bg-secondary overflow-hidden" dir="ltr">
   320→              <div className="h-full rounded-full bg-primary" style={{ width: `${Math.min((c.volume / Math.max(c.avgVolume ?? 1, 1)) * 50, 100)}%` }} />
   321→            </div>
   322→          </div>
   323→          {c.avgTurnover30 !== null && (
   324→            <Stat label={lang === "ar" ? "متوسط قيمة التداول (٣٠ يوماً)" : "Avg turnover (30d)"} value={`EGP ${fmtValue(c.avgTurnover30)}`} />
   325→          )}
   326→          <p className="text-[11px] text-muted-foreground">{tt(T.unusualNote, lang)}</p>
   327→        </section>
   328→      )}
   329→
   330→      {/* FUNDAMENTALS */}
   331→      {activePanel === "fundamentals" && (
   332→        <div className="space-y-4">
   333→          <section className="rounded-lg border bg-card p-4">
   334→            <h2 className="font-bold mb-1">{tt(T.fundamentals, lang)}</h2>
   335→            <p className="text-xs text-muted-foreground mb-3">
   336→              {tt(T.vsSector, lang)} — {lang === "ar" ? data.sectorAgg.nameAr : data.sectorAgg.nameEn}{" "}
   337→              <span className="num">({data.sectorAgg.count} {tt(T.companies, lang)})</span>
   338→            </p>
   339→            <div className="grid grid-cols-2 sm:grid-cols-3 gap-x-4 gap-y-3">
   340→              <VsSector label="P/E" value={c.pe} agg={data.sectorAgg.pe} fmt={(v) => fmtNum(v, 1)} />
   341→              <VsSector label={lang === "ar" ? "م/د (القيمة الدفترية)" : "P/B (book)"} value={c.pb} agg={data.sectorAgg.pb} fmt={(v) => fmtNum(v, 2)} />
   342→              <VsSector label={lang === "ar" ? "العائد على حقوق الملكية" : "ROE"} value={c.roe} agg={data.sectorAgg.roe} fmt={(v) => `${fmtNum(v, 1)}%`} higherBetter />
   343→              <VsSector label="EPS" value={c.eps} agg={data.sectorAgg.eps} fmt={(v) => fmtNum(v)} higherBetter />
   344→              <VsSector label={lang === "ar" ? "عائد التوزيعات" : "Div yield"} value={c.divYield} agg={data.sectorAgg.divYield} fmt={(v) => `${fmtNum(v, 1)}%`} higherBetter />
   345→              <VsSector label={tt(T.beta, lang)} value={c.beta} agg={data.sectorAgg.beta} fmt={(v) => fmtNum(v, 2)} />
   346→              <MetricOnly label={tt(T.netMargin, lang)} value={c.netMarginTTM !== null ? `${fmtNum(c.netMarginTTM, 1)}%` : "—"} />
   347→              <MetricOnly label={lang === "ar" ? "صافي الربح (١٢ شهراً)" : "Net income (TTM)"} value={c.netIncomeTTM !== null ? `EGP ${fmtValue(c.netIncomeTTM)}` : "—"} />
   348→              <MetricOnly label={tt(T.revenueTtm, lang)} value={c.revenueTTM !== null ? `EGP ${fmtValue(c.revenueTTM)}` : "—"} />
   349→              <MetricOnly label={lang === "ar" ? "الدين / حقوق الملكية" : "Debt / equity"} value={c.debtToEquity !== null ? fmtNum(c.debtToEquity, 2) : "—"} />
   350→              <MetricOnly label={lang === "ar" ? "هامش الربح الإجمالي" : "Gross margin"} value={c.grossMarginTTM !== null ? `${fmtNum(c.grossMarginTTM, 1)}%` : "—"} />
   351→              <MetricOnly label={lang === "ar" ? "صافي الدين" : "Net debt"} value={c.netDebt !== null ? `EGP ${fmtValue(c.netDebt)}` : "—"} />
   352→              <MetricOnly label={tt(T.floatShares, lang)} value={c.floatShares !== null ? fmtInt(c.floatShares) : "—"} />
   353→              <MetricOnly label={lang === "ar" ? "الموظفون" : "Employees"} value={c.employees !== null && c.employees !== undefined ? fmtInt(c.employees) : "—"} />
   354→              <MetricOnly label={tt(T.marketCap, lang)} value={`EGP ${fmtValue(c.marketCap)}`} />
   355→            </div>
   356→          </section>
   357→
   358→          {/* peers */}
   359→          <section className="rounded-lg border bg-card overflow-hidden">
   360→            <div className="border-b px-4 py-3">
   361→              <h2 className="font-bold">{tt(T.peers, lang)}</h2>
   362→            </div>
   363→            <div className="overflow-x-auto thin-scroll">
   364→              <table className="w-full text-sm">
   365→                <thead className="border-b bg-card">
   366→                  <tr className="text-[11px] text-muted-foreground">
   367→                    <th className="text-start font-medium px-3 py-2">{tt(T.colTicker, lang)}</th>
   368→                    <th className="text-start font-medium px-3 py-2 hidden md:table-cell">{tt(T.colName, lang)}</th>
   369→                    <th className="text-end font-medium px-3 py-2">{tt(T.colClose, lang)}</th>
   370→                    <th className="text-end font-medium px-3 py-2">{tt(T.colChange, lang)}</th>
   371→                    <th className="text-end font-medium px-3 py-2 hidden sm:table-cell">{tt(T.marketCap, lang)}</th>
   372→                    <th className="text-end font-medium px-3 py-2 hidden md:table-cell">{tt(T.colPe, lang)}</th>
   373→                  </tr>
   374→                </thead>
   375→                <tbody className="divide-y">
   376→                  {data.peers.map((p) => (
   377→                    <tr key={p.ticker} className="hover:bg-accent/30 cursor-pointer transition-colors"
   378→                      onClick={() => navigate("company", { ticker: p.ticker, panel: "fundamentals" })}>

   509→function VsSector({ label, value, agg, fmt, higherBetter }: {
   510→  label: string; value: number | null | undefined; agg: number | null | undefined;
   511→  fmt: (v: number) => string; higherBetter?: boolean;
   512→}) {
   513→  const has = typeof value === "number" && typeof agg === "number";
   514→  let better: boolean | null = null;
   515→  if (has) {
   516→    better = higherBetter ? value > agg : value < agg;
   517→  }
   518→  return (
   519→    <div className="border-b border-dotted pb-2">
   520→      <p className="text-xs text-muted-foreground">{label}</p>
   521→      <div className="flex items-baseline gap-2 flex-wrap">
   522→        <span className="num text-sm font-semibold">{value !== null && value !== undefined ? fmt(value) : "—"}</span>
   523→        {has && (
   524→          <span className={`num text-[10px] px-1 rounded ${better ? "text-up bg-up-soft" : "text-down bg-down-soft"}`}>
   525→            {better ? "▲" : "▼"} {fmt(agg!)}
   526→          </span>
   527→        )}
   528→      </div>
   529→    </div>
   530→  );
   531→}
