   223→  if (pct > 0) return { bg: `oklch(0.55 0.16 150 / ${alpha.toFixed(2)})`, fg: intensity > 0.55 ? "#fff" : "var(--up)" };
   224→  return { bg: `oklch(0.52 0.19 25 / ${alpha.toFixed(2)})`, fg: intensity > 0.55 ? "#fff" : "var(--down)" };
