    87→/** The terracotta asterisk mark — an original 12-ray sunburst that plays the
    88→ *  role of the assistant's avatar (inspired by reference AI chats, not a
    89→ *  copy of any logo). */
    90→function ClaudeMark({ className }: { className?: string }) {
    91→  const rays = Array.from({ length: 12 }, (_, i) => i * 30);
    92→  return (
    93→    <svg viewBox="0 0 24 24" fill="none" className={className} aria-hidden>
    94→      {rays.map((a) => {
    95→        const rad = (a * Math.PI) / 180;
    96→        const x1 = 12 + Math.sin(rad) * 3.4;
    97→        const y1 = 12 - Math.cos(rad) * 3.4;
    98→        const x2 = 12 + Math.sin(rad) * 10.2;
    99→        const y2 = 12 - Math.cos(rad) * 10.2;
   100→        return <line key={a} x1={x1} y1={y1} x2={x2} y2={y2} stroke="currentColor" strokeWidth="2.4" strokeLinecap="round" />;
   101→      })}
   102→    </svg>
   103→  );
   104→}

   170→      <div className="flex items-center gap-2">

   176→      {m.steps && !m.error && <StepChips steps={m.steps} lang={lang} />}

   178→        <p className="text-sm leading-relaxed text-down">{m.content}</p>

   185→        <div className="flex justify-end opacity-0 transition-opacity group-hover:opacity-100 focus-within:opacity-100 max-sm:opacity-60">
   186→          <CopyAnswer text={m.content} lang={lang} />
   187→        </div>
   188→      )}
   189→    </div>
   190→  );
   191→}

   193→export function AgentView() {
   194→  const { lang, toast } = useApp();
   195→  const [messages, setMessages] = useState<AgentMsg[]>([]);
   196→  const [ready, setReady] = useState(false);
   197→  const [input, setInput] = useState("");
   198→  const [busy, setBusy] = useState(false);
   199→  const [deep, setDeep] = useState(false);
   200→  const [elapsed, setElapsed] = useState(0);
   201→  const [liveSteps, setLiveSteps] = useState<AgentStep[]>([]);
   202→  const [liveNote, setLiveNote] = useState<string | null>(null);
   203→  const [streamText, setStreamText] = useState("");
   204→  const [historyOpen, setHistoryOpen] = useState(false);
   205→  const [historyLoading, setHistoryLoading] = useState(false);
   206→  const [historyList, setHistoryList] = useState<HistoryRow[]>([]);
   207→  const [usage, setUsage] = useState<UsageSummary | null>(null);
   208→  const chatIdRef = useRef<string>("");
   209→  const lastQueryRef = useRef<string | null>(null);
   210→  const bottomRef = useRef<HTMLDivElement | null>(null);
   211→  const abortRef = useRef<AbortController | null>(null);
   212→
   213→  // restore the chat from the device (SSR-safe mount read); a shared
   214→  // ?q=… link prefills the composer (never auto-sends — it would burn the
   215→  // recipient's AI quota without their consent)
   216→  useEffect(() => {
   217→    try {
   218→      const raw = localStorage.getItem(CHAT_KEY);
   219→      if (raw) {
   220→        const arr = JSON.parse(raw);
   221→        if (Array.isArray(arr)) {
   222→          setMessages(
   223→            arr
   224→              .filter((m): m is AgentMsg => m && typeof m === "object" && (m.role === "user" || m.role === "assistant") && typeof m.content === "string")
   225→              .slice(-MAX_STORED)
   226→          );
   227→        }
   228→      }
   229→      let id = localStorage.getItem(CHAT_ID_KEY);
   230→      if (!id) {
   231→        id = newChatId();
   232→        localStorage.setItem(CHAT_ID_KEY, id);
   233→      }
   234→      chatIdRef.current = id;
   235→      setDeep(localStorage.getItem(THINK_KEY) === "1");
   236→    } catch {}
   237→    const q = bootParam("q");
   238→    if (q) setInput(q.slice(0, 2000));
   239→    setReady(true);
   240→  }, []);
   241→
   242→  const persist = (next: AgentMsg[]) => {
   243→    setMessages(next);
   244→    try {
   245→      localStorage.setItem(CHAT_KEY, JSON.stringify(next.slice(-MAX_STORED)));
   246→    } catch {}
   247→  };
   248→
   249→  const toggleDeep = () => {
   250→    setDeep((d) => {
   251→      const next = !d;
   252→      try {
   253→        localStorage.setItem(THINK_KEY, next ? "1" : "0");
   254→      } catch {}
   255→      return next;
   256→    });
   257→  };
   258→
   259→  // elapsed-seconds timer while the agent works
   260→  useEffect(() => {
   261→    if (!busy) {
   262→      setElapsed(0);
   263→      return;
   264→    }
   265→    const t = setInterval(() => setElapsed((e) => e + 1), 1000);
   266→    return () => clearInterval(t);
   267→  }, [busy]);
   268→
   269→  // keep the newest message in view (also while the answer streams in)
   270→  useEffect(() => {
   271→    bottomRef.current?.scrollIntoView({ behavior: "smooth", block: "end" });
   272→  }, [messages, busy, liveSteps, streamText]);
   273→
   274→  const ask = async (text: string) => {
   275→    const q = text.trim();
   276→    if (!q || busy) return;
   277→    lastQueryRef.current = q;
   278→    const userMsg: AgentMsg = { role: "user", content: q, ts: Date.now() };
   279→    const history = [...messages, userMsg];
   280→    persist(history);
   281→    setInput("");
   282→    setBusy(true);
   283→    setLiveSteps([]);
   284→    setLiveNote(null);
   285→    setStreamText("");
   286→    const ac = new AbortController();
   287→    abortRef.current = ac;
   288→    let streamSoFar = "";
   289→    try {
   290→      const res = await fetch("/api/agent", {
   291→        method: "POST",
   292→        headers: { "Content-Type": "application/json" },
   293→        signal: ac.signal,
   294→        body: JSON.stringify({
   295→          messages: history.filter((m) => !m.error).map((m) => ({ role: m.role, content: m.content })),
   296→          lang,
   297→          deviceId: getDeviceId(), // usage metering + per-user hourly limit
   298→          deep, // the composer's extended-thinking toggle (Task 22-b)
   299→        }),
   300→      });
   301→      const ct = res.headers.get("content-type") ?? "";
   302→      // plain-JSON error path (429 rate limit / 400 validation / gateway html)
   303→      if (!res.ok || !ct.includes("text/event-stream")) {
   304→        const json = (await res.json().catch(() => ({}))) as { error?: string };
   305→        persist([
   306→          ...history,
   307→          {
   308→            role: "assistant",
   309→            content: json.error ? `${tt(T.agentError, lang)} (${json.error})` : tt(T.agentError, lang),
   310→            error: true,
   311→            ts: Date.now(),
   312→          },
   313→        ]);
   314→        return;
   315→      }
   316→      // SSE stream: tool steps arrive as they run, the answer previews live
   317→      const reader = res.body?.getReader();
   318→      if (!reader) throw new Error("no stream body");
   319→      const dec = new TextDecoder();
   320→      let buf = "";
   321→      let gotTerminal = false;
   322→      const stepsAcc: AgentStep[] = [];
   323→      const handleEvent = (evt: Record<string, unknown>) => {
   324→        if (evt.type === "step") {
   325→          stepsAcc.push({ tool: String(evt.tool ?? ""), args: (evt.args as Record<string, unknown>) ?? {}, ok: evt.ok !== false });
   326→          setLiveSteps([...stepsAcc]);
   327→        } else if (evt.type === "status") {
   328→          setLiveNote(typeof evt.note === "string" ? evt.note : null);
   329→        } else if (evt.type === "delta" && typeof evt.text === "string") {
   330→          streamSoFar += evt.text;
   331→          setStreamText(streamSoFar);
   332→        } else if (evt.type === "done" && typeof evt.answer === "string") {
   333→          gotTerminal = true;
   334→          const finalSteps = Array.isArray(evt.steps) ? (evt.steps as AgentStep[]) : stepsAcc;
   335→          const next: AgentMsg[] = [...history, { role: "assistant", content: evt.answer, steps: finalSteps, ts: Date.now() }];
   336→          persist(next);
   337→          saveChat(next); // server-side history (fire-and-forget)
   338→        } else if (evt.type === "error") {
   339→          gotTerminal = true;
   340→          persist([
   341→            ...history,
   342→            {
   343→              role: "assistant",
   344→              content: evt.message ? `${tt(T.agentError, lang)} (${evt.message})` : tt(T.agentError, lang),
   345→              error: true,
   346→              ts: Date.now(),
   347→            },
   348→          ]);
   349→        }
   350→      };
   351→      for (;;) {
   352→        const { done, value } = await reader.read();
   353→        if (done) break;
   354→        buf += dec.decode(value, { stream: true });
   355→        let sep: number;
   356→        while ((sep = buf.indexOf("\n\n")) !== -1) {
   357→          const rawEvt = buf.slice(0, sep);
   358→          buf = buf.slice(sep + 2);
   359→          for (const line of rawEvt.split("\n")) {
   360→            if (!line.startsWith("data:")) continue;
   361→            try {
   362→              handleEvent(JSON.parse(line.slice(5).trim()) as Record<string, unknown>);
   363→            } catch {
   364→              /* partial line — the next chunk completes it */
   365→            }
   366→          }
   367→        }
   368→      }
   369→      if (!gotTerminal) throw new Error("stream ended without a terminal event");
   370→    } catch (err) {
   371→      // user pressed stop — keep whatever streamed as the (partial) answer
   372→      if (err instanceof DOMException && err.name === "AbortError") {
   373→        if (streamSoFar.trim()) {
   374→          persist([...history, { role: "assistant", content: streamSoFar, ts: Date.now() }]);
   375→        } else {
   376→          persist([...history, { role: "assistant", content: tt(T.agentStopped, lang), error: true, ts: Date.now() }]);
   377→        }
   378→      } else {
   379→        persist([...history, { role: "assistant", content: tt(T.agentError, lang), error: true, ts: Date.now() }]);
   380→      }
   381→    } finally {
   382→      abortRef.current = null;
   383→      setBusy(false);
   384→      setLiveSteps([]);
   385→      setLiveNote(null);
   386→      setStreamText("");
   387→    }
   388→  };
   389→
   390→  const stop = () => {
   391→    abortRef.current?.abort();
   392→  };
   393→
   394→  const retry = () => {
   395→    if (lastQueryRef.current && !busy) void ask(lastQueryRef.current);
   396→  };
   397→
   398→  // ── server-side chat history (auto-saved after every exchange) ──
   399→
   400→  const saveChat = (msgs: AgentMsg[]) => {
   401→    const id = chatIdRef.current;
   402→    if (!id || msgs.length === 0) return;
   403→    const title = (msgs.find((m) => m.role === "user")?.content ?? "").slice(0, 80);
   404→    void fetch("/api/agent/chats", {
   405→      method: "PUT",
   406→      headers: { "Content-Type": "application/json" },
   407→      body: JSON.stringify({
   408→        deviceId: getDeviceId(),
   409→        id,
   410→        title,
   411→        messages: msgs.filter((m) => !m.error).slice(-80),
   412→      }),
   413→    }).catch(() => {}); // best-effort — the next exchange retries
   414→  };
   415→
   416→  const loadHistory = async () => {
   417→    setHistoryLoading(true);
   418→    try {
   419→      const res = await fetch(`/api/agent/chats?deviceId=${encodeURIComponent(getDeviceId())}`);
   420→      const json = (await res.json()) as { chats?: HistoryRow[] };
   421→      setHistoryList(json.chats ?? []);
   422→    } catch {}
   423→    setHistoryLoading(false);
   424→    // usage metering strip (best-effort, never blocks the panel)
   425→    try {
   426→      const res = await fetch("/api/usage");
   427→      if (res.ok) setUsage((await res.json()) as UsageSummary);
   428→    } catch {}
   429→  };
   430→
   431→  const loadChat = async (id: string) => {
   432→    if (busy) return;
   433→    try {
   434→      const res = await fetch(`/api/agent/chats?deviceId=${encodeURIComponent(getDeviceId())}&id=${encodeURIComponent(id)}`);
   435→      if (!res.ok) return;
   436→      const json = (await res.json()) as { messages?: AgentMsg[] };
   437→      const msgs = (json.messages ?? [])
   438→        .filter((m) => m && (m.role === "user" || m.role === "assistant") && typeof m.content === "string")
   439→        .slice(-MAX_STORED);
   440→      persist(msgs);
   441→      chatIdRef.current = id;
   442→      try {
   443→        localStorage.setItem(CHAT_ID_KEY, id);
   444→      } catch {}
   445→      setHistoryOpen(false);
   446→    } catch {}
   447→  };
   448→
   449→  const deleteChat = async (id: string) => {
   450→    try {
   451→      await fetch(`/api/agent/chats?deviceId=${encodeURIComponent(getDeviceId())}&id=${encodeURIComponent(id)}`, {
   452→        method: "DELETE",
   453→      });
   454→      setHistoryList((h) => h.filter((c) => c.id !== id));
   455→      if (chatIdRef.current === id) {
   456→        startNewChat();
   457→        toast(tt(T.agentChatDeleted, lang));
   458→      }
   459→    } catch {}
   460→  };
   461→
   462→  const startNewChat = () => {
   463→    const id = newChatId();
   464→    chatIdRef.current = id;
   465→    try {
   466→      localStorage.setItem(CHAT_ID_KEY, id);
   467→    } catch {}
   468→    persist([]);
   469→    setHistoryOpen(false);
   470→  };
   471→
   472→  const newChat = () => {
   473→    startNewChat();
   474→    toast(tt(T.agentClear, lang));
   475→  };
   476→
   477→  const suggestions = SUGGESTIONS.slice(0, 6);
   478→
   479→  /** the composer's bottom-left chips: tools · extended thinking · model */
   480→  const toolbar = (
   481→    <>
   482→      {/* tools chip */}
   483→      <Popover>
   484→        <PopoverTrigger asChild>
   485→          <button
   486→            type="button"
   487→            aria-label={tt(T.agentAttachTitle, lang)}
   488→            title={tt(T.agentAttachTitle, lang)}
   489→            className="inline-flex h-7 items-center gap-1.5 rounded-full border px-2.5 text-[11px] transition-colors hover:opacity-80"
   490→            style={{ color: "var(--chat-muted)" }}
   491→          >
   492→            <Wrench className="h-3 w-3" aria-hidden />
   493→            <span className="hidden sm:inline">{tt(T.agentToolsInfo, lang)}</span>
   494→          </button>
   495→        </PopoverTrigger>
   496→        <PopoverContent align="start" className="w-64 p-3 space-y-1.5">
   497→          <p className="text-[11px] font-semibold">{tt(T.agentToolsInfo, lang)}</p>
