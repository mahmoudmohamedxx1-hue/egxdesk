    31→    31→const RATE_LIMIT = 60; // requests per hour (long uncapped answers + web search)
    32→    32→const rateMap = new Map<string, number[]>();
    33→    33→
    34→    34→function rateLimited(ip: string): boolean {
    35→    35→  const now = Date.now();
    36→    36→  const window = 60 * 60_000;
    37→    37→  const arr = (rateMap.get(ip) ?? []).filter((t) => now - t < window);
    38→    38→  if (arr.length >= RATE_LIMIT) {
    39→    39→    rateMap.set(ip, arr);
    40→    40→    return true;
    41→    41→  }
    42→    42→  arr.push(now);
    43→    43→  rateMap.set(ip, arr);
    44→    44→  if (rateMap.size > 500) {
    45→    45→    // drop stale entries so the map can never grow unbounded
    46→    46→    for (const [k, v] of rateMap) if (v.every((t) => now - t >= window)) rateMap.delete(k);
    47→    47→  }
    48→    48→  return false;
    49→    49→}

    77→    77→function clampLimit(raw: unknown, max: number, dflt: number): number {
    78→    78→  const n = typeof raw === "number" ? Math.floor(raw) : Number(raw);

    83→    83→/** First balanced JSON object in the model output (handles ```json fences),
    84→    84→ *  with two repair passes for common LLM quirks: control characters (raw
    85→    85→ *  newlines) inside string literals, and replies truncated mid-string (no
    86→    86→ *  closing brace) — the tail is then recovered with a regex. */
    87→    87→function tryParse(s: string): Record<string, unknown> | null {
    88→    88→  try {
    89→    89→    const p = JSON.parse(s);
    90→    90→    return p && typeof p === "object" ? (p as Record<string, unknown>) : null;
    91→    91→  } catch {
    92→    92→    return null;
    93→    93→  }
    94→    94→}
    95→    95→
    96→    96→/** Escape raw \n \r \t that appear INSIDE string literals (invalid JSON that
    97→    97→ *  models sometimes emit); whitespace outside strings is left alone. */
    98→    98→function repairControlChars(s: string): string {
    99→    99→  let out = "";
   100→   100→  let inStr = false;
   101→   101→  let esc = false;
   102→   102→  for (const ch of s) {
   103→   103→    if (esc) {
   104→   104→      out += ch;
   105→   105→      esc = false;
   106→   106→      continue;
   107→   107→    }
   108→   108→    if (ch === "\\") {
   109→   109→      out += ch;
   110→   110→      esc = true;
   111→   111→      continue;
   112→   112→    }
   113→   113→    if (ch === '"') {
   114→   114→      inStr = !inStr;
   115→   115→      out += ch;
   116→   116→      continue;
   117→   117→    }
   118→   118→    if (inStr && (ch === "\n" || ch === "\r" || ch === "\t")) {
   119→   119→      out += "\\n";
   120→   120→      continue;
   121→   121→    }
   122→   122→    out += ch;
   123→   123→  }
   124→   124→  return out;
   125→   125→}
   126→   126→
   127→   127→/** Extract every balanced top-level {...} span in order (used by extractJson
   128→   128→ *  so reasoning prose around the reply object is ignored). */
   129→   129→function* topLevelJsonObjects(s: string): Generator<string> {
   130→   130→  let i = 0;
   131→   131→  while (i < s.length) {
   132→   132→    const start = s.indexOf("{", i);
   133→   133→    if (start === -1) return;
   134→   134→    let depth = 0;
   135→   135→    let inStr = false;
   136→   136→    let esc = false;
   137→   137→    let closed = false;
   138→   138→    for (let j = start; j < s.length; j++) {
   139→   139→      const c = s[j];
   140→   140→      if (esc) {
   141→   141→        esc = false;
   142→   142→        continue;
   143→   143→      }
   144→   144→      if (c === "\\") {
   145→   145→        if (inStr) esc = true;
   146→   146→        continue;
   147→   147→      }
   148→   148→      if (c === '"') {
   149→   149→        inStr = !inStr;
   150→   150→        continue;
   151→   151→      }
   152→   152→      if (inStr) continue;
   153→   153→      if (c === "{") depth++;
   154→   154→      else if (c === "}") {
   155→   155→        depth--;
   156→   156→        if (depth === 0) {
   157→   157→          yield s.slice(start, j + 1);
   158→   158→          i = j + 1;
   159→   159→          closed = true;
   160→   160→          break;
   161→   161→        }
   162→   162→      }
   163→   163→    }
   164→   164→    if (!closed) return; // truncated from here on — stop
   165→   165→  }
   166→   166→}

   168→   168→function extractJson(raw: string): Record<string, unknown> | null {
   169→   169→  if (!raw) return null;
   170→   170→  let s = raw.trim();
   171→   171→  // thinking-enabled models sometimes inline chain-of-thought — strip it first
   172→   172→  s = s.replace(/<think>[\s\S]*?<\/think>/gi, "");
   173→   173→  const fence = s.match(/```(?:json)?\s*([\s\S]*?)```/i);
   174→   174→  if (fence) s = fence[1].trim();
   175→   175→
   176→   176→  // 1) scan every balanced object; the FIRST one that follows the reply
   177→   177→  //    protocol (has "tool" or "final") wins — prose or examples before it
   178→   178→  //    are ignored. A parsed non-protocol object is kept as a last resort so
   179→   179→  //    the caller can issue a format correction (original behavior).
   180→   180→  let firstParsed: Record<string, unknown> | null = null;
   181→   181→  for (const span of topLevelJsonObjects(s)) {
   182→   182→    const p = tryParse(span) ?? tryParse(repairControlChars(span));
   183→   183→    if (!p) continue;
   184→   184→    if (firstParsed === null) firstParsed = p;
   185→   185→    if ("tool" in p || "final" in p) return p;
   186→   186→  }
   187→   187→  if (firstParsed) return firstParsed;
   188→   188→
   189→   189→  // 2) truncated reply (no closing brace): recover {"final": "… from the tail
   190→   190→  const mFinal = s.match(/"final"\s*:\s*"([\s\S]*)/);
   191→   191→  if (mFinal) {
   192→   192→    const tail = mFinal[1].replace(/"\s*\}?\s*$/, "").trim();
   193→   193→    if (tail.length > 0) return { final: tail };
   194→   194→  }
   195→   195→
   196→   196→  // 3) a tool call that survived truncation: {"tool": "name"…
   197→   197→  const mTool = s.match(/"tool"\s*:\s*"([A-Za-z_][A-Za-z0-9_]*)"/);
   198→   198→  if (mTool) {
   199→   199→    const mArgs = s.match(/"args"\s*:\s*(\{[\s\S]*?)[}]?\s*$/);
   200→   200→    let args: Record<string, unknown> = {};
   201→   201→    if (mArgs) {
   202→   202→      const repaired = repairControlChars(`${mArgs[1]}}`);
   203→   203→      const parsed = tryParse(repaired);
   204→   204→      if (parsed) args = parsed;
   205→   205→    }
   206→   206→    return { tool: mTool[1], args };
   207→   207→  }
   208→   208→  return null;
   209→   209→}
