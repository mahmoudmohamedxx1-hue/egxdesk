    16→/** POST /api/agent — the in-app EGX analyst agent (inspired by the tool-loop
    17→ *  pattern of open-source agent frameworks like shubhamsaboo/awesome-llm-apps
    18→ *  and the agent-skills repos): an LLM with STRICT-JSON tool calling over our
    19→ *  own live data layer — quotes, screening, technicals, statements, dividends,
    20→ *  news, calendar, rates, insider filings, flows. The loop runs server-side
    21→ *  (z-ai-web-dev-sdk never reaches the client), max ~10 tool calls per
    22→ *  question, then the model writes the final markdown answer. Gateway 429s
    23→ *  are retried with backoff; every request is metered in UsageEvent.
    24→ *
    25→ *  Honesty by design: tools return only real (delayed ~15-min) data; the
    26→ *  system prompt forbids invented numbers; the response carries a fixed
    27→ *  disclaimer that the client always renders. */
    28→
    29→export const runtime = "nodejs";

    31→// ── rate limiting + usage metering (per IP or device, persisted in SQLite) ──
    32→// 60 agent questions/hour protects a personal tool from floods; since Task 19
    33→// the counter lives in the UsageEvent table, so it survives restarts (the old
    34→// in-memory map reset on every deploy). The upstream LLM gateway ALSO
    35→// throttles bursts (~9-10 calls / ~90-120s window → 429) — those are retried
    36→// with backoff below, and every request is metered for /api/usage.
    37→
    38→const RATE_LIMIT = 60; // agent requests per hour, per IP or deviceId
    39→
    40→const rateMap = new Map<string, number[]>();
    41→
    42→function rateLimitedMemory(ip: string): boolean {
    43→  const now = Date.now();
    44→  const window = 60 * 60_000;
    45→  const arr = (rateMap.get(ip) ?? []).filter((t) => now - t < window);
    46→  if (arr.length >= RATE_LIMIT) {
    47→    rateMap.set(ip, arr);
    48→    return true;
    49→  }
    50→  arr.push(now);
    51→  rateMap.set(ip, arr);
    52→  if (rateMap.size > 500) {
    53→    // drop stale entries so the map can never grow unbounded
    54→    for (const [k, v] of rateMap) if (v.every((t) => now - t >= window)) rateMap.delete(k);
    55→  }
    56→  return false;
    57→}

    59→async function overLimit(ip: string, deviceId: string | null): Promise<boolean> {
    60→  try {
    61→    const since = new Date(Date.now() - 60 * 60_000);
    62→    const count = await db.usageEvent.count({
    63→      where: { createdAt: { gte: since }, OR: [{ ip }, ...(deviceId ? [{ deviceId }] : [])] },
    64→    });
    65→    return count >= RATE_LIMIT;
    66→  } catch {
    67→    return rateLimitedMemory(ip); // SQLite unreachable → in-memory fallback
    68→  }
    69→}

    71→// ── platform 429 handling: a mid-loop throttle must not kill a question ──
    72→
    73→const RETRY_BACKOFF_MS = [12_000, 25_000];
    74→const RETRY_BUDGET_MS = 70_000;
    75→
    76→const sleep = (ms: number) => new Promise((r) => setTimeout(r, ms));

    78→function isThrottleError(err: unknown): boolean {
    79→  const msg = err instanceof Error ? err.message : String(err);
    80→  return msg.includes("429") || /too many requests/i.test(msg);
    81→}

    83→async function createChat(
    84→  zai: Zai,
    85→  opts: { messages: { role: "user" | "assistant"; content: string }[]; thinking: "enabled" | "disabled" },
    86→  retry: { budgetLeft: number }
    87→): Promise<string> {
    88→  for (let attempt = 0; ; attempt++) {
    89→    try {
    90→      const completion = await zai.chat.completions.create({
    91→        messages: opts.messages,
    92→        thinking: { type: opts.thinking },
    93→      });
    94→      return completion.choices[0]?.message?.content ?? "";
    95→    } catch (err) {
    96→      const wait = RETRY_BACKOFF_MS[attempt];
    97→      if (isThrottleError(err) && wait !== undefined && retry.budgetLeft >= wait) {
    98→        retry.budgetLeft -= wait;
    99→        await sleep(wait);
   100→        continue;
   101→      }
   102→      throw err;
   103→    }
   104→  }
   105→}

   133→function clampLimit(raw: unknown, max: number, dflt: number): number {
   134→  const n = typeof raw === "number" ? Math.floor(raw) : Number(raw);
   135→  if (!Number.isFinite(n) || n <= 0) return dflt;
   136→  return Math.min(n, max);

   139→/** First balanced JSON object in the model output (handles ```json fences),
   140→ *  with two repair passes for common LLM quirks: control characters (raw
   141→ *  newlines) inside string literals, and replies truncated mid-string (no
   142→ *  closing brace) — the tail is then recovered with a regex. */
   143→function tryParse(s: string): Record<string, unknown> | null {
   144→  try {
   145→    const p = JSON.parse(s);
   146→    return p && typeof p === "object" ? (p as Record<string, unknown>) : null;
   147→  } catch {
   148→    return null;
   149→  }
   150→}
   151→
   152→/** Escape raw \n \r \t that appear INSIDE string literals (invalid JSON that
   153→ *  models sometimes emit); whitespace outside strings is left alone. */
   154→function repairControlChars(s: string): string {
   155→  let out = "";
   156→  let inStr = false;
   157→  let esc = false;
   158→  for (const ch of s) {
   159→    if (esc) {
   160→      out += ch;
   161→      esc = false;
   162→      continue;
   163→    }
   164→    if (ch === "\\") {
   165→      out += ch;
   166→      esc = true;
   167→      continue;
   168→    }
   169→    if (ch === '"') {
   170→      inStr = !inStr;
   171→      out += ch;
   172→      continue;
   173→    }
   174→    if (inStr && (ch === "\n" || ch === "\r" || ch === "\t")) {
   175→      out += "\\n";
   176→      continue;
   177→    }
   178→    out += ch;
   179→  }
   180→  return out;
   181→}
   182→
   183→/** Extract every balanced top-level {...} span in order (used by extractJson
   184→ *  so reasoning prose around the reply object is ignored). */
   185→function* topLevelJsonObjects(s: string): Generator<string> {
   186→  let i = 0;
   187→  while (i < s.length) {
   188→    const start = s.indexOf("{", i);
   189→    if (start === -1) return;
   190→    let depth = 0;
   191→    let inStr = false;
   192→    let esc = false;
   193→    let closed = false;
   194→    for (let j = start; j < s.length; j++) {
   195→      const c = s[j];
   196→      if (esc) {
   197→        esc = false;
   198→        continue;
   199→      }
   200→      if (c === "\\") {
   201→        if (inStr) esc = true;
   202→        continue;
   203→      }
   204→      if (c === '"') {
   205→        inStr = !inStr;
   206→        continue;
   207→      }
   208→      if (inStr) continue;
   209→      if (c === "{") depth++;
   210→      else if (c === "}") {
   211→        depth--;
   212→        if (depth === 0) {
   213→          yield s.slice(start, j + 1);
   214→          i = j + 1;
   215→          closed = true;
   216→          break;
   217→        }
   218→      }
   219→    }
   220→    if (!closed) return; // truncated from here on — stop
   221→  }
   222→}

   224→function extractJson(raw: string): Record<string, unknown> | null {
   225→  if (!raw) return null;
   226→  let s = raw.trim();
   227→  // thinking-enabled models sometimes inline chain-of-thought — strip it first
   228→  s = s.replace(/<think>[\s\S]*?<\/think>/gi, "");
   229→  const fence = s.match(/```(?:json)?\s*([\s\S]*?)```/i);
   230→  if (fence) s = fence[1].trim();
   231→
   232→  // 1) scan every balanced object; the FIRST one that follows the reply
   233→  //    protocol (has "tool" or "final") wins — prose or examples before it
   234→  //    are ignored. A parsed non-protocol object is kept as a last resort so
   235→  //    the caller can issue a format correction (original behavior).
   236→  let firstParsed: Record<string, unknown> | null = null;
   237→  for (const span of topLevelJsonObjects(s)) {
   238→    const p = tryParse(span) ?? tryParse(repairControlChars(span));
   239→    if (!p) continue;
   240→    if (firstParsed === null) firstParsed = p;
   241→    if ("tool" in p || "final" in p) return p;
   242→  }
   243→  if (firstParsed) return firstParsed;
   244→
   245→  // 2) truncated reply (no closing brace): recover {"final": "… from the tail
   246→  const mFinal = s.match(/"final"\s*:\s*"([\s\S]*)/);
   247→  if (mFinal) {
   248→    const tail = mFinal[1].replace(/"\s*\}?\s*$/, "").trim();
   249→    if (tail.length > 0) return { final: tail };
   250→  }
   251→
   252→  // 3) a tool call that survived truncation: {"tool": "name"…
   253→  const mTool = s.match(/"tool"\s*:\s*"([A-Za-z_][A-Za-z0-9_]*)"/);
   254→  if (mTool) {
   255→    const mArgs = s.match(/"args"\s*:\s*(\{[\s\S]*?)[}]?\s*$/);
   256→    let args: Record<string, unknown> = {};
   257→    if (mArgs) {
   258→      const repaired = repairControlChars(`${mArgs[1]}}`);
   259→      const parsed = tryParse(repaired);
   260→      if (parsed) args = parsed;
   261→    }
   262→    return { tool: mTool[1], args };
   263→  }
   264→  return null;
   265→}

   665→export async function POST(req: Request) {
   666→  const ip =
   667→    req.headers.get("x-forwarded-for")?.split(",")[0]?.trim() ||
   668→    req.headers.get("x-real-ip") ||
   669→    "local";
   670→
   671→  let body: { messages?: unknown; lang?: unknown; debug?: unknown; deviceId?: unknown };
   672→  try {
   673→    body = (await req.json()) as { messages?: unknown; lang?: unknown; deviceId?: unknown };
   674→  } catch {
   675→    return NextResponse.json({ error: "invalid json" }, { status: 400 });
   676→  }
   677→  const deviceId =
   678→    typeof body.deviceId === "string" && body.deviceId.length >= 8 ? body.deviceId.slice(0, 64) : null;
   679→
   680→  // persisted hourly limit (per IP or device) — UsageEvent survives restarts
   681→  if (await overLimit(ip, deviceId)) {
   682→    return NextResponse.json({ error: "rate limited — try again later" }, { status: 429 });
   683→  }
   684→
   685→  // validate the conversation the client sends (last 24 turns, last is user)
   686→  const lang = body.lang === "en" ? "en" : "ar";
   687→  const rawMsgs = Array.isArray(body.messages) ? body.messages : [];
   688→  const history: ChatMsg[] = rawMsgs
   689→    .filter(
   690→      (m): m is ChatMsg =>
   691→        m !== null &&
   692→        typeof m === "object" &&
   693→        ((m as ChatMsg).role === "user" || (m as ChatMsg).role === "assistant") &&
   694→        typeof (m as ChatMsg).content === "string" &&
   695→        (m as ChatMsg).content.length > 0 &&
   696→        (m as ChatMsg).content.length <= 8000
   697→    )
   698→    .slice(-24);
   699→  if (history.length === 0 || history[history.length - 1].role !== "user") {
   700→    return NextResponse.json({ error: "messages required (last must be user)" }, { status: 400 });
   701→  }
   702→
   703→  const zai = await getZai();
   704→  const msgs: { role: "user" | "assistant"; content: string }[] = [
   705→    { role: "assistant", content: systemPrompt(lang) },
   706→    ...history.map((m) => ({ role: m.role, content: m.content })),
   707→  ];
   708→  const steps: AgentStep[] = [];
   709→  let corrections = 0;
   710→  const debugRaw: string[] = [];
   711→  const debug = body.debug === true;
   712→
   713→  // usage metering — one UsageEvent row per request, written by finish()
   714→  const t0 = Date.now();
   715→  const usage = { llmCalls: 0, webSearches: 0 };
   716→  const retry = { budgetLeft: RETRY_BUDGET_MS };
   717→  const finish = (payload: Record<string, unknown>, status = 200) => {
   718→    void db.usageEvent
   719→      .create({
   720→        data: {
   721→          ip,
   722→          deviceId,
   723→          route: "agent",
   724→          llmCalls: usage.llmCalls,
   725→          toolCalls: steps.length,
   726→          webSearches: usage.webSearches,
   727→          ok: status === 200,
   728→          ms: Date.now() - t0,
   729→        },
   730→      })
   731→      .catch(() => {}); // metering must never break a reply
   732→    return NextResponse.json(payload, { status, headers: { "Cache-Control": "no-store" } });
   733→  };
   734→
   735→  for (let round = 0; round < MAX_TOOL_CALLS + 3; round++) {
   736→    if (steps.length >= MAX_TOOL_CALLS) break;
   737→    let raw = "";
   738→    try {
   739→      // reasoning depth: round 0 (fast JSON tool-picking or a quick
   740→      // conversational reply) runs with thinking off; every later round —
   741→      // i.e. once real tool data is on the table, the synthesis moment — runs
   742→      // with chain-of-thought ON so the final answer is genuine reasoning,
   743→      // not a shallow template.
   744→      const deep = round > 0;
   745→      raw = await createChat(zai, { messages: msgs, thinking: deep ? "enabled" : "disabled" }, retry);
   746→      usage.llmCalls++;
   747→    } catch (err) {
   748→      // gateway 429s retry with backoff inside createChat; if throttling
   749→      // persists, answer honestly instead of a bare "model unavailable"
   750→      const throttled = isThrottleError(err);
   751→      return finish(
   752→        {
   753→          error: throttled
   754→            ? lang === "ar"
   755→              ? "خدمة الذكاء الاصطناعي مشغولة مؤقتًا (ضغط على المزود) — جرّب بعد دقيقة"
   756→              : "The AI service is briefly busy (provider throttling) — please retry in a minute"
   757→            : "model unavailable",
   758→          detail: err instanceof Error ? err.message : "unknown",
   759→        },
   760→        throttled ? 503 : 502
   761→      );
   762→    }
   763→    if (debug) debugRaw.push(raw.slice(0, 800));
   764→
   765→    const parsed = extractJson(raw);
   766→    if (!parsed || (!("tool" in parsed) && !("final" in parsed))) {
   767→      corrections++;
   768→      if (corrections > 2) break;
   769→      msgs.push({
   770→        role: "user",
   771→        content:
   772→          'Format error. Reply with exactly ONE JSON object, no fences: {"tool": "<name>", "args": {...}} to call a tool, or {"final": "<markdown answer>"} to answer.',
   773→      });
   774→      continue;
   775→    }
   776→
   777→    if (typeof parsed.final === "string" && parsed.final.trim().length > 0) {
   778→      return finish({ answer: parsed.final.trim(), steps, model: "GLM", disclaimer: true, ...(debug ? { debugRaw } : {}) });
   779→    }
   780→
   781→    const toolName = typeof parsed.tool === "string" ? parsed.tool : "";
   782→    const tool = TOOLS.find((t) => t.name === toolName);
   783→    if (!tool) {
   784→      msgs.push({
   785→        role: "user",
   786→        content: `Unknown tool "${toolName}". Available tools: ${TOOLS.map((t) => t.name).join(", ")}.`,
   787→      });
   788→      continue;
   789→    }
   790→
   791→    const args = (parsed.args && typeof parsed.args === "object" ? parsed.args : {}) as Record<string, unknown>;
   792→    let result: unknown;
   793→    try {
   794→      result = await tool.run(args);
   795→    } catch (err) {
   796→      result = { error: err instanceof Error ? err.message : "tool failed" };
   797→    }
   798→    const ok = !(result && typeof result === "object" && "error" in (result as Record<string, unknown>));
   799→    steps.push({ tool: toolName, args, ok });
   800→    if (toolName === "web_search" && ok) usage.webSearches++;
   801→
   802→    msgs.push({ role: "user", content: JSON.stringify(result).slice(0, 9000) });
   803→  }
   804→
   805→  // loop exhausted without a final answer — NEVER waste the collected data:
   806→  // force one synthesis round from the tool results already in context
   807→  if (steps.length > 0) {
   808→    msgs.push({
   809→      role: "user",
   810→      content:
   811→        'Tool budget exhausted. Reply NOW with your final answer using ONLY the tool data collected above — do not request more tools; if a requested stock was not found, say so plainly. Format: {"final": "<markdown answer>"}',
   812→    });
   813→    try {
   814→      const raw = await createChat(zai, { messages: msgs, thinking: "enabled" }, retry);
   815→      usage.llmCalls++;
   816→      if (debug) debugRaw.push(raw.slice(0, 800));
   817→      const parsed = extractJson(raw);
   818→      if (parsed && typeof parsed.final === "string" && parsed.final.trim().length > 0) {
   819→        return finish({ answer: parsed.final.trim(), steps, model: "GLM", disclaimer: true, ...(debug ? { debugRaw } : {}) });
   820→      }
   821→    } catch {
   822→      // fall through to the honest fallback below
   823→    }
   824→  }
   825→
   826→  // no tools ever ran / synthesis also failed — honest fallback, never invented
   827→  const fallback =
   828→    lang === "ar"
   829→      ? "وصلتُ لحد الأدوات المتاحة دون إجابة كاملة. جرّب إعادة السؤال بصيغة أبسط (مثال: «ما حالة السوق الآن؟» أو «quote لسهم COMI»)."
   830→      : "I ran out of tool budget without a complete answer. Try rephrasing (e.g. \"market overview\" or \"quote for COMI\").";
   831→  return finish({ answer: fallback, steps, model: "GLM", disclaimer: true, ...(debug ? { debugRaw } : {}) });
   832→}
