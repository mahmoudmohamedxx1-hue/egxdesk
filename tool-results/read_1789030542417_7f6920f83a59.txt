     1→"use client";
     2→
     3→import { useState } from "react";
     4→import { useApp } from "../market/app-context";
     5→import { useLiveData } from "../market/use-live-data";
     6→import type { CompanyRow, NewsRow, SessionMeta } from "../market/types";
     7→import { T, tt, dn } from "@/lib/i18n";

     9→import { WatchStar } from "../market/watch-star";
    10→import { ChangeCell } from "../market/change-cell";
    11→import { PerfChart, RangeBar } from "../market/perf-chart";
    12→import { PriceChart } from "../market/price-chart";
    13→import { StatementsPanel } from "../market/statements-panel";
    14→import { TechnicalPanel } from "../market/technical-panel";
    15→import { DividendsPanel } from "../market/dividends-panel";
    16→import { SetAlertButton } from "../market/alerts-panel";
    17→import { ValuationPanel } from "../market/valuation-panel";
    18→import { Skeleton } from "@/components/ui/skeleton";
    19→import { Button } from "@/components/ui/button";
    20→import { Volume2, Calculator, TrendingUp, TrendingDown, ExternalLink, RefreshCw, Zap, CalendarClock, FileSpreadsheet } from "lucide-react";
    21→
    22→type CompanyData = {
    23→  session: SessionMeta;
    24→  company: CompanyRow & {
    25→    industry: string | null;
    26→    netMarginTTM: number | null;
    27→    revenueTTM: number | null;
    28→    floatShares: number | null;
    29→    avgTurnover30: number | null;
    30→    high1M: number | null;
    31→    low1M: number | null;
    32→    beta: number | null;
    33→    updateMode: string | null;
    34→    employees?: number | null;
    35→  };
    36→  sectorAgg: {
    37→    count: number;
    38→    nameAr: string;
    39→    nameEn: string;
    40→    pe: number | null;
    41→    pb: number | null;
    42→    roe: number | null;
    43→    eps: number | null;
    44→    divYield: number | null;
    45→    beta: number | null;
    46→    avgChangePct: number | null;
    47→  };
    48→  peers: CompanyRow[];
    49→  news: NewsRow[];
    50→  disclosures: NewsRow[];
    51→  signals: {
    52→    streak: { direction: "up" | "down"; count: number; since: string } | null;
    53→    nextEarnings: string | null;
    54→    unusualVolume: boolean;
    55→    volumeRatio: number | null;
    56→    near52High: boolean;
    57→    near52Low: boolean;
    58→    computedFrom: string;
    59→  } | null;
    60→};
    61→
    62→export function CompanyView({ ticker, panel }: { ticker: string; panel: string }) {
    63→  const { lang, navigate } = useApp();
    64→  const { data, error, refresh } = useLiveData<CompanyData>(`/api/company/${encodeURIComponent(ticker)}`, 60_000);
    65→  const [activePanel, setActivePanel] = useState(panel);
    66→  const [speaking, setSpeaking] = useState(false);
    67→  const [prevPanel, setPrevPanel] = useState(panel);
    68→  if (prevPanel !== panel) {
    69→    // URL panel changed (e.g. navigating from a peer link) — adjust state during render
    70→    setPrevPanel(panel);
    71→    setActivePanel(panel);
    72→  }
    73→
    74→  if (error && !data) {
    75→    return (
    76→      <div className="rounded-lg border bg-card p-8 text-center space-y-3">
    77→        <p className="font-medium">{tt(T.errorLoad, lang)}</p>
    78→        <Button size="sm" onClick={refresh}>
    79→          <RefreshCw className="h-3.5 w-3.5 me-1.5" />
    80→          {tt(T.retry, lang)}
    81→        </Button>
    82→      </div>
    83→    );
    84→  }
    85→
    86→  if (!data || !data.company) {
    87→    if (data && (data as { error?: string }).error === "no such company") {
    88→      return (
    89→        <div className="py-16 text-center space-y-2">
    90→          <p className="text-lg font-semibold">{lang === "ar" ? "لا توجد شركة بهذا الرمز" : "No such company"}</p>
    91→          <button onClick={() => navigate("market")} className="text-sm text-primary hover:underline">
    92→            {tt(T.browseMarket, lang)}
    93→          </button>
    94→        </div>
    95→      );
    96→    }
    97→    return (
    98→      <div className="space-y-4">
    99→        <Skeleton className="h-16 w-full" />
   100→        <Skeleton className="h-48 w-full" />
   101→        <Skeleton className="h-64 w-full" />
   102→      </div>
   103→    );
   104→  }
   105→
   106→  const c = data.company;
   107→
   108→  function speakQuote() {
   109→    if (typeof window === "undefined" || !window.speechSynthesis) return;
   110→    if (speaking) { window.speechSynthesis.cancel(); setSpeaking(false); return; }
   111→    const dir = c.changePct >= 0
   112→      ? (lang === "ar" ? "مرتفع" : "up")
   113→      : (lang === "ar" ? "منخفض" : "down");
   114→    const text = lang === "ar"
   115→      ? `${c.ticker}. ${dn(c, lang)}. آخر سعر ${fmtNum(c.close)} جنيه، ${dir} بنسبة ${fmtNum(Math.abs(c.changePct))} بالمئة.`
   116→      : `${c.ticker}. ${c.name}. Last ${fmtNum(c.close)} Egyptian pounds, ${dir} ${fmtNum(Math.abs(c.changePct))} percent.`;
   117→    const u = new SpeechSynthesisUtterance(text);
   118→    u.lang = lang === "ar" ? "ar-EG" : "en-US";
   119→    u.onend = () => setSpeaking(false);
   120→    u.onerror = () => setSpeaking(false);
   121→    window.speechSynthesis.speak(u);
   122→    setSpeaking(true);
   123→  }
   124→
   125→  const panels = [
   126→    { key: "overview", t: T.panelOverview },
   127→    { key: "technical", t: T.panelTechnical },
   128→    { key: "statements", t: T.panelStatements },
   129→    { key: "dividends", t: T.panelDividends },
   130→    { key: "valuation", t: { ar: "التقييم", en: "Valuation" } },
   131→    { key: "fundamentals", t: T.panelFundamentals },
   132→    { key: "disclosures", t: T.panelDisclosures },
   133→    { key: "activity", t: T.panelActivity },
   134→    { key: "news", t: T.panelNews },
   135→  ];
   136→
   137→  const perf = [
   138→    { label: "1W", value: c.perfW },
   139→    { label: "1M", value: c.perf1M },
   140→    { label: "3M", value: c.perf3M },
   141→    { label: "6M", value: c.perf6M },
   142→    { label: "YTD", value: c.perfYTD },
   143→    { label: "1Y", value: c.perfY },
   144→    { label: "3Y", value: c.perf3Y },
   145→    { label: "5Y", value: c.perf5Y },
   146→  ];
   147→
   148→  return (
   149→    <div className="space-y-5">
   150→      {/* ticker row */}
   151→      <div className="flex items-center flex-wrap gap-2">
   152→        <span className="num text-lg font-bold">{c.ticker}</span>
   153→        <WatchStar ticker={c.ticker} />
   154→        <button
   155→          onClick={speakQuote}
   156→          className="inline-flex items-center gap-1.5 rounded-md border px-2.5 py-1.5 text-xs hover:bg-accent/50 transition-colors"
   157→        >
   158→          <Volume2 className={`h-3.5 w-3.5 ${speaking ? "animate-pulse" : ""}`} />
   159→          {tt(T.listenBrief, lang)}
   160→        </button>
   161→        <button
   162→          onClick={() => navigate("tools")}
   163→          className="inline-flex items-center gap-1.5 rounded-md border px-2.5 py-1.5 text-xs hover:bg-accent/50 transition-colors"
   164→        >
   165→          <Calculator className="h-3.5 w-3.5" />
   166→          {lang === "ar" ? "احسب عائد الكوبون" : "Compute coupon return"}
   167→        </button>
   168→        <SetAlertButton ticker={c.ticker} close={c.close} />
   169→      </div>
   170→
   171→      {/* identity + quote */}
   172→      <div className="rounded-lg border bg-card p-4 sm:p-5">
   173→        <div className="flex flex-wrap items-start justify-between gap-4">
   174→          <div className="min-w-0 space-y-1">
   175→            <p className="text-xs text-muted-foreground">
   176→              {lang === "ar" ? c.sectorAr : c.sectorEn}
   177→              {c.industry ? ` · ${c.industry}` : ""}
   178→            </p>
   179→            <h1 className="text-xl sm:text-2xl font-bold tracking-tight">{dn(c, lang)}</h1>
   180→            {lang === "ar" && c.nameAr && c.nameAr !== c.name && (
   181→              <p className="text-xs text-muted-foreground truncate">{c.name}</p>
   182→            )}
   183→            {lang === "en" && c.nameAr && (
   184→              <p className="text-xs text-muted-foreground truncate" dir="rtl">{c.nameAr}</p>
   185→            )}
   186→            <p className="num text-xs text-muted-foreground">EGX · EGP · {tt(T.delayed, lang)}</p>
   187→          </div>
   188→          <div className="text-end space-y-0.5">
   189→            <p className="text-xs text-muted-foreground">{tt(T.lastClose, lang)}</p>
   190→            <p className="num text-3xl font-bold tracking-tight">{fmtNum(c.close)}</p>
   191→            <p className={`num text-sm font-semibold ${directionClass(c.changePct)} flex items-center gap-1.5 justify-end`}>
   192→              {c.changePct >= 0 ? <TrendingUp className="h-3.5 w-3.5" /> : <TrendingDown className="h-3.5 w-3.5" />}
   193→              {c.changePct >= 0 ? "+" : ""}{fmtNum(c.changeAbs)} ({fmtPct(c.changePct)})
   194→            </p>
   195→            <p className="num text-[10px] text-muted-foreground">{data.session.lastSession}</p>
   196→          </div>
   197→        </div>
   198→
   199→        {/* stat strip */}
   200→        <div className="mt-4 grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-6 gap-x-4 gap-y-3 border-t pt-4">
   201→          <Stat label={tt(T.marketCap, lang)} value={`EGP ${fmtValue(c.marketCap)}`} />
   202→          <Stat label={tt(T.week, lang)} value={fmtPct(c.perfW)} cls={directionClass(c.perfW)} />
   203→          <Stat label={tt(T.month, lang)} value={fmtPct(c.perf1M)} cls={directionClass(c.perf1M)} />
   204→          <Stat label={tt(T.ytd, lang)} value={fmtPct(c.perfYTD)} cls={directionClass(c.perfYTD)} />
   205→          <Stat label={tt(T.volume, lang)} value={fmtInt(c.volume)} sub={`${tt(T.inSession, lang)} · ${tt(T.avg10, lang)} ${fmtInt(c.avgVolume)}`} />
   206→          <Stat label="P/E" value={c.pe ? fmtNum(c.pe, 1) : "—"} sub={c.eps ? `EPS ${fmtNum(c.eps)}` : undefined} />
   207→        </div>
   208→      </div>
   209→
   210→      {/* panel tabs */}
   211→      <div className="flex items-center gap-1 overflow-x-auto thin-scroll border-b" role="tablist">
   212→        {panels.map((p) => (
   213→          <button
   214→            key={p.key}
   215→            role="tab"
   216→            aria-selected={activePanel === p.key}
   217→            onClick={() => setActivePanel(p.key)}
   218→            className={`whitespace-nowrap px-3 py-2 text-sm -mb-px border-b-2 transition-colors ${
   219→              activePanel === p.key ? "border-primary font-semibold" : "border-transparent text-muted-foreground hover:text-foreground"
   220→            }`}
   221→          >
   222→            {tt(p.t, lang)}
   223→            {p.key === "news" && data.news.length > 0 && (
   224→              <span className="num ms-1.5 text-xs text-muted-foreground">{data.news.length}</span>
   225→            )}
   226→            {p.key === "disclosures" && data.disclosures.length > 0 && (
   227→              <span className="num ms-1.5 text-xs text-muted-foreground">{data.disclosures.length}</span>
   228→            )}
   229→          </button>
   230→        ))}
   231→      </div>
   232→
   233→      {/* OVERVIEW */}
   234→      {activePanel === "overview" && (
   235→        <div className="space-y-5">
   236→          <section className="rounded-lg border bg-card p-4">
   237→            <div className="flex items-center justify-between mb-3 flex-wrap gap-2">
   238→              <h2 className="font-bold">{tt(T.priceChart, lang)}</h2>
   239→              <span className="num text-xs text-muted-foreground">{c.ticker} · EGX</span>
   240→            </div>
   241→            <PriceChart key={c.ticker} symbol={c.ticker} defaultRange="6M" />
   242→          </section>
   243→
   244→          {data.signals && <SignalsCard signals={data.signals} lang={lang} />}
   245→
   246→          <section className="rounded-lg border bg-card p-4">
   247→            <div className="flex items-center justify-between mb-3 flex-wrap gap-2">
   248→              <h2 className="font-bold">{tt(T.performance, lang)}</h2>
   249→            </div>
   250→            <PerfChart perf={perf} lang={lang} />
   251→            <p className="mt-2 text-[11px] text-muted-foreground">{tt(T.perfNote, lang)}</p>
   252→          </section>
   253→
   254→          <section className="rounded-lg border bg-card p-4">
   255→            <h2 className="font-bold mb-3">{tt(T.range52, lang)}</h2>
   256→            <RangeBar
   257→              low={c.low52}
   258→              high={c.high52}
   259→              close={c.close}
   260→              labelLow={lang === "ar" ? "أدنى ٥٢ أسبوعاً" : "52-week low"}
   261→              labelHigh={lang === "ar" ? "أعلى ٥٢ أسبوعاً" : "52-week high"}
   262→            />
   263→            {c.high1M !== null && c.low1M !== null && (
   264→              <p className="num mt-3 text-[11px] text-muted-foreground">
   265→                {lang === "ar" ? "مدى الشهر:" : "1-month range:"} {fmtNum(c.low1M, 1)} – {fmtNum(c.high1M, 1)}
   266→              </p>
   267→            )}
   268→          </section>
   269→        </div>
   270→      )}
   271→
   272→      {/* TECHNICAL ANALYSIS */}
   273→      {activePanel === "technical" && <TechnicalPanel key={c.ticker} ticker={c.ticker} />}
   274→
   275→      {/* STATEMENTS */}
   276→      {activePanel === "statements" && (
   277→        <StatementsPanel key={c.ticker} ticker={c.ticker} />
   278→      )}
   279→
   280→      {/* DIVIDENDS (G3) */}
   281→      {activePanel === "dividends" && (
   282→        <DividendsPanel key={c.ticker} ticker={c.ticker} close={c.close} />
   283→      )}
   284→
   285→      {/* VALUATION (G10) */}
   286→      {activePanel === "valuation" && (
   287→        <ValuationPanel key={c.ticker} company={c} sectorAgg={data.sectorAgg} />
   288→      )}
   289→
   290→      {/* DISCLOSURES (press-derived) */}
   291→      {activePanel === "disclosures" && (
   292→        <section className="rounded-lg border bg-card divide-y">
   293→          <div className="px-4 py-3">
   294→            <h2 className="font-bold flex items-center gap-2">
   295→              <FileSpreadsheet className="h-4 w-4 text-primary" />
   296→              {tt(T.panelDisclosures, lang)}
   297→            </h2>
   298→            <p className="mt-1 text-xs text-muted-foreground max-w-3xl leading-relaxed">{tt(T.disclosuresNote, lang)}</p>
   299→          </div>
   300→          {data.disclosures.length === 0 && (
   301→            <p className="px-4 py-10 text-center text-sm text-muted-foreground">{tt(T.noDisclosures, lang)}</p>
   302→          )}
   303→          {data.disclosures.map((n) => (
   304→            <article key={n.id} className="px-4 py-3">
   305→              <div className="flex items-center gap-2 mb-1 flex-wrap">
   306→                <span className="rounded-sm bg-secondary px-1.5 py-0.5 text-[10px] font-medium">{n.source}</span>
   307→                <span className="num text-[11px] text-muted-foreground">{fmtDateAr(n.publishedAt)} · {fmtTimeAr(n.publishedAt)}</span>
   308→              </div>
   309→              <a href={n.link} target="_blank" rel="noopener noreferrer" className="text-sm leading-snug font-medium hover:underline inline-flex items-start gap-1.5">
   310→                {n.title}
   311→                <ExternalLink className="h-3 w-3 mt-1 shrink-0 text-muted-foreground" />
   312→              </a>
   313→              {n.snippet && <p className="mt-1 text-xs text-muted-foreground line-clamp-2">{n.snippet}</p>}
   314→            </article>
   315→          ))}
   316→        </section>
   317→      )}
   318→
   319→      {/* ACTIVITY */}
   320→      {activePanel === "activity" && (
   321→        <section className="rounded-lg border bg-card p-4 space-y-4">
   322→          <h2 className="font-bold">{tt(T.panelActivity, lang)}</h2>
   323→          <div className="grid grid-cols-2 sm:grid-cols-4 gap-4">
   324→            <Stat label={tt(T.volume, lang)} value={fmtInt(c.volume)} sub={tt(T.inSession, lang)} />
   325→            <Stat label={tt(T.avg10, lang)} value={fmtInt(c.avgVolume)} />
   326→            <Stat label={tt(T.valueTraded, lang)} value={`EGP ${fmtValue(c.valueTraded)}`} sub={tt(T.inSession, lang)} />
   327→            <Stat label={lang === "ar" ? "الحجم ÷ المعتاد" : "Vol ÷ usual"} value={c.volumeRatio !== null ? `${fmtNum(c.volumeRatio, 1)}×` : "—"} sub={tt(T.inSession, lang)} />
   328→          </div>
   329→          <div className="rounded-md bg-secondary/70 p-3 flex items-center justify-between">
   330→            <span className="text-xs text-muted-foreground">{lang === "ar" ? "الحجم ÷ المعتاد" : "Volume ÷ usual"}</span>
   331→            <span className="num text-lg font-bold text-primary">
   332→              {c.avgVolume ? fmtNum(c.volume / c.avgVolume, 1) + "×" : "—"}
   333→            </span>
   334→          </div>
   335→          <div>
   336→            <div className="flex items-center justify-between mb-1 text-xs">
   337→              <span className="text-muted-foreground">{lang === "ar" ? "مقابل متوسط ١٠ جلسات" : "vs 10-session average"}</span>
   338→              <span className="num">{fmtInt(c.volume)} / {fmtInt(c.avgVolume)}</span>
   339→            </div>
   340→            <div className="h-2 rounded-full bg-secondary overflow-hidden" dir="ltr">
   341→              <div className="h-full rounded-full bg-primary" style={{ width: `${Math.min((c.volume / Math.max(c.avgVolume ?? 1, 1)) * 50, 100)}%` }} />
   342→            </div>
   343→          </div>
   344→          {c.avgTurnover30 !== null && (
   345→            <Stat label={lang === "ar" ? "متوسط قيمة التداول (٣٠ يوماً)" : "Avg turnover (30d)"} value={`EGP ${fmtValue(c.avgTurnover30)}`} />
   346→          )}
   347→          <p className="text-[11px] text-muted-foreground">{tt(T.unusualNote, lang)}</p>
   348→        </section>
   349→      )}
   350→
   351→      {/* FUNDAMENTALS */}
   352→      {activePanel === "fundamentals" && (
   353→        <div className="space-y-4">
   354→          <section className="rounded-lg border bg-card p-4">
   355→            <h2 className="font-bold mb-1">{tt(T.fundamentals, lang)}</h2>
   356→            <p className="text-xs text-muted-foreground mb-3">
   357→              {tt(T.vsSector, lang)} — {lang === "ar" ? data.sectorAgg.nameAr : data.sectorAgg.nameEn}{" "}
   358→              <span className="num">({data.sectorAgg.count} {tt(T.companies, lang)})</span>
   359→            </p>
   360→            <div className="grid grid-cols-2 sm:grid-cols-3 gap-x-4 gap-y-3">
   361→              <VsSector label="P/E" value={c.pe} agg={data.sectorAgg.pe} fmt={(v) => fmtNum(v, 1)} />
   362→              <VsSector label={lang === "ar" ? "م/د (القيمة الدفترية)" : "P/B (book)"} value={c.pb} agg={data.sectorAgg.pb} fmt={(v) => fmtNum(v, 2)} />
   363→              <VsSector label={lang === "ar" ? "العائد على حقوق الملكية" : "ROE"} value={c.roe} agg={data.sectorAgg.roe} fmt={(v) => `${fmtNum(v, 1)}%`} higherBetter />
   364→              <VsSector label="EPS" value={c.eps} agg={data.sectorAgg.eps} fmt={(v) => fmtNum(v)} higherBetter />
   365→              <VsSector label={lang === "ar" ? "عائد التوزيعات" : "Div yield"} value={c.divYield} agg={data.sectorAgg.divYield} fmt={(v) => `${fmtNum(v, 1)}%`} higherBetter />
   366→              <VsSector label={tt(T.beta, lang)} value={c.beta} agg={data.sectorAgg.beta} fmt={(v) => fmtNum(v, 2)} />
   367→              <MetricOnly label={tt(T.netMargin, lang)} value={c.netMarginTTM !== null ? `${fmtNum(c.netMarginTTM, 1)}%` : "—"} />
   368→              <MetricOnly label={lang === "ar" ? "صافي الربح (١٢ شهراً)" : "Net income (TTM)"} value={c.netIncomeTTM !== null ? `EGP ${fmtValue(c.netIncomeTTM)}` : "—"} />
   369→              <MetricOnly label={tt(T.revenueTtm, lang)} value={c.revenueTTM !== null ? `EGP ${fmtValue(c.revenueTTM)}` : "—"} />
   370→              <MetricOnly label={lang === "ar" ? "الدين / حقوق الملكية" : "Debt / equity"} value={c.debtToEquity !== null ? fmtNum(c.debtToEquity, 2) : "—"} />
   371→              <MetricOnly label={lang === "ar" ? "هامش الربح الإجمالي" : "Gross margin"} value={c.grossMarginTTM !== null ? `${fmtNum(c.grossMarginTTM, 1)}%` : "—"} />
   372→              <MetricOnly label={lang === "ar" ? "صافي الدين" : "Net debt"} value={c.netDebt !== null ? `EGP ${fmtValue(c.netDebt)}` : "—"} />
   373→              <MetricOnly label={tt(T.floatShares, lang)} value={c.floatShares !== null ? fmtInt(c.floatShares) : "—"} />
   374→              <MetricOnly label={lang === "ar" ? "الموظفون" : "Employees"} value={c.employees !== null && c.employees !== undefined ? fmtInt(c.employees) : "—"} />
   375→              <MetricOnly label={tt(T.marketCap, lang)} value={`EGP ${fmtValue(c.marketCap)}`} />
   376→            </div>
   377→          </section>
   378→
   379→          {/* peers */}
   380→          <section className="rounded-lg border bg-card overflow-hidden">
   381→            <div className="border-b px-4 py-3">
   382→              <h2 className="font-bold">{tt(T.peers, lang)}</h2>
   383→            </div>
   384→            <div className="overflow-x-auto thin-scroll">
   385→              <table className="w-full text-sm">
   386→                <thead className="border-b bg-card">

   530→function VsSector({ label, value, agg, fmt, higherBetter }: {
   531→  label: string; value: number | null | undefined; agg: number | null | undefined;
   532→  fmt: (v: number) => string; higherBetter?: boolean;
   533→}) {
   534→  const has = typeof value === "number" && typeof agg === "number";
   535→  let better: boolean | null = null;
   536→  if (has) {
   537→    better = higherBetter ? value > agg : value < agg;
   538→  }
   539→  return (
   540→    <div className="border-b border-dotted pb-2">
   541→      <p className="text-xs text-muted-foreground">{label}</p>
   542→      <div className="flex items-baseline gap-2 flex-wrap">
   543→        <span className="num text-sm font-semibold">{value !== null && value !== undefined ? fmt(value) : "—"}</span>
   544→        {has && (
   545→          <span className={`num text-[10px] px-1 rounded ${better ? "text-up bg-up-soft" : "text-down bg-down-soft"}`}>
   546→            {better ? "▲" : "▼"} {fmt(agg!)}
   547→          </span>
   548→        )}
   549→      </div>
   550→    </div>
   551→  );
   552→}
