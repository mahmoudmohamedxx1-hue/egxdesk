    96→    96→        const x1 = 12 + Math.sin(rad) * 3.4;
    97→    97→        const y1 = 12 - Math.cos(rad) * 3.4;

    99→    99→        const y2 = 12 - Math.cos(rad) * 10.2;

   101→   101→      })}

   302→   302→      // plain-JSON error path (429 rate limit / 400 validation / gateway html)
   303→   303→      if (!res.ok || !ct.includes("text/event-stream")) {

   315→   315→      }
   316→   316→      // SSE stream: tool steps arrive as they run, the answer previews live
   317→   317→      const reader = res.body?.getReader();
   318→   318→      if (!reader) throw new Error("no stream body");
   319→   319→      const dec = new TextDecoder();
   320→   320→      let buf = "";
   321→   321→      let gotTerminal = false;
   322→   322→      const stepsAcc: AgentStep[] = [];
   323→   323→      const handleEvent = (evt: Record<string, unknown>) => {
   324→   324→        if (evt.type === "step") {
   325→   325→          stepsAcc.push({ tool: String(evt.tool ?? ""), args: (evt.args as Record<string, unknown>) ?? {}, ok: evt.ok !== false });
   326→   326→          setLiveSteps([...stepsAcc]);
   327→   327→        } else if (evt.type === "status") {
   328→   328→          setLiveNote(typeof evt.note === "string" ? evt.note : null);
   329→   329→        } else if (evt.type === "delta" && typeof evt.text === "string") {
   330→   330→          streamSoFar += evt.text;
   331→   331→          setStreamText(streamSoFar);
   332→   332→        } else if (evt.type === "done" && typeof evt.answer === "string") {
   333→   333→          gotTerminal = true;
   334→   334→          const finalSteps = Array.isArray(evt.steps) ? (evt.steps as AgentStep[]) : stepsAcc;
   335→   335→          const next: AgentMsg[] = [...history, { role: "assistant", content: evt.answer, steps: finalSteps, ts: Date.now() }];
   336→   336→          persist(next);
   337→   337→          saveChat(next); // server-side history (fire-and-forget)
   338→   338→        } else if (evt.type === "error") {
   339→   339→          gotTerminal = true;
   340→   340→          persist([
   341→   341→            ...history,
   342→   342→            {
   343→   343→              role: "assistant",
   344→   344→              content: evt.message ? `${tt(T.agentError, lang)} (${evt.message})` : tt(T.agentError, lang),
   345→   345→              error: true,
   346→   346→              ts: Date.now(),
   347→   347→            },
   348→   348→          ]);
   349→   349→        }
   350→   350→      };
   351→   351→      for (;;) {
   352→   352→        const { done, value } = await reader.read();
   353→   353→        if (done) break;
   354→   354→        buf += dec.decode(value, { stream: true });
   355→   355→        let sep: number;
   356→   356→        while ((sep = buf.indexOf("\n\n")) !== -1) {
   357→   357→          const rawEvt = buf.slice(0, sep);
   358→   358→          buf = buf.slice(sep + 2);
   359→   359→          for (const line of rawEvt.split("\n")) {
   360→   360→            if (!line.startsWith("data:")) continue;
   361→   361→            try {
   362→   362→              handleEvent(JSON.parse(line.slice(5).trim()) as Record<string, unknown>);
   363→   363→            } catch {
   364→   364→              /* partial line — the next chunk completes it */
   365→   365→            }
   366→   366→          }
   367→   367→        }
   368→   368→      }
   369→   369→      if (!gotTerminal) throw new Error("stream ended without a terminal event");
   370→   370→    } catch (err) {
   371→   371→      // user pressed stop — keep whatever streamed as the (partial) answer
   372→   372→      if (err instanceof DOMException && err.name === "AbortError") {
   373→   373→        if (streamSoFar.trim()) {
   374→   374→          persist([...history, { role: "assistant", content: streamSoFar, ts: Date.now() }]);
   375→   375→        } else {
   376→   376→          persist([...history, { role: "assistant", content: tt(T.agentStopped, lang), error: true, ts: Date.now() }]);
   377→   377→        }
   378→   378→      } else {
   379→   379→        persist([...history, { role: "assistant", content: tt(T.agentError, lang), error: true, ts: Date.now() }]);
   380→   380→      }
   381→   381→    } finally {
   382→   382→      abortRef.current = null;
   383→   383→      setBusy(false);
   384→   384→      setLiveSteps([]);
   385→   385→      setLiveNote(null);
   386→   386→      setStreamText("");
   387→   387→    }
   388→   388→  };
   389→   389→

   391→   391→    abortRef.current?.abort();
   392→   392→  };
   393→   393→
   394→   394→  const retry = () => {
   395→   395→    if (lastQueryRef.current && !busy) void ask(lastQueryRef.current);

   409→   409→        id,
   410→   410→        title,
   411→   411→        messages: msgs.filter((m) => !m.error).slice(-80),

   415→   415→
   416→   416→  const loadHistory = async () => {
