   311→   311→        persist([

   322→   322→      // SSE stream: tool steps arrive as they run, the answer previews live
   323→   323→      const reader = res.body?.getReader();
   324→   324→      if (!reader) throw new Error("no stream body");
   325→   325→      const dec = new TextDecoder();
   326→   326→      let buf = "";
   327→   327→      let gotTerminal = false;
   328→   328→      const stepsAcc: AgentStep[] = [];
   329→   329→      const handleEvent = (evt: Record<string, unknown>) => {
   330→   330→        if (evt.type === "step") {
   331→   331→          stepsAcc.push({ tool: String(evt.tool ?? ""), args: (evt.args as Record<string, unknown>) ?? {}, ok: evt.ok !== false });
   332→   332→          setLiveSteps([...stepsAcc]);
   333→   333→        } else if (evt.type === "status") {
   334→   334→          setLiveNote(typeof evt.note === "string" ? evt.note : null);
   335→   335→        } else if (evt.type === "delta" && typeof evt.text === "string") {
   336→   336→          streamSoFar += evt.text;
   337→   337→          setStreamText(streamSoFar);
   338→   338→        } else if (evt.type === "done" && typeof evt.answer === "string") {
   339→   339→          gotTerminal = true;
   340→   340→          const finalSteps = Array.isArray(evt.steps) ? (evt.steps as AgentStep[]) : stepsAcc;
   341→   341→          const next: AgentMsg[] = [...history, { role: "assistant", content: evt.answer, steps: finalSteps, ts: Date.now() }];
   342→   342→          persist(next);
   343→   343→          saveChat(next); // server-side history (fire-and-forget)
   344→   344→        } else if (evt.type === "error") {
   345→   345→          gotTerminal = true;
   346→   346→          persist([
   347→   347→            ...history,
   348→   348→            {
   349→   349→              role: "assistant",
   350→   350→              content: evt.message ? `${tt(T.agentError, lang)} (${evt.message})` : tt(T.agentError, lang),
   351→   351→              error: true,
   352→   352→              ts: Date.now(),
   353→   353→            },
   354→   354→          ]);
   355→   355→        }
   356→   356→      };
   357→   357→      for (;;) {
   358→   358→        const { done, value } = await reader.read();
   359→   359→        if (done) break;
   360→   360→        buf += dec.decode(value, { stream: true });
   361→   361→        let sep: number;
   362→   362→        while ((sep = buf.indexOf("\n\n")) !== -1) {
   363→   363→          const rawEvt = buf.slice(0, sep);
   364→   364→          buf = buf.slice(sep + 2);
   365→   365→          for (const line of rawEvt.split("\n")) {
   366→   366→            if (!line.startsWith("data:")) continue;
   367→   367→            try {
   368→   368→              handleEvent(JSON.parse(line.slice(5).trim()) as Record<string, unknown>);
   369→   369→            } catch {
   370→   370→              /* partial line — the next chunk completes it */
   371→   371→            }
   372→   372→          }
   373→   373→        }
   374→   374→      }
   375→   375→      if (!gotTerminal) throw new Error("stream ended without a terminal event");
   376→   376→    } catch (err) {
   377→   377→      // user pressed stop — keep whatever streamed as the (partial) answer
   378→   378→      if (err instanceof DOMException && err.name === "AbortError") {
   379→   379→        if (streamSoFar.trim()) {
   380→   380→          persist([...history, { role: "assistant", content: streamSoFar, ts: Date.now() }]);
   381→   381→        } else {
   382→   382→          persist([...history, { role: "assistant", content: tt(T.agentStopped, lang), error: true, ts: Date.now() }]);
   383→   383→        }
   384→   384→      } else {
   385→   385→        persist([...history, { role: "assistant", content: tt(T.agentError, lang), error: true, ts: Date.now() }]);
   386→   386→      }
   387→   387→    } finally {
   388→   388→      abortRef.current = null;
   389→   389→      setBusy(false);
   390→   390→      setLiveSteps([]);
   391→   391→      setLiveNote(null);
   392→   392→      setStreamText("");
   393→   393→    }
   394→   394→  };
   395→   395→
   396→   396→  const stop = () => {
   397→   397→    abortRef.current?.abort();
   398→   398→  };
   399→   399→
   400→   400→  const retry = () => {
   401→   401→    if (lastQueryRef.current && !busy) void ask(lastQueryRef.current);
   402→   402→  };
   403→   403→
   404→   404→  // ── server-side chat history (auto-saved after every exchange) ──
   405→   405→
   406→   406→  const saveChat = (msgs: AgentMsg[]) => {
   407→   407→    const id = chatIdRef.current;
   408→   408→    if (!id || msgs.length === 0) return;
   409→   409→    const title = (msgs.find((m) => m.role === "user")?.content ?? "").slice(0, 80);
   410→   410→    void fetch("/api/agent/chats", {
   411→   411→      method: "PUT",
   412→   412→      headers: { "Content-Type": "application/json" },
   413→   413→      body: JSON.stringify({
